import re
import os
import shutil

def task_func(source_dir, target_dir, file_pattern=r'\b[A-Za-z0-9]+\.(txt|doc|docx)\b'):
    source_dir = os.path.abspath(source_dir)
    target_dir = os.path.abspath(target_dir)

    if not os.path.isdir(source_dir):
        raise NotADirectoryError(f"Source directory does not exist or is not a directory: {source_dir}")

    os.makedirs(target_dir, exist_ok=True)

    pattern = re.compile(file_pattern)

    matched_paths = []
    seen = set()

    for root, _, files in os.walk(source_dir):
        for filename in files:
            full_path = os.path.normpath(os.path.join(root, filename))
            # Normalize separators for regexes that expect Windows-style paths.
            regex_path = full_path.replace("/", "\\")
            if pattern.search(regex_path):
                if full_path not in seen:
                    seen.add(full_path)
                    matched_paths.append(full_path)

    config_path = os.path.join(target_dir, "matched_files.cfg")
    with open(config_path, "w", encoding="utf-8") as config_file:
        if matched_paths:
            config_file.write("\n".join(matched_paths))
            config_file.write("\n")

    return config_path
