import pytz
import numpy as np
from dateutil.parser import parse
import math
from datetime import datetime, timedelta

MOON_PHASES_YEARS = np.array([1987, 1994, 2001, 2008, 2015, 2022])


def _resolve_timezone(tz_name):
    try:
        return pytz.timezone(tz_name)
    except Exception as exc:
        raise ValueError(f"Invalid timezone: {tz_name!r}") from exc


def _cycle_reference_year(year):
    base_year = int(MOON_PHASES_YEARS[0])
    cycle_span = int(MOON_PHASES_YEARS[1] - MOON_PHASES_YEARS[0]) if len(MOON_PHASES_YEARS) > 1 else 7
    return int(base_year + cycle_span * math.floor((year - base_year) / cycle_span))


def task_func(date_str, from_tz, to_tz):
    from_tz_obj = _resolve_timezone(from_tz)
    to_tz_obj = _resolve_timezone(to_tz)

    try:
        dt = parse(date_str)
    except Exception as exc:
        raise ValueError(f"Invalid date string: {date_str!r}") from exc

    if dt.tzinfo is None:
        try:
            dt = from_tz_obj.localize(dt, is_dst=None)
        except (pytz.AmbiguousTimeError, pytz.NonExistentTimeError) as exc:
            raise ValueError(f"Cannot localize naive datetime {date_str!r} in timezone {from_tz!r}") from exc

    dt = dt.astimezone(to_tz_obj)

    cycle_year = _cycle_reference_year(dt.year)

    try:
        reference_start = to_tz_obj.localize(datetime(cycle_year, 1, 1, 0, 0, 0), is_dst=None)
    except (pytz.AmbiguousTimeError, pytz.NonExistentTimeError):
        reference_start = to_tz_obj.localize(datetime(cycle_year, 1, 1, 12, 0, 0), is_dst=None) - timedelta(hours=12)

    elapsed_days = (dt - reference_start).total_seconds() / 86400.0

    synodic_month = 29.53058867  # average lunar synodic month in days
    lunar_position = (elapsed_days % synodic_month) / synodic_month

    # 0.0 -> new moon, 1.0 -> full moon
    phase = 0.5 * (1.0 - math.cos(2.0 * math.pi * lunar_position))

    return float(max(0.0, min(1.0, phase)))
