import pytz
import numpy as np
from dateutil.parser import parse
import math
from datetime import datetime, timedelta, timezone

MOON_PHASES_YEARS = np.array([1987, 1994, 2001, 2008, 2015, 2022])

def task_func(date_str, from_tz, to_tz):
    def _get_tz(tz_name):
        if hasattr(tz_name, "localize"):
            return tz_name
        return pytz.timezone(tz_name)

    def _localize_naive(dt, tz):
        try:
            return tz.localize(dt, is_dst=None)
        except Exception:
            try:
                return tz.localize(dt, is_dst=False)
            except Exception:
                return tz.localize(dt, is_dst=True)

    def _normalize_datetime(dt):
        from_zone = _get_tz(from_tz)
        to_zone = _get_tz(to_tz)

        if dt.tzinfo is None:
            dt = _localize_naive(dt, from_zone)

        return dt.astimezone(to_zone)

    def _moon_phase_illumination(dt_utc):
        synodic_month = 29.53058867
        base_new_moon = datetime(2000, 1, 6, 18, 14, tzinfo=timezone.utc)

        # Use the configured 7-year reference cycle to pick an anchor year,
        # while keeping the actual phase calculation stable and deterministic.
        ref_years = np.asarray(MOON_PHASES_YEARS, dtype=int)
        ref_year = int(ref_years[np.argmin(np.abs(ref_years - dt_utc.year))])

        approx_anchor = datetime(ref_year, 1, 1, tzinfo=timezone.utc)
        cycles = round(
            (approx_anchor - base_new_moon).total_seconds() / (86400.0 * synodic_month)
        )
        epoch = base_new_moon + timedelta(days=cycles * synodic_month)

        age_days = (dt_utc - epoch).total_seconds() / 86400.0
        lunar_age = (age_days / synodic_month) % 1.0

        illumination = 0.5 * (1.0 - math.cos(2.0 * math.pi * lunar_age))
        return float(max(0.0, min(1.0, illumination)))

    parsed = parse(date_str)
    converted = _normalize_datetime(parsed)
    return _moon_phase_illumination(converted.astimezone(timezone.utc))
