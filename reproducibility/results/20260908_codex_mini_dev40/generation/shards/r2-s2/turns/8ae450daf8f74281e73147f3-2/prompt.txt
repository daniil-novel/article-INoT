import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(s1, s2):
    s1 = pd.Series(s1).dropna()
    s2 = pd.Series(s2).dropna()

    common_values = set(s1.tolist()).intersection(set(s2.tolist()))
    intersection_count = len(common_values)

    df = pd.DataFrame(
        {
            "series": ["s1"] * len(s1) + ["s2"] * len(s2),
            "value": pd.concat([s1.reset_index(drop=True), s2.reset_index(drop=True)], ignore_index=True),
        }
    )

    fig, ax = plt.subplots()
    sns.swarmplot(data=df, x="series", y="value", order=["s1", "s2"], ax=ax)

    for value in sorted(common_values):
        ax.axhline(y=value, color="red", linestyle="--", linewidth=1, alpha=0.7)

    return ax, intersection_count
