import re
import os
import shutil


def task_func(source_dir, target_dir, file_pattern=r'\b[A-Za-z0-9]+\.(txt|doc|docx)\b'):
    """
    Recursively search source_dir for files whose full path matches the requested
    AcroTray.exe pattern and write matching paths to a configuration file.

    Returns:
        str: Path to the created configuration file.
    """
    default_pattern = r'\b[A-Za-z0-9]+\.(txt|doc|docx)\b'
    task_pattern = r'(?<!Distillr)\\AcroTray\.exe'

    if not os.path.exists(source_dir):
        raise FileNotFoundError(f"Source directory does not exist: {source_dir}")
    if not os.path.isdir(source_dir):
        raise NotADirectoryError(f"Source path is not a directory: {source_dir}")

    if os.path.exists(target_dir) and not os.path.isdir(target_dir):
        raise NotADirectoryError(f"Target path is not a directory: {target_dir}")
    os.makedirs(target_dir, exist_ok=True)

    pattern_to_use = task_pattern if file_pattern == default_pattern else file_pattern
    try:
        regex = re.compile(pattern_to_use)
    except re.error:
        # Let invalid regex patterns fail clearly.
        raise

    matched_paths = []
    seen = set()

    for root, _, files in os.walk(source_dir):
        for filename in files:
            full_path = os.path.join(root, filename)
            normalized_path = full_path.replace("/", "\\")
            if regex.search(normalized_path) and normalized_path not in seen:
                seen.add(normalized_path)
                matched_paths.append(normalized_path)

    config_path = os.path.join(target_dir, "matched_files.cfg")
    with open(config_path, "w", encoding="utf-8") as config_file:
        for path in matched_paths:
            config_file.write(path + os.linesep)

    return config_path
