import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(s1, s2):
    s1 = pd.Series(s1).dropna()
    s2 = pd.Series(s2).dropna()

    s1 = pd.to_numeric(s1, errors="coerce").dropna()
    s2 = pd.to_numeric(s2, errors="coerce").dropna()

    intersection = set(s1.unique()) & set(s2.unique())
    intersection_count = len(intersection)

    combined = pd.DataFrame(
        {
            "series": ["s1"] * len(s1) + ["s2"] * len(s2),
            "value": pd.concat([s1, s2], ignore_index=True),
        }
    )

    fig, ax = plt.subplots()

    if not combined.empty:
        sns.swarmplot(
            data=combined,
            x="series",
            y="value",
            order=["s1", "s2"],
            ax=ax,
            size=6,
        )

        for val in sorted(intersection):
            ax.axhline(
                y=val,
                color="red",
                linestyle="--",
                linewidth=1.2,
                alpha=0.75,
                zorder=0,
            )

    ax.set_xlabel("Series")
    ax.set_ylabel("Value")
    ax.set_title("Swarm plot comparison of s1 and s2")
    ax.set_xticks([0, 1])
    ax.set_xticklabels(["s1", "s2"])

    return ax, intersection_count
