import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame):
    # Keep only numeric columns
    numeric_df = df.select_dtypes(include=[np.number]).copy()

    # Prepare output DataFrame with the required columns
    result = pd.DataFrame(
        np.nan,
        index=df.index,
        columns=["principal component 1", "principal component 2"],
        dtype=float,
    )

    # Create the plot axes
    fig, ax = plt.subplots()

    # Set required axis labels
    ax.set_xlabel("principal component")
    ax.set_ylabel("principal component 2")

    # If there is no numeric data, return empty PCA result with axes
    if numeric_df.shape[0] == 0 or numeric_df.shape[1] == 0:
        return result, ax

    # Replace infinite values with NaN so they are treated as missing
    numeric_df = numeric_df.replace([np.inf, -np.inf], np.nan)

    # Impute missing values with column means
    means = numeric_df.mean(axis=0, skipna=True)
    numeric_df = numeric_df.fillna(means)

    # If a column is still all-NaN after mean imputation, fill with 0.0
    numeric_df = numeric_df.fillna(0.0)

    # Determine how many components can be computed safely
    n_components = min(2, numeric_df.shape[0], numeric_df.shape[1])
    if n_components <= 0:
        return result, ax

    # Run PCA
    try:
        pca = PCA(n_components=n_components)
        transformed = pca.fit_transform(numeric_df.to_numpy())
    except Exception:
        # Graceful fallback if PCA cannot be computed
        return result, ax

    # Store the available principal components
    result.iloc[:, :n_components] = transformed

    # Draw scatter plot of the first two components when both are available
    valid_mask = result["principal component 1"].notna() & result["principal component 2"].notna()
    if valid_mask.any():
        ax.scatter(
            result.loc[valid_mask, "principal component 1"],
            result.loc[valid_mask, "principal component 2"],
        )

    return result, ax
