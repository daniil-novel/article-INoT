import pandas as pd
import numpy as np

def task_func(file_path="data.csv", columns=["A", "B", "C"]):
    import matplotlib.pyplot as plt

    if columns is None:
        raise ValueError("columns must contain at least one column name.")
    if isinstance(columns, str):
        columns = [columns]
    else:
        columns = list(columns)

    if not columns:
        raise ValueError("columns must contain at least one column name.")

    df = pd.read_csv(file_path)

    def _convert_numeric_values(value):
        if pd.isna(value) or isinstance(value, bool):
            return value
        if isinstance(value, (int, float, np.integer, np.floating)):
            return float(value)
        try:
            text = str(value).strip()
            if text == "":
                return value
            return float(text)
        except (TypeError, ValueError):
            return value

    df = df.apply(lambda col: col.map(_convert_numeric_values))

    valid_columns = [col for col in columns if col in df.columns]
    if not valid_columns:
        raise ValueError("None of the requested columns were found in the CSV file.")

    fig, ax = plt.subplots()

    plotted_columns = []
    numeric_cache = {}

    for col in valid_columns:
        numeric_series = pd.to_numeric(df[col], errors="coerce").astype(float)
        numeric_cache[col] = numeric_series
        if numeric_series.notna().any():
            ax.plot(df.index, numeric_series, label=col)
            plotted_columns.append(col)

    if not plotted_columns:
        raise ValueError("No numeric data found in the requested columns.")

    ax.set_xlabel("Index")
    ax.set_ylabel("Value")
    ax.set_title("Line chart of selected columns")
    ax.legend()
    fig.tight_layout()

    target_col = plotted_columns[0]
    cube_root_series = pd.Series(
        np.cbrt(numeric_cache[target_col].to_numpy(dtype=float)),
        index=df.index,
        name=f"{target_col}_cube_root",
    )

    return df, ax, cube_root_series
