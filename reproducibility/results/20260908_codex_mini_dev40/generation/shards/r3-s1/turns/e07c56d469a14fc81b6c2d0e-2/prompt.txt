import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    import os

    # Validate inputs
    try:
        mean = float(mean)
        std_dev = float(std_dev)
    except (TypeError, ValueError):
        raise ValueError("mean and std_dev must be numeric values.")

    if not np.isfinite(mean) or not np.isfinite(std_dev):
        raise ValueError("mean and std_dev must be finite values.")

    if std_dev <= 0:
        raise ValueError("std_dev must be greater than 0.")

    # Generate sample
    rng = np.random.default_rng()
    sample_size = 1000
    sample = rng.normal(loc=mean, scale=std_dev, size=sample_size)

    # Analyze distribution
    skewness = float(stats.skew(sample, bias=False))
    kurtosis = float(stats.kurtosis(sample, fisher=True, bias=False))

    saved_paths = []

    if save_plots:
        # Histogram
        hist_path = os.path.abspath("histogram.png")
        fig1, ax1 = plt.subplots(figsize=(8, 5))
        ax1.hist(sample, bins=30, edgecolor="black", alpha=0.75)
        ax1.set_title("Histogram of Sample")
        ax1.set_xlabel("Value")
        ax1.set_ylabel("Frequency")
        ax1.grid(True, alpha=0.3)
        fig1.tight_layout()
        fig1.savefig(hist_path, dpi=150)
        plt.close(fig1)

        # QQ plot
        qq_path = os.path.abspath("qq_plot.png")
        fig2, ax2 = plt.subplots(figsize=(8, 5))
        stats.probplot(sample, dist="norm", plot=ax2)
        ax2.set_title("QQ Plot of Sample")
        ax2.grid(True, alpha=0.3)
        fig2.tight_layout()
        fig2.savefig(qq_path, dpi=150)
        plt.close(fig2)

        saved_paths = [hist_path, qq_path]

    return skewness, kurtosis, saved_paths
