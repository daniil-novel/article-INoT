import subprocess
import os
import shutil
import sys

# Constants
DIRECTORY = r'c:\Program Files\VMware\VMware Server'
BACKUP_DIRECTORY = r'c:\Program Files\VMware\VMware Server\Backup'


def task_func(filename):
    if filename is None:
        return -1

    filename = str(filename).strip()
    if not filename:
        return -1

    # Reject path traversal and path separators; only allow a plain file name.
    if os.path.basename(filename) != filename:
        return -1
    if any(sep and sep in filename for sep in (os.sep, os.altsep)):
        return -1

    source_path = os.path.join(DIRECTORY, filename)
    backup_path = os.path.join(BACKUP_DIRECTORY, filename)

    try:
        if not os.path.isfile(source_path):
            return -1

        os.makedirs(BACKUP_DIRECTORY, exist_ok=True)

        shutil.copy2(source_path, backup_path)

        completed = subprocess.run(
            [backup_path],
            cwd=BACKUP_DIRECTORY,
            check=False
        )
        return completed.returncode

    except (OSError, shutil.SameFileError, ValueError):
        return -1
