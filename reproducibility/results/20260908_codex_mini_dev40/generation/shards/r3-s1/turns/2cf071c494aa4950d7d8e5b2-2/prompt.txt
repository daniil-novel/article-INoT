import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

import os
import tempfile
import uuid


def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    """
    Generate a random sample from a normal distribution, compute skewness and
    kurtosis, and optionally save a histogram and QQ plot.

    Returns:
        tuple: (skewness: float, kurtosis: float, plot_paths: list)
    """
    # Validate inputs
    for name, value in ("mean", mean), ("std_dev", std_dev):
        if not np.isscalar(value) or isinstance(value, bool) or not np.isfinite(value):
            raise ValueError(f"{name} must be a finite numeric scalar.")

    mean = float(mean)
    std_dev = float(std_dev)

    if std_dev <= 0:
        raise ValueError("std_dev must be greater than 0.")

    # Generate random sample
    sample_size = 1000
    sample = np.random.normal(loc=mean, scale=std_dev, size=sample_size)

    # Compute skewness and kurtosis (Fisher kurtosis: normal distribution ~= 0)
    skewness = float(stats.skew(sample, bias=False))
    kurtosis = float(stats.kurtosis(sample, fisher=True, bias=False))

    plot_paths = []

    if save_plots:
        output_dir = tempfile.gettempdir()
        unique_id = uuid.uuid4().hex

        # Histogram
        hist_path = os.path.join(output_dir, f"histogram_{unique_id}.png")
        try:
            fig, ax = plt.subplots(figsize=(8, 5))
            ax.hist(sample, bins=30, density=True, alpha=0.75, color="steelblue", edgecolor="black")
            ax.set_title("Histogram of Normal Sample")
            ax.set_xlabel("Value")
            ax.set_ylabel("Density")
            ax.grid(True, alpha=0.3)
            fig.tight_layout()
            fig.savefig(hist_path, dpi=150)
            plot_paths.append(hist_path)
        finally:
            plt.close(fig)

        # QQ plot
        qq_path = os.path.join(output_dir, f"qq_plot_{unique_id}.png")
        try:
            fig, ax = plt.subplots(figsize=(8, 5))
            stats.probplot(sample, dist="norm", plot=ax)
            ax.set_title("QQ Plot of Normal Sample")
            fig.tight_layout()
            fig.savefig(qq_path, dpi=150)
            plot_paths.append(qq_path)
        finally:
            plt.close(fig)

    return skewness, kurtosis, plot_paths
