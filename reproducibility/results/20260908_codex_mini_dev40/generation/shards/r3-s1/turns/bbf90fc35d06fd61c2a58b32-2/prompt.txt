```python
import subprocess
import os
import shutil
import sys
# Constants
DIRECTORY = 'c:\Program Files\VMware\VMware Server'
BACKUP_DIRECTORY = 'c:\Program Files\VMware\VMware Server\Backup'
def task_func(filename):
    if not filename:
        return -1

    safe_name = os.path.basename(str(filename).strip())
    if not safe_name or safe_name in ('.', '..'):
        return -1

    source_path = os.path.join(DIRECTORY, safe_name)
    backup_path = os.path.join(BACKUP_DIRECTORY, safe_name)

    try:
        if not os.path.isfile(source_path):
            return -1

        os.makedirs(BACKUP_DIRECTORY, exist_ok=True)
        shutil.copy2(source_path, backup_path)

        ext = os.path.splitext(safe_name)[1].lower()
        if ext in ('.py', '.pyw'):
            cmd = [sys.executable, backup_path]
        else:
            cmd = [backup_path]

        completed = subprocess.run(cmd, check=False)
        return completed.returncode
    except Exception:
        return -1
```
