import pandas as pd
import json

def task_func(file_path, key):
    with open(file_path, "r", encoding="utf-8") as f:
        data = json.load(f)

    if isinstance(data, dict):
        records = [data]
    elif isinstance(data, list):
        records = data
    else:
        raise ValueError("Input JSON must be a list of objects or a single object.")

    cleaned_records = []
    for record in records:
        if not isinstance(record, dict):
            raise ValueError("All records in the JSON array must be objects.")
        cleaned_record = record.copy()
        cleaned_record.pop(key, None)
        cleaned_records.append(cleaned_record)

    df = pd.DataFrame(cleaned_records)
    df.to_json(file_path, orient="records", indent=4, force_ascii=False)

    return df
