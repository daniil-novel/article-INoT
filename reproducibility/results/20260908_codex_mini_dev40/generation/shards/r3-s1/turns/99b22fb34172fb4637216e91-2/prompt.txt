```python
import os
import subprocess
import random

# Constants
SCRIPTS = ['script1.sh', 'script2.sh', 'script3.sh']
SCRIPTS_DIR = '/path/to/scripts'

def task_func():
    if not SCRIPTS:
        raise ValueError("SCRIPTS must not be empty")

    selected_script = random.choice(SCRIPTS)
    script_path = os.path.abspath(os.path.join(SCRIPTS_DIR, selected_script))

    subprocess.run(["bash", script_path], check=True)

    return {"script": script_path}
```
