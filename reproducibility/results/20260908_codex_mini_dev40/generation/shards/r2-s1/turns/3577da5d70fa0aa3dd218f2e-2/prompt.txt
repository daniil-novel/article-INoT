import subprocess
import os
import shutil
import sys

# Constants
DIRECTORY = 'c:\\Program Files\\VMware\\VMware Server'
BACKUP_DIRECTORY = 'c:\\Program Files\\VMware\\VMware Server\\Backup'


def task_func(filename):
    if not filename:
        return -1

    base_name = os.path.basename(filename)
    if not base_name:
        return -1

    source_path = os.path.join(DIRECTORY, base_name)
    backup_path = os.path.join(BACKUP_DIRECTORY, base_name)

    if not os.path.isfile(source_path):
        return -1

    try:
        os.makedirs(BACKUP_DIRECTORY, exist_ok=True)
        shutil.copy2(source_path, backup_path)
    except (OSError, IOError):
        return -1

    try:
        if backup_path.lower().endswith('.py'):
            completed = subprocess.run([sys.executable, backup_path], check=False)
        else:
            completed = subprocess.run([backup_path], check=False)
        return completed.returncode
    except (OSError, subprocess.SubprocessError):
        return -1
