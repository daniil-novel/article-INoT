import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

from pathlib import Path
from uuid import uuid4


def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    def _is_finite_number(value):
        return isinstance(value, (int, float, np.number)) and not isinstance(value, bool) and np.isfinite(value)

    if not _is_finite_number(mean):
        raise ValueError("mean must be a finite numeric value")
    if not _is_finite_number(std_dev):
        raise ValueError("std_dev must be a finite numeric value")
    if std_dev <= 0:
        raise ValueError("std_dev must be greater than 0")

    rng = np.random.default_rng()
    sample = rng.normal(loc=float(mean), scale=float(std_dev), size=1000)

    skewness = stats.skew(sample, bias=False)
    kurtosis = stats.kurtosis(sample, fisher=True, bias=False)

    plot_paths = []
    if save_plots:
        unique_id = uuid4().hex
        hist_path = Path.cwd() / f"histogram_{unique_id}.png"
        qq_path = Path.cwd() / f"qq_plot_{unique_id}.png"

        fig, ax = plt.subplots(figsize=(8, 6))
        ax.hist(sample, bins="auto", density=True, alpha=0.75, color="steelblue", edgecolor="black")
        ax.set_title("Histogram of Normal Sample")
        ax.set_xlabel("Value")
        ax.set_ylabel("Density")
        fig.tight_layout()
        fig.savefig(hist_path, dpi=300)
        plt.close(fig)
        plot_paths.append(str(hist_path))

        fig, ax = plt.subplots(figsize=(8, 6))
        stats.probplot(sample, dist="norm", plot=ax)
        ax.set_title("QQ Plot of Normal Sample")
        fig.tight_layout()
        fig.savefig(qq_path, dpi=300)
        plt.close(fig)
        plot_paths.append(str(qq_path))

    return float(skewness), float(kurtosis), plot_paths
