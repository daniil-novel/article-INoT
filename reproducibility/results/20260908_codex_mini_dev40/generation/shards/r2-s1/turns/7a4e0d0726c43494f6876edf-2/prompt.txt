import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    """
    Generate a random sample from a normal distribution, compute skewness and kurtosis,
    and optionally save a histogram and QQ plot.

    Returns:
        tuple: (skewness: float, kurtosis: float, plot_paths: list[str])
    """
    # Validate inputs
    if not np.isscalar(mean) or isinstance(mean, bool):
        raise TypeError("mean must be a numeric scalar.")
    if not np.isscalar(std_dev) or isinstance(std_dev, bool):
        raise TypeError("std_dev must be a numeric scalar.")

    mean = float(mean)
    std_dev = float(std_dev)

    if std_dev <= 0:
        raise ValueError("std_dev must be a positive number.")

    # Generate a random normal sample
    sample_size = 5000
    sample = np.random.normal(loc=mean, scale=std_dev, size=sample_size)

    # Compute skewness and kurtosis
    skewness = float(stats.skew(sample))
    # Use Fisher's definition (excess kurtosis): normal distribution => 0
    kurt = float(stats.kurtosis(sample, fisher=True))

    plot_paths = []

    if save_plots:
        # Histogram
        hist_path = "histogram.png"
        plt.figure(figsize=(8, 6))
        plt.hist(sample, bins="auto", edgecolor="black", alpha=0.75)
        plt.title("Histogram of Normal Sample")
        plt.xlabel("Value")
        plt.ylabel("Frequency")
        plt.tight_layout()
        plt.savefig(hist_path, dpi=150)
        plt.close()
        plot_paths.append(hist_path)

        # QQ plot
        qq_path = "qq_plot.png"
        plt.figure(figsize=(8, 6))
        stats.probplot(sample, dist="norm", plot=plt)
        plt.title("QQ Plot of Normal Sample")
        plt.tight_layout()
        plt.savefig(qq_path, dpi=150)
        plt.close()
        plot_paths.append(qq_path)

    return skewness, kurt, plot_paths
