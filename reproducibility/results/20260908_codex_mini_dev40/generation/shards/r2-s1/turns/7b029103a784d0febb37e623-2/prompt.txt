import nltk
# Download necessary NLTK data (if not already present)
nltk.download('punkt')
nltk.download('averaged_perceptron_tagger')
from collections import Counter

def task_func(content):
    if not isinstance(content, str):
        return {}

    tokens = nltk.word_tokenize(content)
    if not tokens:
        return {}

    # Remove the last word-like token, keeping trailing punctuation if present
    last_word_idx = None
    for i in range(len(tokens) - 1, -1, -1):
        if any(ch.isalnum() for ch in tokens[i]):
            last_word_idx = i
            break

    if last_word_idx is None:
        return {}

    remaining_tokens = tokens[:last_word_idx] + tokens[last_word_idx + 1:]
    if not remaining_tokens:
        return {}

    tagged_tokens = nltk.pos_tag(remaining_tokens)
    return dict(Counter(tag for _, tag in tagged_tokens))
