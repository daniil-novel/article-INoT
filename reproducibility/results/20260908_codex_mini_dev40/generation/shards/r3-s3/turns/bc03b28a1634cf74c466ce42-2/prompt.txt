import pandas as pd
import json
import os
import shutil

def task_func(path):
    os.makedirs(path, exist_ok=True)
    processed_dir = os.path.join(path, "processed")
    os.makedirs(processed_dir, exist_ok=True)

    frames = []

    json_files = sorted(
        f for f in os.listdir(path)
        if f.lower().endswith(".json") and os.path.isfile(os.path.join(path, f))
    )

    for filename in json_files:
        file_path = os.path.join(path, filename)

        with open(file_path, "r", encoding="utf-8") as f:
            data = json.load(f)

        if isinstance(data, list):
            df_file = pd.DataFrame(data)
        elif isinstance(data, dict):
            df_file = pd.DataFrame([data])
        else:
            df_file = pd.DataFrame({"value": [data]})

        df_file["Source"] = filename
        frames.append(df_file)

        target_path = os.path.join(processed_dir, filename)
        if os.path.exists(target_path):
            os.remove(target_path)
        shutil.move(file_path, target_path)

    if frames:
        df = pd.concat(frames, ignore_index=True)
    else:
        df = pd.DataFrame()

    return df
