import numpy as np
import datetime

def task_func(arr):
    a = np.asarray(arr)
    out = np.empty(a.shape, dtype=object)

    def reverse_words(x):
        if isinstance(x, str):
            words = x.split()
            return " ".join(words[::-1])
        return x

    for idx in np.ndindex(a.shape):
        out[idx] = reverse_words(a[idx])

    return out
