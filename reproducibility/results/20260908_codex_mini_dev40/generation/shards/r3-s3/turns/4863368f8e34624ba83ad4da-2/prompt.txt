import pandas as pd
import json
import os
import shutil

def task_func(path):
    os.makedirs(path, exist_ok=True)
    processed_dir = os.path.join(path, "processed")
    os.makedirs(processed_dir, exist_ok=True)

    frames = []
    json_files = sorted(
        filename
        for filename in os.listdir(path)
        if filename.lower().endswith(".json")
        and os.path.isfile(os.path.join(path, filename))
    )

    for filename in json_files:
        file_path = os.path.join(path, filename)
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                data = json.load(f)

            try:
                if isinstance(data, (dict, list)):
                    frame = pd.json_normalize(data)
                else:
                    frame = pd.DataFrame([data])
            except Exception:
                frame = pd.DataFrame(data if isinstance(data, list) else [data])

            dest_path = os.path.join(processed_dir, filename)
            if os.path.exists(dest_path):
                if os.path.isfile(dest_path) or os.path.islink(dest_path):
                    os.remove(dest_path)
                else:
                    continue

            shutil.move(file_path, dest_path)

            frame = frame.copy()
            frame["Source"] = filename
            frame = frame[["Source"] + [c for c in frame.columns if c != "Source"]]
            frames.append(frame)
        except (json.JSONDecodeError, OSError, ValueError):
            continue

    if frames:
        df = pd.concat(frames, ignore_index=True, sort=False)
    else:
        df = pd.DataFrame(columns=["Source"])

    return df
