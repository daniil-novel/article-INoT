import numpy as np
import matplotlib.pyplot as plt
import cmath

def task_func(x, y):
    if not isinstance(x, np.ndarray) or not isinstance(y, np.ndarray):
        raise TypeError("x and y must be numpy.ndarray")

    if len(x) != len(y):
        raise ValueError("x and y must have the same length")

    x_vals = np.ravel(x)
    y_vals = np.ravel(y)

    X, Y = np.meshgrid(x_vals, y_vals)
    phase_values = np.angle(X + 1j * Y)

    fig, ax = plt.subplots()
    mesh = ax.pcolormesh(X, Y, phase_values, shading="auto", cmap="twilight")
    fig.colorbar(mesh, ax=ax, label="Phase (radians)")
    ax.set_xlabel("x")
    ax.set_ylabel("y")
    ax.set_title("Phase of complex function")

    return ax, phase_values
