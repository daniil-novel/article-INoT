import numpy as np
import secrets
import hashlib
import base64

def task_func(num, from_base, to_base, alphabet):
    DIGITS = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
    STD_B64 = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    # Validate bases
    if not isinstance(from_base, int) or not isinstance(to_base, int):
        raise ValueError("from_base and to_base must be integers")
    if from_base < 2 or to_base < 2:
        raise ValueError("Invalid base for conversion")
    if from_base > len(DIGITS) or to_base > len(DIGITS):
        raise ValueError("Base too large for supported conversion")

    # Validate custom alphabet for base64
    if not isinstance(alphabet, str) or len(alphabet) != 64 or len(set(alphabet)) != 64:
        raise ValueError("alphabet must be a 64-character string with unique characters")

    if not isinstance(num, str) or num == "":
        raise ValueError("Invalid number format for conversion")

    # Handle optional sign
    sign = 1
    s = num
    if s[0] in "+-":
        if s[0] == "-":
            sign = -1
        s = s[1:]
    if s == "":
        raise ValueError("Invalid number format for conversion")

    # Validate and parse input number in from_base
    if from_base <= 36:
        valid = set(DIGITS[:from_base])
        value = 0
        for ch in s:
            ch_u = ch.upper()
            if ch_u not in valid:
                raise ValueError("Invalid number format for conversion")
            value = value * from_base + DIGITS.index(ch_u)
    else:
        valid = set(DIGITS[:from_base])
        value = 0
        for ch in s:
            if ch not in valid:
                raise ValueError("Invalid number format for conversion")
            value = value * from_base + DIGITS.index(ch)

    value *= sign

    # Convert integer to target base
    def int_to_base(n, base):
        if n == 0:
            return "0"
        negative = n < 0
        n = abs(n)
        out = []
        while n > 0:
            out.append(DIGITS[n % base])
            n //= base
        result = "".join(reversed(out))
        return "-" + result if negative else result

    converted = int_to_base(value, to_base)

    # Add random salt and hash the result
    salt = secrets.token_hex(16)
    payload = f"{converted}:{salt}".encode("utf-8")
    digest = hashlib.sha256(payload).digest()

    # Base64 encode with custom alphabet
    trans = bytes.maketrans(STD_B64, alphabet.encode("utf-8"))
    encoded = base64.b64encode(digest).translate(trans).decode("ascii")

    return encoded, salt
