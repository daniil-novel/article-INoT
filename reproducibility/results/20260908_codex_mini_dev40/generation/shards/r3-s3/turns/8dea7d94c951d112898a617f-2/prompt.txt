import urllib.request
import zipfile
import os
import urllib.error


def task_func(
    url: str,
    save_path: str = "downloaded_file.zip",
    extract_path: str = "extracted_files",
) -> str:
    error_message = None

    try:
        try:
            os.makedirs(extract_path, exist_ok=True)
        except OSError as e:
            return f"Error: File I/O issue - {e}"

        try:
            with urllib.request.urlopen(url) as response, open(save_path, "wb") as out_file:
                while True:
                    chunk = response.read(8192)
                    if not chunk:
                        break
                    out_file.write(chunk)
        except urllib.error.URLError as e:
            reason = getattr(e, "reason", e)
            error_message = f"URL Error: {reason}"
        except OSError as e:
            error_message = f"Error: File I/O issue - {e}"

        if error_message is None:
            try:
                with zipfile.ZipFile(save_path, "r") as zip_ref:
                    zip_ref.extractall(extract_path)
            except zipfile.BadZipFile:
                error_message = "Error: Corrupted or invalid ZIP file"
            except OSError as e:
                error_message = f"Error: File I/O issue - {e}"

    finally:
        if os.path.exists(save_path):
            try:
                os.remove(save_path)
            except OSError as e:
                if error_message is None:
                    error_message = f"Error: Failed to delete ZIP file - {e}"

    return error_message if error_message is not None else extract_path
