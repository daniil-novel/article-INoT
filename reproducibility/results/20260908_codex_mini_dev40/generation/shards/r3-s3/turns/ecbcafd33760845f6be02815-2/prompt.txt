import urllib.request
import zipfile
import os
import urllib.error

def task_func(
    url: str,
    save_path: str = "downloaded_file.zip",
    extract_path: str = "extracted_files",
) -> str:
    error_message = None
    cleanup_error = None

    def _safe_extract(zf: zipfile.ZipFile, destination: str) -> None:
        abs_destination = os.path.abspath(destination)
        for member in zf.infolist():
            target_path = os.path.abspath(os.path.join(destination, member.filename))
            try:
                common = os.path.commonpath([abs_destination, target_path])
            except ValueError as exc:
                raise OSError(f"Unsafe ZIP entry detected: {member.filename}") from exc
            if common != abs_destination:
                raise OSError(f"Unsafe ZIP entry detected: {member.filename}")
        zf.extractall(destination)

    try:
        os.makedirs(extract_path, exist_ok=True)

        with urllib.request.urlopen(url) as response, open(save_path, "wb") as out_file:
            while True:
                chunk = response.read(8192)
                if not chunk:
                    break
                out_file.write(chunk)

        with zipfile.ZipFile(save_path, "r") as zf:
            _safe_extract(zf, extract_path)

    except urllib.error.URLError as exc:
        reason = getattr(exc, "reason", exc)
        error_message = f"URL Error: {reason}"
    except zipfile.BadZipFile:
        error_message = "ZIP Error: Corrupted or invalid ZIP file."
    except OSError as exc:
        error_message = f"File Error: {exc}"
    except Exception as exc:
        error_message = f"Error: {exc}"
    finally:
        if os.path.exists(save_path):
            try:
                os.remove(save_path)
            except OSError as exc:
                cleanup_error = f"File Error: Failed to delete temporary ZIP file: {exc}"

    if error_message is not None:
        return error_message
    if cleanup_error is not None:
        return cleanup_error
    return extract_path
