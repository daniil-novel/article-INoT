import numpy as np
import secrets
import hashlib
import base64

def task_func(num, from_base, to_base, alphabet):
    """
    Convert `num` from `from_base` to `to_base`, hash the converted value with a random salt,
    and encode the SHA-256 digest using base64 with a custom alphabet.

    Returns:
        tuple: (custom_base64_encoded_hash, salt)
    """
    if from_base < 2 or to_base < 2:
        raise ValueError("from_base and to_base must be at least 2")

    if not isinstance(num, str):
        raise ValueError("num must be a string")

    if not isinstance(alphabet, str) or len(alphabet) != 64 or len(set(alphabet)) != 64:
        raise ValueError("alphabet must be a 64-character string with unique characters")

    # Digits for conversion: supports bases up to 94 using printable ASCII characters.
    digits = "".join(chr(i) for i in range(33, 127))
    if from_base > len(digits) or to_base > len(digits):
        raise ValueError("base exceeds supported range")

    # For common bases (<= 36), accept case-insensitive input like hex "ff".
    working_num = num.upper() if from_base <= 36 else num

    digit_to_val = {ch: idx for idx, ch in enumerate(digits)}

    if working_num == "":
        raise ValueError("num must not be empty")

    # Parse input number in from_base with validation.
    value = 0
    for ch in working_num:
        if ch not in digit_to_val or digit_to_val[ch] >= from_base:
            raise ValueError("num contains characters not valid in the specified from_base")
        value = value * from_base + digit_to_val[ch]

    # Convert integer to to_base.
    if value == 0:
        converted = digits[0]
    else:
        out = []
        while value > 0:
            value, rem = divmod(value, to_base)
            out.append(digits[rem])
        converted = "".join(reversed(out))

    # Random salt ensures different results across invocations.
    salt = secrets.token_hex(16)

    # Hash the converted representation plus salt.
    payload = f"{converted}:{salt}".encode("utf-8")
    digest = hashlib.sha256(payload).digest()

    # Standard base64 -> custom alphabet translation.
    standard_alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    b64_text = base64.b64encode(digest).decode("ascii")
    custom_b64 = b64_text.translate(str.maketrans(standard_alphabet, alphabet))

    return custom_b64, salt
