import numpy as np
import random
from sklearn.preprocessing import MinMaxScaler

def task_func(list_of_lists, seed=42):
    if not list_of_lists:
        return []

    rng = random.Random(seed)

    filled_rows = []
    row_lengths = []

    for row in list_of_lists:
        if len(row) == 0:
            row = [rng.randint(0, 100) for _ in range(5)]
        row = [float(x) for x in row]
        filled_rows.append(row)
        row_lengths.append(len(row))

    flat_values = np.array([v for row in filled_rows for v in row], dtype=float).reshape(-1, 1)

    scaler = MinMaxScaler(feature_range=(0, 1))
    scaled_flat = scaler.fit_transform(flat_values).ravel()

    result = []
    idx = 0
    for length in row_lengths:
        result.append([float(x) for x in scaled_flat[idx:idx + length]])
        idx += length

    return result
