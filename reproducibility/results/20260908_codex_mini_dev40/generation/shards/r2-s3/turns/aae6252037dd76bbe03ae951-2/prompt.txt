import numpy as np
import datetime

def task_func(arr):
    arr_np = np.asarray(arr)

    def reverse_words(value):
        text = value.decode() if isinstance(value, (bytes, bytearray)) else str(value)
        words = text.split()
        return " ".join(words[::-1]) if words else ""

    result = np.empty(arr_np.shape, dtype=object)

    if arr_np.size == 0:
        return result

    for idx in np.ndindex(arr_np.shape):
        result[idx] = reverse_words(arr_np[idx])

    return result
