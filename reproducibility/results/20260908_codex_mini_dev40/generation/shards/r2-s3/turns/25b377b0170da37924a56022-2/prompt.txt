import numpy as np
import random
from sklearn.preprocessing import MinMaxScaler

def task_func(list_of_lists, seed=42):
    if not list_of_lists:
        return []

    rng = random.Random(seed)

    filled_lists = []
    lengths = []

    for inner in list_of_lists:
        if len(inner) == 0:
            inner = [rng.randint(0, 100) for _ in range(5)]
        else:
            inner = list(inner)
        filled_lists.append(inner)
        lengths.append(len(inner))

    flat_values = np.array([value for inner in filled_lists for value in inner], dtype=float).reshape(-1, 1)

    scaler = MinMaxScaler(feature_range=(0, 1))
    scaled_flat = scaler.fit_transform(flat_values).ravel()

    result = []
    index = 0
    for length in lengths:
        result.append([float(x) for x in scaled_flat[index:index + length]])
        index += length

    return result
