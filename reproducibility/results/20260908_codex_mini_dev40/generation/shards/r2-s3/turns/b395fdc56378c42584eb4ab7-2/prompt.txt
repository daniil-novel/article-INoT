import numpy as np
import datetime

def task_func(arr):
    arr = np.asarray(arr)

    def reverse_words(x):
        s = str(x)
        if not s:
            return ""
        words = s.split()
        return " ".join(words[::-1]) if words else ""

    return np.vectorize(reverse_words, otypes=[str])(arr)
