import struct
import zlib

# Constants
KEY = '470FC614'


def task_func(hex_string=KEY):
    if not isinstance(hex_string, str):
        raise ValueError("hex_string must be a string")

    s = hex_string.strip()
    if s.startswith(("0x", "0X")):
        s = s[2:]

    if len(s) != 8:
        raise ValueError("hex_string must represent exactly 4 bytes (8 hex digits)")

    try:
        raw_bytes = bytes.fromhex(s)
    except ValueError as exc:
        raise ValueError("hex_string contains invalid hex characters") from exc

    float_value = struct.unpack(">f", raw_bytes)[0]
    binary32_bytes = struct.pack(">f", float_value)
    return zlib.compress(binary32_bytes)
