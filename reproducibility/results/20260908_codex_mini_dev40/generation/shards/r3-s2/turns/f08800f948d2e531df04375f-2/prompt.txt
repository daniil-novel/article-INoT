import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame):
    numeric_df = df.select_dtypes(include=[np.number]).copy()
    result_columns = ["principal component 1", "principal component 2"]

    fig, ax = plt.subplots()

    # Gracefully handle cases with no numeric data
    if numeric_df.shape[0] == 0 or numeric_df.shape[1] == 0:
        pca_df = pd.DataFrame(columns=result_columns, index=df.index)
        ax.set_xlabel("principal component")
        ax.set_ylabel("principal component 2")
        return pca_df, ax

    # Fill missing values with column means; if a column is entirely NaN, fall back to 0
    numeric_df = numeric_df.fillna(numeric_df.mean()).fillna(0)

    # PCA needs at least 1 component and cannot exceed min(n_samples, n_features)
    n_components = min(2, numeric_df.shape[0], numeric_df.shape[1])

    if n_components == 0:
        pca_df = pd.DataFrame(columns=result_columns, index=df.index)
        ax.set_xlabel("principal component")
        ax.set_ylabel("principal component 2")
        return pca_df, ax

    pca = PCA(n_components=n_components)
    transformed = pca.fit_transform(numeric_df)

    # Pad with zeros if only one principal component is available
    if n_components == 1:
        transformed = np.hstack([transformed, np.zeros((transformed.shape[0], 1))])

    pca_df = pd.DataFrame(
        transformed[:, :2],
        columns=result_columns,
        index=df.index
    )

    sns.scatterplot(
        data=pca_df,
        x="principal component 1",
        y="principal component 2",
        ax=ax
    )
    ax.set_xlabel("principal component")
    ax.set_ylabel("principal component 2")

    return pca_df, ax
