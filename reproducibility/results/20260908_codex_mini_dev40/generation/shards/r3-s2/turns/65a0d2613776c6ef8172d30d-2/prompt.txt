import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame):
    # Select numeric columns only
    numeric_df = df.select_dtypes(include=np.number).copy()

    if numeric_df.shape[1] < 2:
        raise ValueError("PCA requires at least two numeric columns.")

    if numeric_df.shape[0] == 0:
        raise ValueError("Input DataFrame contains no rows to process.")

    # Replace missing values with each column's mean
    col_means = numeric_df.mean()
    numeric_df = numeric_df.fillna(col_means)

    # Fallback for columns that are entirely NaN
    numeric_df = numeric_df.fillna(0)

    # Perform PCA on the first two principal components
    pca = PCA(n_components=2)
    transformed = pca.fit_transform(numeric_df.to_numpy())

    pca_df = pd.DataFrame(
        transformed,
        columns=["principal component 1", "principal component 2"],
        index=df.index,
    )

    # Create scatter plot
    fig, ax = plt.subplots()
    sns.scatterplot(
        x=pca_df["principal component 1"],
        y=pca_df["principal component 2"],
        ax=ax,
    )
    ax.set_xlabel("principal component")
    ax.set_ylabel("principal component 2")

    return pca_df, ax
