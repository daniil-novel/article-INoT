import re
import os
import shutil

def task_func(source_dir, target_dir, file_pattern=r'\b[A-Za-z0-9]+\.(txt|doc|docx)\b'):
    """
    Recursively scan source_dir for files whose full paths match file_pattern,
    write matched paths to a configuration file in target_dir, and return the
    configuration file path.
    """
    source_dir = os.fspath(source_dir)
    target_dir = os.fspath(target_dir)

    if not os.path.isdir(source_dir):
        raise FileNotFoundError(f"Source directory does not exist: {source_dir}")

    os.makedirs(target_dir, exist_ok=True)

    pattern = re.compile(file_pattern)
    matched_paths = []

    for root, _, files in os.walk(source_dir):
        for name in files:
            full_path = os.path.join(root, name)
            if pattern.search(full_path):
                matched_paths.append(os.path.normpath(full_path))

    # Remove duplicates while preserving deterministic output order
    matched_paths = sorted(set(matched_paths))

    config_path = os.path.join(target_dir, "matched_files.cfg")
    with open(config_path, "w", encoding="utf-8") as config_file:
        for path in matched_paths:
            config_file.write(path + "\n")

    return config_path
