from collections import Counter
from operator import itemgetter
import itertools

def task_func(word_dict):
    counter = Counter()
    for value in word_dict.values():
        if isinstance(value, str):
            counter.update(ch for ch in value if ch.isalpha())
        else:
            try:
                for item in value:
                    if isinstance(item, str):
                        if len(item) == 1:
                            if item.isalpha():
                                counter[item] += 1
                        else:
                            counter.update(ch for ch in item if ch.isalpha())
            except TypeError:
                continue
    return dict(counter)
