import pytz
import numpy as np
from dateutil.parser import parse
import math
MOON_PHASES_YEARS = np.array([1987, 1994, 2001, 2008, 2015, 2022])

from datetime import datetime, timedelta
import calendar


def _safe_localize(tz, naive_dt):
    try:
        return tz.localize(naive_dt, is_dst=None)
    except pytz.AmbiguousTimeError:
        return tz.localize(naive_dt, is_dst=True)
    except pytz.NonExistentTimeError:
        return tz.localize(naive_dt + timedelta(hours=1), is_dst=True)


def _clamp_day(year, month, day):
    return min(day, calendar.monthrange(year, month)[1])


def _julian_day(dt):
    dt_utc = dt.astimezone(pytz.utc)
    year = dt_utc.year
    month = dt_utc.month
    day = (
        dt_utc.day
        + (dt_utc.hour
           + (dt_utc.minute
              + (dt_utc.second + dt_utc.microsecond / 1_000_000.0) / 60.0) / 60.0) / 24.0
    )

    if month <= 2:
        year -= 1
        month += 12

    a = year // 100
    b = 2 - a + a // 4

    return (
        math.floor(365.25 * (year + 4716))
        + math.floor(30.6001 * (month + 1))
        + day
        + b
        - 1524.5
    )


def task_func(date_str, from_tz, to_tz):
    try:
        dt = parse(date_str)
    except Exception as exc:
        raise ValueError(f"Invalid date string: {date_str!r}") from exc

    try:
        src_tz = pytz.timezone(from_tz)
    except Exception as exc:
        raise ValueError(f"Invalid from_tz: {from_tz!r}") from exc

    try:
        dst_tz = pytz.timezone(to_tz)
    except Exception as exc:
        raise ValueError(f"Invalid to_tz: {to_tz!r}") from exc

    if dt.tzinfo is None:
        dt = _safe_localize(src_tz, dt)
    else:
        dt = dt.astimezone(src_tz)

    local_dt = dt.astimezone(dst_tz)

    # Map the year onto the 7-year lunar cycle anchored by MOON_PHASES_YEARS.
    base_year = int(MOON_PHASES_YEARS[0])
    cycle_length = int(np.diff(MOON_PHASES_YEARS).mean()) if len(MOON_PHASES_YEARS) > 1 else 7
    reference_year = base_year + ((local_dt.year - base_year) % cycle_length)

    safe_day = _clamp_day(reference_year, local_dt.month, local_dt.day)
    ref_naive = datetime(
        reference_year,
        local_dt.month,
        safe_day,
        local_dt.hour,
        local_dt.minute,
        local_dt.second,
        local_dt.microsecond,
    )
    ref_local = _safe_localize(dst_tz, ref_naive)

    jd = _julian_day(ref_local)

    # Moon illumination approximation in [0, 1]
    synodic_month = 29.530588853
    new_moon_jd = 2451550.1
    age = ((jd - new_moon_jd) % synodic_month) / synodic_month
    illumination = 0.5 * (1.0 - math.cos(2.0 * math.pi * age))

    return float(max(0.0, min(1.0, illumination)))
