import string
import random
from collections import Counter

def task_func(n, seed=None):
    if seed is not None:
        random.seed(seed)
    if n < 0:
        raise ValueError("n must be non-negative")
    letters = [random.choice(string.ascii_lowercase) for _ in range(n)]
    return Counter(letters)
