import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(s1, s2):
    """
    Create a swarm plot comparing two Series and highlight their shared values.

    Returns
    -------
    ax : matplotlib.axes.Axes
        The axes containing the swarm plot.
    intersection_count : int
        Number of unique intersecting data points between s1 and s2.
    """
    # Normalize inputs to pandas Series
    s1 = pd.Series(s1).dropna()
    s2 = pd.Series(s2).dropna()

    # Compute unique intersection values
    intersection_values = set(s1.unique()) & set(s2.unique())
    intersection_count = len(intersection_values)

    # Build a long-form dataframe for plotting
    plot_df = pd.DataFrame({
        "series": ["s1"] * len(s1) + ["s2"] * len(s2),
        "value": list(s1) + list(s2),
    })

    # Create plot
    fig, ax = plt.subplots()
    if not plot_df.empty:
        sns.swarmplot(data=plot_df, x="series", y="value", ax=ax)

    # Highlight intersecting values with red dashed horizontal lines
    for value in intersection_values:
        ax.axhline(y=value, color="red", linestyle="--", linewidth=1, alpha=0.7)

    ax.set_xlabel("Series")
    ax.set_ylabel("Value")
    ax.set_title("Swarm Plot of Two Series with Intersections Highlighted")

    return ax, intersection_count
