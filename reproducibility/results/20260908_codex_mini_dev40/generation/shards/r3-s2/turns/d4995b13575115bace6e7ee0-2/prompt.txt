```python
import pandas as pd
import numpy as np

def task_func(file_path="data.csv", columns=["A", "B", "C"]):
    import matplotlib.pyplot as plt

    df = pd.read_csv(file_path)

    if not columns:
        raise ValueError("columns must contain at least one column name.")

    missing = [col for col in columns if col not in df.columns]
    if missing:
        raise KeyError(f"Missing columns in CSV: {missing}")

    # Convert the specified columns to floats where possible
    selected = df.loc[:, columns].apply(pd.to_numeric, errors="coerce").astype(float)
    df = df.copy()
    df.loc[:, columns] = selected

    # Plot the selected columns
    ax = selected.plot(kind="line")
    ax.set_xlabel("Index")
    ax.set_ylabel("Value")
    ax.set_title("Line chart of selected columns")

    # Compute cube-root values, preserving sign for negative numbers
    cube_root_series = pd.Series(
        np.cbrt(selected.to_numpy(dtype=float).ravel()),
        name="cube_root",
    )

    return df, ax, cube_root_series
```
