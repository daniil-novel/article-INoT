import re
import os
import shutil

def task_func(source_dir, target_dir, file_pattern=r'\b[A-Za-z0-9]+\.(txt|doc|docx)\b'):
    source_dir = os.path.abspath(source_dir)
    target_dir = os.path.abspath(target_dir)

    if not os.path.isdir(source_dir):
        raise FileNotFoundError(f"Source directory does not exist: {source_dir}")

    os.makedirs(target_dir, exist_ok=True)

    pattern = re.compile(file_pattern)

    matches = []
    for root, _, files in os.walk(source_dir):
        for filename in files:
            full_path = os.path.abspath(os.path.normpath(os.path.join(root, filename)))
            if pattern.search(full_path):
                matches.append(full_path)

    matches.sort()

    config_path = os.path.join(target_dir, "matched_paths.cfg")
    with open(config_path, "w", encoding="utf-8", newline="\n") as config_file:
        for path in matches:
            config_file.write(path + "\n")

    return config_path
