import pandas as pd
from sklearn.cluster import KMeans
import matplotlib.pyplot as plt

def task_func(df, n_clusters=3, random_state=0):
    """
    Convert 'date' to ordinal, cluster by 'date' and 'value', and plot the result.

    Returns
    -------
    matplotlib.axes.Axes
        The Axes object containing the scatter plot of the clusters.
    """
    if df.empty:
        raise ValueError("The DataFrame is empty.")
    required_columns = {"date", "value"}
    if not required_columns.issubset(df.columns):
        raise ValueError("The DataFrame lacks required columns: 'date' and 'value'.")

    if not isinstance(n_clusters, int) or n_clusters <= 0:
        raise ValueError("n_clusters must be a positive integer.")
    if n_clusters > len(df):
        raise ValueError("n_clusters cannot exceed the number of rows in the DataFrame.")

    data = df.copy()

    # Convert and validate date
    dates = pd.to_datetime(data["date"], errors="coerce")
    if dates.isna().any():
        raise ValueError("The 'date' column contains invalid or missing datetime values.")
    data["date_ordinal"] = dates.map(lambda x: x.toordinal())

    # Convert and validate value
    values = pd.to_numeric(data["value"], errors="coerce")
    if values.isna().any():
        raise ValueError("The 'value' column contains invalid or missing numeric values.")
    data["value"] = values

    features = data[["date_ordinal", "value"]]

    model = KMeans(n_clusters=n_clusters, random_state=random_state)
    labels = model.fit_predict(features)
    data["cluster"] = labels

    fig, ax = plt.subplots()
    scatter = ax.scatter(
        data["date_ordinal"],
        data["value"],
        c=data["cluster"],
        cmap="viridis",
        s=50,
    )

    ax.set_xlabel("Date (ordinal)")
    ax.set_ylabel("Value")
    ax.set_title("KMeans Clustering of Value vs Date")

    return ax
