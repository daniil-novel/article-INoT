import pandas as pd
import matplotlib.pyplot as plt

# Constants
COLUMN_NAMES = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']


def task_func(data):
    df = pd.DataFrame(data).copy()

    # Assign column names for the original data
    n_cols = df.shape[1]
    if n_cols > 0:
        if n_cols <= len(COLUMN_NAMES):
            df.columns = COLUMN_NAMES[:n_cols]
        else:
            extra_names = [f"Col{i}" for i in range(len(COLUMN_NAMES) + 1, n_cols + 1)]
            df.columns = COLUMN_NAMES + extra_names

    # Compute row-wise averages robustly
    numeric_df = df.apply(pd.to_numeric, errors='coerce') if not df.empty else df
    df["Average"] = numeric_df.mean(axis=1) if n_cols > 0 else pd.Series(dtype=float)

    # Plot averages against row indices
    fig, ax = plt.subplots()
    ax.plot(df.index, df["Average"], marker="o")
    ax.set_xlabel("Row Index")
    ax.set_ylabel("Average")
    ax.set_title("Row Averages")
    ax.grid(True, linestyle="--", alpha=0.5)

    return df, ax
