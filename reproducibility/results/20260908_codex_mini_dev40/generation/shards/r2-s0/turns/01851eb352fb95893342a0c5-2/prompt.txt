```python
import pandas as pd
import matplotlib.pyplot as plt

# Constants
COLUMN_NAMES = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']


def task_func(data):
    df = pd.DataFrame(data)

    # Allow a truly empty input to produce an empty, well-formed result
    if df.shape[1] == 0 and df.empty:
        df = pd.DataFrame(columns=COLUMN_NAMES)
    else:
        if df.shape[1] != len(COLUMN_NAMES):
            raise ValueError(
                f"Each row must contain exactly {len(COLUMN_NAMES)} values."
            )
        df.columns = COLUMN_NAMES
        df = df.apply(pd.to_numeric, errors='raise')

    df["Average"] = df.mean(axis=1)

    fig, ax = plt.subplots()
    ax.plot(df.index, df["Average"], marker="o")
    ax.set_xlabel("Row Index")
    ax.set_ylabel("Average")
    ax.set_title("Row Averages")

    return df, ax
```
