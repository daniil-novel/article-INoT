import re
import os
from pathlib import Path
import glob


def task_func(directory_path: str, regex_pattern: str = r'\\(.+?\\)|\\w') -> dict:
    directory = Path(directory_path)
    if not directory.exists() or not directory.is_dir():
        return {}

    # Normalize the supplied default pattern into the intended tokenization:
    # - text inside parentheses becomes one match
    # - everything outside parentheses is matched character-by-character
    if regex_pattern in {r'\\(.+?\\)|\\w', r'\(.+?\)|\w'}:
        regex_pattern = r'\((.*?)\)|[\s\S]'

    compiled = re.compile(regex_pattern)
    result = {}

    for path in sorted(directory.iterdir(), key=lambda p: p.name):
        if not path.is_file():
            continue

        try:
            content = path.read_text(encoding="utf-8", errors="ignore")
        except OSError:
            continue

        matches = []
        for match in compiled.finditer(content):
            groups = match.groups()
            token = next((group for group in groups if group is not None), match.group(0)) if groups else match.group(0)
            if len(token) >= 2 and token.startswith("(") and token.endswith(")"):
                token = token[1:-1]
            matches.append(token)

        result[path.name] = matches

    return result
