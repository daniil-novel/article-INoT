import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


def task_func(array_length=100):
    if isinstance(array_length, bool) or not isinstance(array_length, int):
        raise TypeError("array_length must be an integer.")
    if array_length <= 0:
        raise ValueError("array_length must be a positive integer.")

    array1 = np.random.random(array_length)
    array2 = np.random.random(array_length)

    statistics = pd.DataFrame(
        {
            "Array1": [
                np.mean(array1),
                np.median(array1),
                np.std(array1),
            ],
            "Array2": [
                np.mean(array2),
                np.median(array2),
                np.std(array2),
            ],
        },
        index=["Mean", "Median", "Standard Deviation"],
    )

    fig, ax = plt.subplots(figsize=(8, 5))
    indices = np.arange(len(statistics.index))
    width = 0.35

    ax.bar(indices - width / 2, statistics["Array1"], width, label="Array1")
    ax.bar(indices + width / 2, statistics["Array2"], width, label="Array2")

    ax.set_xticks(indices)
    ax.set_xticklabels(statistics.index)
    ax.set_ylabel("Value")
    ax.set_title("Comparison of Array Statistics")
    ax.legend()

    return statistics, ax
