import csv
import random
import string


def task_func(file_path,
          num_rows,
          gender=['Male', 'Female', 'Non-Binary'],
          countries=['USA', 'UK', 'Canada', 'Australia', 'India'],
          seed=None):
    if not isinstance(num_rows, int):
        raise TypeError("num_rows must be an integer")

    if gender is None or len(gender) == 0:
        raise ValueError("gender must be a non-empty sequence")
    if countries is None or len(countries) == 0:
        raise ValueError("countries must be a non-empty sequence")

    if seed is not None:
        random.seed(seed)

    headers = ['Name', 'Age', 'Gender', 'Country']

    with open(file_path, mode='w', newline='', encoding='utf-8') as f:
        writer = csv.writer(f)
        writer.writerow(headers)

        if num_rows <= 0:
            return file_path

        for _ in range(num_rows):
            name = ''.join(random.choices(string.ascii_uppercase, k=5))
            age = random.randint(20, 60)
            chosen_gender = random.choice(gender)
            country = random.choice(countries)
            writer.writerow([name, age, chosen_gender, country])

    return file_path
