import pandas as pd
import numpy as np

def task_func(rows=100, columns=3):
    if not isinstance(rows, int) or isinstance(rows, bool):
        raise ValueError("rows must be a non-negative integer")
    if not isinstance(columns, int) or isinstance(columns, bool):
        raise ValueError("columns must be a non-negative integer")
    if rows < 0 or columns < 0:
        raise ValueError("rows and columns must be non-negative")

    def make_column_name(n):
        name = ""
        while True:
            n, rem = divmod(n, 26)
            name = chr(ord('a') + rem) + name
            if n == 0:
                break
            n -= 1
        return name

    col_names = [make_column_name(i) for i in range(columns)]
    letters = np.array(list("abcdefghijklmnopqrstuvwxyz"))
    data = np.random.choice(letters, size=(rows, columns))

    return pd.DataFrame(data, columns=col_names)
