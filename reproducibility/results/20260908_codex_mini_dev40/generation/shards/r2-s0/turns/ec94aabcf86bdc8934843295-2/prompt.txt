from random import randint, seed
import pandas as pd
from collections.abc import Mapping, Iterable

def task_func(goals, penalties, rng_seed=None):
    FINE_COST = 100

    if rng_seed is not None:
        seed(rng_seed)

    def _is_nonstring_iterable(value):
        return isinstance(value, Iterable) and not isinstance(value, (str, bytes, bytearray, Mapping))

    def _normalize_top_level(value):
        if isinstance(value, Mapping):
            return list(value.keys()), list(value.values())
        if _is_nonstring_iterable(value):
            return None, list(value)
        return None, [value]

    def _resolve_count(spec):
        if isinstance(spec, bool):
            raise ValueError("Counts must be non-negative integers.")

        if isinstance(spec, int):
            if spec < 0:
                raise ValueError("Counts must be non-negative integers.")
            return spec

        if isinstance(spec, (list, tuple)) and len(spec) == 2:
            low, high = spec
            if isinstance(low, bool) or isinstance(high, bool) or not isinstance(low, int) or not isinstance(high, int):
                raise ValueError("Range bounds must be integers.")
            if low < 0 or high < 0:
                raise ValueError("Counts must be non-negative integers.")
            if low > high:
                low, high = high, low
            return randint(low, high)

        raise ValueError("Unsupported count specification.")

    goal_keys, goal_specs = _normalize_top_level(goals)
    penalty_keys, penalty_specs = _normalize_top_level(penalties)

    if goal_keys is not None and penalty_keys is not None:
        if set(goal_keys) != set(penalty_keys):
            raise ValueError("goals and penalties must describe the same teams.")
        teams = goal_keys
        penalty_map = dict(zip(penalty_keys, penalty_specs))
        goal_map = dict(zip(goal_keys, goal_specs))
        goal_specs = [goal_map[t] for t in teams]
        penalty_specs = [penalty_map[t] for t in teams]
    elif goal_keys is not None:
        teams = goal_keys
        if len(goal_specs) != len(penalty_specs):
            raise ValueError("goals and penalties must have the same length.")
    elif penalty_keys is not None:
        teams = penalty_keys
        if len(goal_specs) != len(penalty_specs):
            raise ValueError("goals and penalties must have the same length.")
    else:
        if len(goal_specs) != len(penalty_specs):
            raise ValueError("goals and penalties must have the same length.")
        teams = [f"Team {i + 1}" for i in range(len(goal_specs))]

    rows = []
    for team, goal_spec, penalty_spec in zip(teams, goal_specs, penalty_specs):
        goal_count = _resolve_count(goal_spec)
        penalty_count = _resolve_count(penalty_spec)
        fines = penalty_count * FINE_COST
        rows.append(
            {
                "Team": team,
                "Match Result": f"Goals: {goal_count}, Penalties: {penalty_count}, Fines: {fines}",
            }
        )

    return pd.DataFrame(rows, columns=["Team", "Match Result"])
