import pandas as pd
import matplotlib.pyplot as plt

# Constants
COLUMN_NAMES = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']


def task_func(data):
    """
    Convert a 2D input into a DataFrame with columns A-H, compute row averages,
    append them as an 'Average' column, and plot the averages against row index.

    Parameters
    ----------
    data : 2D array-like or pandas.DataFrame
        Input data with exactly 8 values per row.

    Returns
    -------
    tuple
        (df, ax) where df is the augmented DataFrame and ax is the matplotlib Axes.
    """
    expected_cols = len(COLUMN_NAMES)

    # Normalize input to a DataFrame with exactly 8 columns.
    if isinstance(data, pd.DataFrame):
        df = data.copy()
        if df.shape[1] != expected_cols:
            raise ValueError(f"Expected {expected_cols} columns, got {df.shape[1]}.")
        df.columns = COLUMN_NAMES
    else:
        try:
            df = pd.DataFrame(data, columns=COLUMN_NAMES)
        except ValueError as exc:
            raise ValueError(
                f"Input must be a 2D structure with exactly {expected_cols} values per row."
            ) from exc

    # Ensure all values are numeric.
    try:
        df = df.apply(pd.to_numeric, errors="raise")
    except Exception as exc:
        raise ValueError("All values must be numeric.") from exc

    # Reject missing values.
    if df.isna().any().any():
        raise ValueError("Missing values are not allowed.")

    # Compute row averages and append them.
    df["Average"] = df.mean(axis=1)

    # Plot row averages against row indices.
    fig, ax = plt.subplots()
    ax.plot(df.index, df["Average"], marker="o", linestyle="-")
    ax.set_xlabel("Row Index")
    ax.set_ylabel("Average")
    ax.set_title("Average of Each Row")
    ax.grid(True, alpha=0.3)

    return df, ax
