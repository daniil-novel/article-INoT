With the default pattern:

- `\(.+?\)` matches a parenthesized substring like `(abc)` as **one** match


import re
import os
from pathlib import Path
import glob

def task_func(directory_path: str, regex_pattern: str = r'\(.+?\)|\w') -> dict:
    """
    Extract regex matches from all text files in a directory.

    Returns a dictionary mapping file names (without path) to lists of matches.
    """
    directory = Path(directory_path)
    if not directory.exists() or not directory.is_dir():
        return {}

    pattern = re.compile(regex_pattern)
    result = {}

    for file_path in directory.iterdir():
        if not file_path.is_file():
            continue
        if file_path.suffix.lower() != ".txt":
            continue

        try:
            text = file_path.read_text(encoding="utf-8", errors="ignore")
        except OSError:
            continue

        matches = [match.group(0) for match in pattern.finditer(text)]
        result[file_path.name] = matches

    return result
