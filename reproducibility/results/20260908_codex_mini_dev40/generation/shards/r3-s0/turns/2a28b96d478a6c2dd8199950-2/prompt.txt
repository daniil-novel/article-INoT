from collections import Counter
import random
import matplotlib.pyplot as plt

def task_func(num_rolls, num_dice, plot_path=None, random_seed=0):
    """
    Simulate rolling num_dice six-sided dice num_rolls times, count sum frequencies,
    and plot the distribution as a bar chart.

    Returns:
        tuple: (Counter, matplotlib.axes.Axes)
    """
    if not isinstance(num_rolls, int) or not isinstance(num_dice, int):
        raise TypeError("num_rolls and num_dice must be integers")
    if num_rolls < 0 or num_dice < 0:
        raise ValueError("num_rolls and num_dice must be non-negative integers")

    random.seed(random_seed)

    min_sum = num_dice
    max_sum = 6 * num_dice

    distribution = Counter({s: 0 for s in range(min_sum, max_sum + 1)})

    for _ in range(num_rolls):
        roll_sum = sum(random.randint(1, 6) for _ in range(num_dice))
        distribution[roll_sum] += 1

    sums = list(range(min_sum, max_sum + 1))
    counts = [distribution[s] for s in sums]

    fig, ax = plt.subplots()
    ax.bar(sums, counts)
    ax.set_title("Distribution of Dice Roll Sums")
    ax.set_xlabel("Sum of Dice Roll")
    ax.set_ylabel("Count")
    ax.set_xticks(sums)

    if plot_path is not None:
        fig.savefig(plot_path)

    return distribution, ax
