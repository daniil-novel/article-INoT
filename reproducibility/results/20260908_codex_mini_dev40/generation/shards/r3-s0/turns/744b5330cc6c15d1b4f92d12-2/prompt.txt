from random import randint, seed
import pandas as pd
from numbers import Integral

# Predefined fine per penalty
PENALTY_COST = 100

# Method
def task_func(goals, penalties, rng_seed=None):
    """
    Build a DataFrame with columns:
      - Team
      - Match Result

    For each team, random goals are generated in [0, goals[i]] and random
    penalties in [0, penalties[i]]. Penalties are converted into fines using
    PENALTY_COST.
    """
    if rng_seed is not None:
        seed(rng_seed)

    def _to_list(value, name):
        if isinstance(value, (str, bytes)):
            raise ValueError(f"{name} must be an iterable of non-negative integers")
        try:
            return list(value)
        except TypeError:
            return [value]

    goals_list = _to_list(goals, "goals")
    penalties_list = _to_list(penalties, "penalties")

    if len(goals_list) != len(penalties_list):
        raise ValueError("goals and penalties must have the same length")

    rows = []
    for idx, (goal_bound, penalty_bound) in enumerate(zip(goals_list, penalties_list), start=1):
        if not isinstance(goal_bound, Integral) or isinstance(goal_bound, bool):
            raise ValueError("goals must contain integers")
        if not isinstance(penalty_bound, Integral) or isinstance(penalty_bound, bool):
            raise ValueError("penalties must contain integers")
        if goal_bound < 0 or penalty_bound < 0:
            raise ValueError("goals and penalties must be non-negative")

        team_goals = randint(0, goal_bound)
        team_penalties = randint(0, penalty_bound)
        fine = team_penalties * PENALTY_COST

        rows.append(
            {
                "Team": f"Team {idx}",
                "Match Result": f"Goals: {team_goals}, Penalties: {team_penalties}, Fine: {fine}",
            }
        )

    return pd.DataFrame(rows, columns=["Team", "Match Result"])
