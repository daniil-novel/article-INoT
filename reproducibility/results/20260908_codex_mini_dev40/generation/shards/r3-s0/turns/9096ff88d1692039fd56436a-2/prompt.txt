import pandas as pd
import matplotlib.pyplot as plt

# Constants
COLUMN_NAMES = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']


def task_func(data):
    # Normalize input to a DataFrame
    df = pd.DataFrame(data)

    # Assign column names when reasonable; extend dynamically if needed
    n_cols = df.shape[1]
    if n_cols <= len(COLUMN_NAMES):
        df.columns = COLUMN_NAMES[:n_cols]
    else:
        extra_names = [f"Col{i}" for i in range(len(COLUMN_NAMES) + 1, n_cols + 1)]
        df.columns = COLUMN_NAMES + extra_names

    # Compute row-wise averages using numeric values only
    numeric_df = df.apply(pd.to_numeric, errors="coerce")
    averages = numeric_df.mean(axis=1)

    # Append the Average column
    df["Average"] = averages

    # Create the plot
    fig, ax = plt.subplots()
    ax.plot(df.index, averages, marker="o")
    ax.set_xlabel("Row Index")
    ax.set_ylabel("Average")
    ax.set_title("Row Averages")
    ax.grid(True)

    return df, ax
