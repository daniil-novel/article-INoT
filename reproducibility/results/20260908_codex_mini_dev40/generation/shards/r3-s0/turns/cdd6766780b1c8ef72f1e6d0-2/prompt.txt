import csv
import random

def task_func(file_path,
          num_rows,
          gender=['Male', 'Female', 'Non-Binary'],
          countries=['USA', 'UK', 'Canada', 'Australia', 'India'],
          seed=None):
    if not isinstance(num_rows, int):
        raise TypeError("num_rows must be an integer")
    if not gender:
        raise ValueError("gender must not be empty")
    if not countries:
        raise ValueError("countries must not be empty")

    rng = random.Random(seed) if seed is not None else random
    alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

    with open(file_path, mode="w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["Name", "Age", "Gender", "Country"])

        for _ in range(max(0, num_rows)):
            name = "".join(rng.choice(alphabet) for _ in range(5))
            age = rng.randint(20, 60)
            chosen_gender = rng.choice(gender)
            country = rng.choice(countries)
            writer.writerow([name, age, chosen_gender, country])

    return file_path
