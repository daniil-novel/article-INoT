```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def task_func(array_length=100):
    """
    Generate two random arrays, compute mean/median/std for each,
    store results in a pandas DataFrame, and plot a comparison bar chart.

    Returns
    -------
    statistics : pandas.DataFrame
        DataFrame with rows ['Mean', 'Median', 'Standard Deviation']
        and columns ['Array1', 'Array2'].
    ax : matplotlib.axes.Axes
        The bar chart axes.
    """
    if not isinstance(array_length, int) or array_length <= 0:
        raise ValueError("array_length must be a positive integer.")

    # Generate two random arrays
    array1 = np.random.random(array_length)
    array2 = np.random.random(array_length)

    # Compute statistics
    statistics = pd.DataFrame(
        {
            "Array1": [
                np.mean(array1),
                np.median(array1),
                np.std(array1),
            ],
            "Array2": [
                np.mean(array2),
                np.median(array2),
                np.std(array2),
            ],
        },
        index=["Mean", "Median", "Standard Deviation"],
    )

    # Plot grouped bar chart
    ax = statistics.plot(kind="bar", figsize=(8, 5), rot=0)
    ax.set_title("Comparison of Statistics for Two Random Arrays")
    ax.set_xlabel("Statistics")
    ax.set_ylabel("Value")
    ax.legend(title="Arrays")
    plt.tight_layout()

    return statistics, ax
```
