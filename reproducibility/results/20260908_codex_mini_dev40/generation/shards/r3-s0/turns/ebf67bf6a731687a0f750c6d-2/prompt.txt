from random import randint, seed
import pandas as pd

# Method
def task_func(goals, penalties, rng_seed=None):
    if rng_seed is not None:
        seed(rng_seed)

    def _normalize(value):
        if isinstance(value, int):
            return [value]
        if isinstance(value, (str, bytes)):
            raise TypeError("goals and penalties must be integers or iterables of integers")
        try:
            return list(value)
        except TypeError as exc:
            raise TypeError("goals and penalties must be integers or iterables of integers") from exc

    goals_list = _normalize(goals)
    penalties_list = _normalize(penalties)

    if len(goals_list) != len(penalties_list):
        raise ValueError("goals and penalties must have the same length")

    penalty_cost = 10
    rows = []

    for idx, (goal_limit, penalty_limit) in enumerate(zip(goals_list, penalties_list), start=1):
        if not isinstance(goal_limit, int) or not isinstance(penalty_limit, int):
            raise TypeError("all goal and penalty values must be integers")
        if goal_limit < 0 or penalty_limit < 0:
            raise ValueError("goals and penalties must be non-negative")

        goals_scored = randint(0, goal_limit) if goal_limit > 0 else 0
        penalties_count = randint(0, penalty_limit) if penalty_limit > 0 else 0
        fine = penalties_count * penalty_cost

        rows.append({
            "Team": f"Team {idx}",
            "Match Result": f"Goals: {goals_scored}, Penalties: {penalties_count}, Fine: {fine}"
        })

    return pd.DataFrame(rows, columns=["Team", "Match Result"])
