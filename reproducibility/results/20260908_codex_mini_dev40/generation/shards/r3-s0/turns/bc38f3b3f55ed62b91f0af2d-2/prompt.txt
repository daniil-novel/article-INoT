import pandas as pd
import logging
# Set up basic configuration for logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

import os


def task_func(sheet_name, excel_file_location="test.xlsx", csv_file_location="test.csv"):
    try:
        if not os.path.isfile(excel_file_location):
            logging.error("Excel file does not exist at path: %s", excel_file_location)
            raise FileNotFoundError(
                f"Excel file does not exist at the specified path: {excel_file_location}"
            )

        logging.info(
            "Reading Excel file '%s' from sheet '%s'.",
            excel_file_location,
            sheet_name,
        )

        try:
            excel_df = pd.read_excel(excel_file_location, sheet_name=sheet_name)
        except ValueError as exc:
            message = str(exc).lower()
            if "worksheet" in message and "not found" in message or "sheet" in message and "not found" in message:
                logging.error(
                    "Specified sheet '%s' was not found in Excel file '%s'.",
                    sheet_name,
                    excel_file_location,
                )
                raise ValueError(
                    f"Specified sheet name '{sheet_name}' was not found in the Excel file."
                ) from exc
            raise

        if excel_df.shape[1] == 0:
            logging.info("Excel sheet contains no columns. Writing empty CSV and returning empty dictionary.")
            excel_df.to_csv(csv_file_location, index=False)
            return {}

        numeric_df = pd.DataFrame(index=excel_df.index)
        for column in excel_df.columns:
            numeric_df[column] = pd.to_numeric(excel_df[column], errors="raise")

        logging.info("Writing validated numeric data to CSV file '%s'.", csv_file_location)
        numeric_df.to_csv(csv_file_location, index=False)

        logging.info("Reading data back from CSV file '%s'.", csv_file_location)
        csv_df = pd.read_csv(csv_file_location)

        for column in csv_df.columns:
            csv_df[column] = pd.to_numeric(csv_df[column], errors="raise")

        result = {}
        for column in csv_df.columns:
            column_sum = csv_df[column].sum()
            result[column] = column_sum.item() if hasattr(column_sum, "item") else column_sum

        logging.info("Successfully calculated sums for %d columns.", len(result))
        return result

    except Exception:
        logging.exception("Failed to process Excel to CSV and calculate column sums.")
        raise
