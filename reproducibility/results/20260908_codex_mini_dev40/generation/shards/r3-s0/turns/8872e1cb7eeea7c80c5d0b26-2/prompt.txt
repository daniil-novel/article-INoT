import pandas as pd
import numpy as np

def task_func(rows=100, columns=3):
    """
    Create a pandas DataFrame filled with random lowercase alphabets.
    """
    # Validate input types
    if not isinstance(rows, (int, np.integer)) or not isinstance(columns, (int, np.integer)):
        raise TypeError("rows and columns must be integers")
    
    # Validate input values
    if rows < 0 or columns < 0:
        raise ValueError("rows and columns must be non-negative")

    def _col_name(idx):
        # Convert 0-based index to Excel-like lowercase column names:
        # 0 -> a, 1 -> b, ..., 25 -> z, 26 -> aa, etc.
        name = []
        idx += 1
        while idx > 0:
            idx, rem = divmod(idx - 1, 26)
            name.append(chr(ord('a') + rem))
        return ''.join(reversed(name))

    column_names = [_col_name(i) for i in range(columns)]
    letters = np.array(list("abcdefghijklmnopqrstuvwxyz"))

    data = np.random.choice(letters, size=(rows, columns))
    df = pd.DataFrame(data, columns=column_names)

    return df
