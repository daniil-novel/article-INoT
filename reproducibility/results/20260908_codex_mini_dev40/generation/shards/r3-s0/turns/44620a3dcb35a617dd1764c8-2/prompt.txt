```python
import pandas as pd
import logging
# Set up basic configuration for logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def task_func(sheet_name, excel_file_location="test.xlsx", csv_file_location="test.csv"):
    logging.info("Starting task_func with Excel file '%s' and sheet '%s'.", excel_file_location, sheet_name)

    try:
        df = pd.read_excel(excel_file_location, sheet_name=sheet_name)
    except FileNotFoundError:
        logging.exception("Excel file not found at path: %s", excel_file_location)
        raise
    except ValueError as exc:
        message = str(exc)
        if "Worksheet named" in message or "sheet" in message.lower():
            logging.exception("Specified sheet '%s' was not found in Excel file '%s'.", sheet_name, excel_file_location)
            raise ValueError(f"Sheet '{sheet_name}' was not found in the Excel file.") from exc
        raise

    df.to_csv(csv_file_location, index=False)
    logging.info("Excel data written to CSV file '%s'.", csv_file_location)

    csv_df = pd.read_csv(csv_file_location)
    column_sums = csv_df.sum(numeric_only=True).to_dict()

    logging.info("Column sums calculated successfully.")
    return column_sums
```
