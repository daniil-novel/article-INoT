import collections
import pandas as pd
def task_func(my_tuple, path_csv_files):
    if my_tuple is None:
        columns = []
    elif isinstance(my_tuple, str):
        columns = [my_tuple]
    else:
        columns = list(my_tuple)

    if path_csv_files is None:
        csv_files = []
    elif isinstance(path_csv_files, (str, bytes)) or not hasattr(path_csv_files, "__iter__"):
        csv_files = [path_csv_files]
    else:
        csv_files = list(path_csv_files)

    result = {col: collections.Counter() for col in columns}

    for csv_path in csv_files:
        try:
            df = pd.read_csv(csv_path)
        except pd.errors.EmptyDataError:
            continue

        for col in columns:
            if col in df.columns:
                values = df[col].dropna().tolist()
                result[col].update(values)

    return {col: dict(counter) for col, counter in result.items()}
