import sqlite3
import random
import os

def task_func(db_path,
          num_entries,
          users=['Alice', 'Bob', 'Charlie', 'Dave', 'Eve'],
          countries=['USA', 'UK', 'Canada', 'Australia', 'India'],
          random_seed=None):
    if not isinstance(num_entries, int):
        raise TypeError("num_entries must be an integer")
    if num_entries < 0:
        raise ValueError("num_entries must be non-negative")
    if not users:
        raise ValueError("users must not be empty")
    if not countries:
        raise ValueError("countries must not be empty")

    if random_seed is not None:
        random.seed(random_seed)

    db_path = str(db_path)
    parent_dir = os.path.dirname(db_path)
    if parent_dir:
        os.makedirs(parent_dir, exist_ok=True)

    conn = sqlite3.connect(db_path)
    try:
        cur = conn.cursor()
        cur.execute("DROP TABLE IF EXISTS users")
        cur.execute(
            """
            CREATE TABLE users (
                id INTEGER PRIMARY KEY,
                name TEXT,
                age INTEGER,
                country TEXT
            )
            """
        )

        for i in range(num_entries):
            cur.execute(
                "INSERT INTO users (id, name, age, country) VALUES (?, ?, ?, ?)",
                (i, random.choice(users), random.randint(20, 60), random.choice(countries))
            )

        conn.commit()
    finally:
        conn.close()

    return db_path
