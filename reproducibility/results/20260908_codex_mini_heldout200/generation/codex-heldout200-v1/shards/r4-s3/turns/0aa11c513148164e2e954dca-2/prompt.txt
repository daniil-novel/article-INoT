import numpy as np
import random
import matplotlib.pyplot as plt

# Constants
LETTERS = list("abcdefghijklmnopqrstuvwxyz")
NUMBERS = list(range(1, 27))

def task_func(n_pairs=26):
    """
    Generate and display a bar chart of random letter-number pairs.

    Parameters
    ----------
    n_pairs : int, optional
        Number of unique pairs to include in the chart. Must be between 1 and 26.

    Returns
    -------
    matplotlib.container.BarContainer
        The bar container object created by plt.bar().

    Raises
    ------
    ValueError
        If n_pairs is outside the range [1, 26].
    """
    if not 1 <= n_pairs <= 26:
        raise ValueError("'n_pairs' must be between 1 and 26, inclusive.")

    # Build the 26 unique letter:number pairs
    pairs = [f"{letter}:{number}" for letter, number in zip(LETTERS, NUMBERS)]

    # Randomly shuffle and select the requested number of pairs
    random.shuffle(pairs)
    selected_pairs = pairs[:n_pairs]

    # Assign random counts from 1 to 9
    counts = [random.randint(1, 9) for _ in range(n_pairs)]

    # Create the bar chart
    bars = plt.bar(selected_pairs, counts)

    # Format chart
    plt.title("Random Letter:Number Pairs Chart")
    plt.xlabel("Letter:Number Pairs")
    plt.ylabel("Counts")

    # Display chart
    plt.show()

    return bars
