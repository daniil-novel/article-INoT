import numpy as np
import matplotlib.pyplot as plt

def task_func(mu, sigma, sample_size):
    if sigma < 0:
        raise ValueError("sigma must be non-negative")

    if isinstance(sample_size, bool):
        raise TypeError("sample_size must be an integer")

    if not float(sample_size).is_integer():
        raise ValueError("sample_size must be an integer")

    sample_size = int(sample_size)

    if sample_size < 0:
        raise ValueError("sample_size must be non-negative")

    samples = np.random.normal(loc=mu, scale=sigma, size=sample_size)

    plt.hist(samples, bins=30, alpha=0.75, color="blue")
    plt.xlabel("Sample values")
    plt.ylabel("Frequency")
    plt.title("Histogram of Generated Samples")
    plt.show()

    return samples
