import numpy as np
import random

def task_func(MIN_WORDS, MAX_WORDS, WORDS_POOL):
    # Validate integer inputs
    if not isinstance(MIN_WORDS, (int, np.integer)) or isinstance(MIN_WORDS, bool):
        raise TypeError("MIN_WORDS must be an integer.")
    if not isinstance(MAX_WORDS, (int, np.integer)) or isinstance(MAX_WORDS, bool):
        raise TypeError("MAX_WORDS must be an integer.")

    if MIN_WORDS <= 0:
        raise ValueError("MIN_WORDS must be greater than 0.")
    if MIN_WORDS > MAX_WORDS:
        raise ValueError("MIN_WORDS must be less than or equal to MAX_WORDS.")

    pool = list(WORDS_POOL)
    if not pool:
        raise ValueError("WORDS_POOL must not be empty.")

    # Choose a random target length in the allowed range
    target_len = random.randint(MIN_WORDS, MAX_WORDS)

    # Build a palindrome sentence
    half_len = target_len // 2
    first_half = [random.choice(pool) for _ in range(half_len)]

    if target_len % 2 == 0:
        words = first_half + first_half[::-1]
    else:
        middle_word = random.choice(pool)
        words = first_half + [middle_word] + first_half[::-1]

    sentence = " ".join(words)

    # Check that generated sentence length is within the specified range
    actual_len = len(sentence.split())
    if not (MIN_WORDS <= actual_len <= MAX_WORDS):
        raise RuntimeError("Generated sentence length is outside the requested range.")

    # Ensure it is a palindrome sentence
    if words != words[::-1]:
        raise RuntimeError("Generated sentence is not a palindrome.")

    return sentence
