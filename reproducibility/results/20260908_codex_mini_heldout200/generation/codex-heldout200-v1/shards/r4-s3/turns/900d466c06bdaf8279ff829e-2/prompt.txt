import numpy as np
from scipy import stats
import matplotlib.pyplot as plt

def task_func(matrix):
    """
    Compute the skewness and kurtosis of the distribution of row maxima
    in `matrix`, and return a histogram plot of that distribution.

    Parameters
    ----------
    matrix : array-like
        2D numeric matrix-like input.

    Returns
    -------
    tuple
        (skewness, kurtosis, ax) where ax is a matplotlib Axes object.
    """
    try:
        data = np.asarray(matrix, dtype=float)
    except Exception as exc:
        raise ValueError("matrix must be a numeric array-like object") from exc

    if data.size == 0:
        raise ValueError("matrix must be non-empty")

    if data.ndim == 0:
        data = data.reshape(1, 1)
    elif data.ndim == 1:
        data = data.reshape(1, -1)
    elif data.ndim != 2:
        raise ValueError("matrix must be 1D or 2D array-like input")

    if data.shape[1] == 0:
        raise ValueError("matrix must contain at least one column")

    # Compute row maxima, ignoring NaNs if present
    row_maxima = np.nanmax(data, axis=1)
    row_maxima = row_maxima[np.isfinite(row_maxima)]

    if row_maxima.size == 0:
        raise ValueError("matrix must contain at least one finite row maximum")

    skewness = stats.skew(row_maxima, bias=False, nan_policy="omit")
    kurt = stats.kurtosis(row_maxima, bias=False, fisher=True, nan_policy="omit")

    fig, ax = plt.subplots()
    ax.hist(
        row_maxima,
        bins="auto",
        density=True,
        alpha=0.7,
        edgecolor="black",
    )

    # Estimate core density with a KDE overlay when possible
    if row_maxima.size > 1 and np.unique(row_maxima).size > 1:
        try:
            kde = stats.gaussian_kde(row_maxima)
            x_min, x_max = np.min(row_maxima), np.max(row_maxima)
            xs = np.linspace(x_min, x_max, 200)
            ax.plot(xs, kde(xs), color="red", linewidth=2)
        except Exception:
            pass

    ax.set_xlabel("Row maximum")
    ax.set_ylabel("Density")
    ax.set_title("Distribution of Row Maxima")

    return float(skewness), float(kurt), ax
