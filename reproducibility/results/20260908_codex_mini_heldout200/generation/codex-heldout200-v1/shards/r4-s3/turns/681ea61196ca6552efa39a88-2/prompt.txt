import collections
import pandas as pd

def task_func(my_tuple, path_csv_files):
    # Normalize requested columns
    if my_tuple is None:
        columns = []
    elif isinstance(my_tuple, (str, bytes)):
        columns = [my_tuple]
    else:
        columns = list(my_tuple)

    # Remove duplicates while preserving order
    seen = set()
    target_columns = []
    for col in columns:
        if col not in seen:
            seen.add(col)
            target_columns.append(col)

    # Normalize paths
    if path_csv_files is None:
        paths = []
    elif isinstance(path_csv_files, (str, bytes)):
        paths = [path_csv_files]
    else:
        paths = list(path_csv_files)

    # Prepare counters
    counts = {col: collections.Counter() for col in target_columns}

    # Aggregate counts across all files
    for path in paths:
        df = pd.read_csv(path)
        for col in target_columns:
            if col in df.columns:
                values = df[col].dropna().tolist()
                counts[col].update(values)

    # Convert Counter objects to plain dictionaries
    return {col: dict(counter) for col, counter in counts.items()}
