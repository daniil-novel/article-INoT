import pandas as pd
import numpy as np

DEFAULT_COLUMNS = ['Element', 'Count']

def task_func(elements, include_index=False):
    elements_as_str = [str(element) for element in elements]
    counts = [len(element) for element in elements_as_str]

    data = {
        'Element': elements_as_str,
        'Count': counts,
    }

    if include_index:
        data = {
            'Index': list(range(len(elements_as_str))),
            'Element': elements_as_str,
            'Count': counts,
        }
        return pd.DataFrame(data, columns=['Index', 'Element', 'Count'])

    return pd.DataFrame(data, columns=DEFAULT_COLUMNS)
