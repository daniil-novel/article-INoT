```python
import numpy as np
import random
import matplotlib.pyplot as plt

# Constants
LETTERS = list("abcdefghijklmnopqrstuvwxyz")
NUMBERS = list(range(1, 27))

def task_func(n_pairs=26):
    """
    Generate and display a bar chart of random letter-number pairs.

    Parameters
    ----------
    n_pairs : int, optional
        Number of unique letter:number pairs to plot. Must be between 1 and 26.

    Returns
    -------
    matplotlib.container.BarContainer
        The container of bars created by matplotlib.

    Raises
    ------
    ValueError
        If n_pairs is not an integer in the range [1, 26].
    """
    if not isinstance(n_pairs, (int, np.integer)) or isinstance(n_pairs, bool):
        raise ValueError("n_pairs must be an integer between 1 and 26 inclusive.")
    if n_pairs < 1 or n_pairs > 26:
        raise ValueError("n_pairs must be an integer between 1 and 26 inclusive.")

    # Build the 26 unique pairs: a:1, b:2, ..., z:26
    pairs = list(zip(LETTERS, NUMBERS))

    # Randomize the order and select the requested number of pairs
    random.shuffle(pairs)
    selected_pairs = pairs[:n_pairs]

    # Prepare labels and random counts
    labels = [f"{letter}:{number}" for letter, number in selected_pairs]
    counts = [random.randint(1, 9) for _ in range(n_pairs)]

    # Plot the bar chart
    bar_container = plt.bar(labels, counts)
    plt.title("Random Letter:Number Pairs Chart")
    plt.xlabel("Letter:Number Pairs")
    plt.ylabel("Counts")

    plt.show()
    return bar_container
```
