import pandas as pd
import random

# Constants
CATEGORIES = ['A', 'B', 'C', 'D', 'E']


def task_func(value_range=(0, 100)):
    """
    Generate a category distribution as a DataFrame.

    Parameters
    ----------
    value_range : tuple[int, int]
        Inclusive range (low, high) used to generate a random count for each
        category in CATEGORIES.

    Returns
    -------
    pandas.DataFrame
        DataFrame with columns:
        - Category: category name
        - Count: generated integer count
    """
    # Validate value_range
    if not isinstance(value_range, (tuple, list)) or len(value_range) != 2:
        raise ValueError("value_range must be a tuple or list of length 2.")

    low, high = value_range

    # Ensure numeric inputs
    if not isinstance(low, (int, float)) or not isinstance(high, (int, float)):
        raise TypeError("value_range bounds must be numeric.")

    # Convert to integers safely
    if int(low) != low or int(high) != high:
        raise ValueError("value_range bounds must be whole numbers.")
    low = int(low)
    high = int(high)

    # Validate bounds
    if low < 0 or high < 0:
        raise ValueError("value_range bounds must be non-negative.")
    if low > high:
        raise ValueError("value_range lower bound must be less than or equal to upper bound.")

    # Handle empty category list
    if not CATEGORIES:
        return pd.DataFrame(columns=["Category", "Count"])

    counts = [random.randint(low, high) for _ in CATEGORIES]
    return pd.DataFrame({"Category": CATEGORIES, "Count": counts})
