```python
import numpy as np
import random
import itertools
import pandas as pd
# Constants
PLANETS = [
    "Mercury",
    "Venus",
    "Earth",
    "Mars",
    "Jupiter",
    "Saturn",
    "Uranus",
    "Neptune",
]
ELEMENTS = [
    "Hydrogen",
    "Helium",
    "Oxygen",
    "Carbon",
    "Nitrogen",
    "Magnesium",
    "Silicon",
    "Iron",
    "Nickel",
]
def task_func():
    data = [
        [f"{random.choice(PLANETS)}:{random.choice(ELEMENTS)}" for _ in range(len(ELEMENTS))]
        for _ in range(len(PLANETS))
    ]
    return pd.DataFrame(data, index=PLANETS, columns=ELEMENTS)
```
