import matplotlib.pyplot as plt
import numpy as np
import scipy.stats as stats


def task_func(x, y, labels):
    def _to_series_list(data):
        if isinstance(data, np.ndarray):
            if data.ndim == 1:
                return [data]
            return [np.asarray(row) for row in data]
        if isinstance(data, (list, tuple)):
            if len(data) == 0:
                return []
            if any(isinstance(item, (list, tuple, np.ndarray)) for item in data):
                return list(data)
            return [data]
        return [data]

    x_series = _to_series_list(x)
    y_series = _to_series_list(y)

    if len(x_series) != len(y_series):
        raise ValueError("x and y must contain the same number of series.")

    if isinstance(labels, (list, tuple, np.ndarray)):
        label_list = list(labels)
    elif labels is None:
        label_list = []
    else:
        label_list = [labels]

    if len(label_list) < len(x_series):
        label_list.extend(
            [f"Compound {i + 1}" for i in range(len(label_list), len(x_series))]
        )
    else:
        label_list = label_list[: len(x_series)]

    fig, ax = plt.subplots()
    cmap = plt.cm.get_cmap("tab10", max(1, len(x_series)))

    plotted_any = False
    for idx, (xi, yi) in enumerate(zip(x_series, y_series)):
        xi = np.asarray(xi, dtype=float).ravel()
        yi = np.asarray(yi, dtype=float).ravel()

        if xi.size != yi.size:
            raise ValueError(f"Series {idx} in x and y must have the same length.")

        mask = np.isfinite(xi) & np.isfinite(yi)
        xi = xi[mask]
        yi = yi[mask]

        if xi.size == 0:
            continue

        order = np.argsort(xi)
        xi = xi[order]
        yi = yi[order]

        ax.plot(
            xi,
            yi,
            label=label_list[idx],
            color=cmap(idx),
            linewidth=2,
        )
        plotted_any = True

    ax.set_xlabel("x")
    ax.set_ylabel("Density")
    ax.set_title("Normal distributions for chemical compounds")
    ax.grid(True, alpha=0.3)

    if plotted_any:
        ax.legend()

    fig.tight_layout()
    return fig
