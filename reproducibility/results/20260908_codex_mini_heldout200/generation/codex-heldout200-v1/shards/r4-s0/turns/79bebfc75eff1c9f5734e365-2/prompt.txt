import pandas as pd
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt


def task_func(df1, df2, features=["feature1", "feature2", "feature3"], target="target"):
    """
    Merge df1 and df2 on 'id', fit a linear regression model, and plot residuals.

    Returns
    -------
    dict
        {
            'coefficients': list,
            'intercept': float,
            'residuals_plot': matplotlib.axes.Axes
        }
    """
    # Validate merge key
    if "id" not in df1.columns or "id" not in df2.columns:
        raise ValueError("Both input dataframes must contain an 'id' column.")

    # Merge dataframes
    merged = pd.merge(df1, df2, on="id", how="inner", suffixes=("_df1", "_df2"))

    if merged.empty:
        raise ValueError("No matching 'id' values were found between df1 and df2.")

    # Helper to resolve column names after merge
    def resolve_column(col_name):
        if col_name in merged.columns:
            return col_name

        candidates = [f"{col_name}_df1", f"{col_name}_df2"]
        existing = [c for c in candidates if c in merged.columns]

        if len(existing) == 1:
            return existing[0]
        if len(existing) > 1:
            # Deterministic preference if the same-named column appears in both inputs
            return existing[0]

        raise ValueError(f"Required column '{col_name}' was not found in the merged dataframe.")

    # Resolve required columns
    resolved_features = [resolve_column(col) for col in features]
    resolved_target = resolve_column(target)

    required_cols = resolved_features + [resolved_target]

    # Keep only required columns and drop rows with missing values
    data = merged[required_cols].dropna()

    if data.empty:
        raise ValueError("No rows remaining after dropping missing values in required columns.")

    # Ensure all required columns are numeric
    non_numeric = [
        col for col in required_cols
        if not pd.api.types.is_numeric_dtype(data[col])
    ]
    if non_numeric:
        raise ValueError(
            "All features and the target must be numeric. Non-numeric columns: "
            + ", ".join(non_numeric)
        )

    X = data[resolved_features]
    y = data[resolved_target]

    # Fit linear regression
    model = LinearRegression()
    model.fit(X, y)

    # Predictions and residuals
    y_pred = model.predict(X)
    residuals = y - y_pred

    # Residuals plot
    fig, ax = plt.subplots()
    ax.scatter(y_pred, residuals)
    ax.set_title("Residuals Plot")
    ax.set_xlabel("Predicted Values")
    ax.set_ylabel("Residuals")

    return {
        "coefficients": model.coef_.tolist(),
        "intercept": float(model.intercept_),
        "residuals_plot": ax,
    }
