import requests
import os
import zipfile
def task_func(url, destination_directory, headers=None):
    from urllib.parse import urlparse, unquote

    if not isinstance(destination_directory, str) or not destination_directory:
        raise ValueError("destination_directory must be a non-empty string")

    destination_directory = os.path.abspath(destination_directory)

    if os.path.exists(destination_directory) and not os.path.isdir(destination_directory):
        raise NotADirectoryError(f"Destination path is not a directory: {destination_directory}")

    os.makedirs(destination_directory, exist_ok=True)

    parsed = urlparse(url)
    base_name = os.path.basename(parsed.path.rstrip("/"))
    base_name = unquote(base_name)

    if not base_name:
        base_name = "downloaded.zip"

    if not base_name.lower().endswith(".zip"):
        base_name = f"{base_name}.zip"

    zip_path = os.path.join(destination_directory, base_name)

    request_headers = headers if headers is not None else {}

    response = requests.get(url, headers=request_headers, stream=True)
    response.raise_for_status()

    with open(zip_path, "wb") as f:
        for chunk in response.iter_content(chunk_size=8192):
            if chunk:
                f.write(chunk)

    extracted_files = []
    dest_root = os.path.abspath(destination_directory)

    try:
        with zipfile.ZipFile(zip_path, "r") as zf:
            for member in zf.infolist():
                member_name = member.filename

                # Skip directory entries
                if member.is_dir() or member_name.endswith("/"):
                    continue

                # Prevent zip-slip path traversal
                target_path = os.path.abspath(os.path.join(destination_directory, member_name))
                if os.path.commonpath([dest_root, target_path]) != dest_root:
                    raise ValueError(f"Unsafe path in zip archive: {member_name}")

                parent_dir = os.path.dirname(target_path)
                if parent_dir:
                    os.makedirs(parent_dir, exist_ok=True)

                with zf.open(member, "r") as source, open(target_path, "wb") as target:
                    while True:
                        data = source.read(8192)
                        if not data:
                            break
                        target.write(data)

                extracted_files.append(member_name)
    except zipfile.BadZipFile as e:
        raise zipfile.BadZipFile(f"Downloaded file is not a valid ZIP archive: {zip_path}") from e

    return extracted_files
