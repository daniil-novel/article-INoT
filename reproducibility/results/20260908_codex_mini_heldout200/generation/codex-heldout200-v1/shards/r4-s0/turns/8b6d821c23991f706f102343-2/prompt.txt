import cv2
import os
from matplotlib import pyplot as plt

def task_func(image_path='image.jpg', histogram_path='histogram.png'):
    if not os.path.exists(image_path):
        raise FileNotFoundError(f"Image file does not exist at the specified path: {image_path}")

    image = cv2.imread(image_path, cv2.IMREAD_UNCHANGED)
    if image is None:
        raise ValueError(f"Unable to read image file: {image_path}")

    # Prepare image for display and histogram computation
    if image.ndim == 2:
        display_image = image
        gray_image = image
        is_grayscale = True
    elif image.ndim == 3 and image.shape[2] == 4:
        display_image = cv2.cvtColor(image, cv2.COLOR_BGRA2RGBA)
        gray_image = cv2.cvtColor(image, cv2.COLOR_BGRA2GRAY)
        is_grayscale = False
    else:
        display_image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        is_grayscale = False

    fig, (ax_img, ax_hist) = plt.subplots(1, 2, figsize=(10, 4))

    # Display original image
    if is_grayscale:
        ax_img.imshow(display_image, cmap='gray')
    else:
        ax_img.imshow(display_image)
    ax_img.set_title('Original Image')
    ax_img.axis('off')

    # Create histogram of pixel intensities
    ax_hist.hist(gray_image.ravel(), bins=256, range=(0, 256), color='blue', alpha=0.75)
    ax_hist.set_title('Histogram of Pixel Intensities')
    ax_hist.set_xlabel('Intensity')
    ax_hist.set_ylabel('Frequency')

    fig.tight_layout()

    # Ensure output directory exists if one is provided
    output_dir = os.path.dirname(histogram_path)
    if output_dir:
        os.makedirs(output_dir, exist_ok=True)

    # Save the histogram figure as PNG
    fig.savefig(histogram_path, format='png')

    # Display the figure
    plt.show()

    return ax_hist
