import pandas as pd
import numpy as np

def task_func(product_dict, product_keys):
    """
    Build a product profit report and plot profit by product.

    Parameters
    ----------
    product_dict : dict
        Mapping from product name to product data. Each product entry is
        expected to contain quantity, price, and profit values.
    product_keys : iterable
        Product names to include in the report, in the desired order.

    Returns
    -------
    tuple
        (DataFrame, Axes or None)
    """
    cols = ['Product', 'Quantity', 'Price', 'Profit', 'Average Price', 'Average Profit']

    if product_dict is None:
        product_dict = {}

    keys = [] if product_keys is None else list(product_keys)

    records = []
    for key in keys:
        if key not in product_dict:
            continue

        entry = product_dict[key]
        try:
            if isinstance(entry, dict):
                quantity = entry.get('quantity', entry.get('Quantity'))
                price = entry.get('price', entry.get('Price'))
                profit = entry.get('profit', entry.get('Profit'))
            else:
                quantity, price, profit = entry[:3]
        except Exception:
            continue

        values = pd.to_numeric(pd.Series([quantity, price, profit]), errors='coerce')
        if values.isna().any():
            continue

        records.append([key, values.iloc[0], values.iloc[1], values.iloc[2]])

    if not records:
        return pd.DataFrame(columns=cols), None

    df = pd.DataFrame(records, columns=['Product', 'Quantity', 'Price', 'Profit'])

    avg_price = float(np.mean(df['Price'].astype(float)))
    avg_profit = float(np.mean(df['Profit'].astype(float)))

    df['Average Price'] = avg_price
    df['Average Profit'] = avg_profit

    import matplotlib.pyplot as plt

    fig, ax = plt.subplots()
    ax.bar(df['Product'], df['Profit'])
    ax.set_title('Profit by Product')
    ax.set_xlabel('Product')
    ax.set_ylabel('Profit')
    if len(df) > 1:
        plt.setp(ax.get_xticklabels(), rotation=45, ha='right')
    fig.tight_layout()

    return df, ax
