import os
import glob
import shutil

def task_func(directory, archive_dir='archive'):
    errors = []

    source_dir = os.path.abspath(directory)
    target_dir = os.path.abspath(archive_dir)

    if not os.path.isdir(source_dir):
        return False, [f"Source directory does not exist: {directory}"]

    if source_dir == target_dir:
        return False, [f"Source and archive directories must be different: {directory}"]

    try:
        os.makedirs(target_dir, exist_ok=True)
    except Exception as exc:
        return False, [f"Failed to create archive directory {archive_dir}: {exc}"]

    json_files = glob.glob(os.path.join(source_dir, "*.json"))

    for source_path in json_files:
        destination_path = os.path.join(target_dir, os.path.basename(source_path))
        try:
            if os.path.exists(destination_path):
                raise FileExistsError(f"destination already exists: {destination_path}")
            shutil.move(source_path, destination_path)
        except Exception as exc:
            errors.append(f"Failed to move {source_path}: {exc}")

    return len(errors) == 0, errors
