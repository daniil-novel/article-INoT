import cv2
import matplotlib.pyplot as plt
import os

def task_func(image_path, kernel_size):
    if not isinstance(kernel_size, int) or isinstance(kernel_size, bool) or kernel_size <= 0:
        raise ValueError("kernel_size must be a positive integer")

    if not isinstance(image_path, str) or not os.path.isfile(image_path):
        raise FileNotFoundError(f"Image file not found: {image_path}")

    image = cv2.imread(image_path)
    if image is None:
        raise FileNotFoundError(f"Image file not found or could not be read: {image_path}")

    blurred_image = cv2.blur(image, (kernel_size, kernel_size))

    fig, axes = plt.subplots(1, 2, figsize=(10, 5))

    if image.ndim == 2:
        axes[0].imshow(image, cmap="gray")
        axes[1].imshow(blurred_image, cmap="gray")
    else:
        axes[0].imshow(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
        axes[1].imshow(cv2.cvtColor(blurred_image, cv2.COLOR_BGR2RGB))

    axes[0].set_title("Original Image")
    axes[1].set_title("Blurred Image")

    axes[0].axis("off")
    axes[1].axis("off")

    plt.tight_layout()

    return blurred_image, axes[0], axes[1]
