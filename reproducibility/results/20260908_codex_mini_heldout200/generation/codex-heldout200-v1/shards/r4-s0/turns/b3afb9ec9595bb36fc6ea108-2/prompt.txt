import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler

def task_func(P, T):
    P_arr = np.asarray(P)
    T_arr = np.asarray(T)

    if P_arr.ndim != 2:
        raise ValueError("P must be a 2D matrix.")
    if T_arr.ndim != 3:
        raise ValueError("T must be a 3D tensor.")

    if P_arr.size == 0 or T_arr.size == 0:
        raise ValueError("P and T must be non-empty.")

    if np.iscomplexobj(P_arr) or np.iscomplexobj(T_arr):
        raise ValueError("P and T must contain real numeric values.")

    if not np.issubdtype(P_arr.dtype, np.number) or not np.issubdtype(T_arr.dtype, np.number):
        raise ValueError("P and T must contain numeric values.")

    if not np.all(np.isfinite(P_arr)) or not np.all(np.isfinite(T_arr)):
        raise ValueError("P and T must not contain NaN or infinite values.")

    if P_arr.shape[1] != T_arr.shape[0]:
        raise ValueError(
            f"Incompatible shapes for multiplication: P.shape={P_arr.shape}, T.shape={T_arr.shape}. "
            "Require P.shape[1] == T.shape[0]."
        )

    # Matrix-tensor multiplication: contract P's second axis with T's first axis.
    result = np.tensordot(P_arr, T_arr, axes=([1], [0]))

    # Flatten to 2D for scaling: samples x features
    if result.ndim == 1:
        result_2d = result.reshape(-1, 1)
    else:
        result_2d = result.reshape(result.shape[0], -1)

    if result_2d.shape[0] == 0 or result_2d.shape[1] == 0:
        raise ValueError("The multiplication result cannot be empty.")

    scaled = StandardScaler().fit_transform(result_2d)

    columns = [f"feature_{i}" for i in range(scaled.shape[1])]
    return pd.DataFrame(scaled, columns=columns)
