```python
import numpy as np
from sklearn import datasets
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression

def task_func(n_samples=100, n_features=10, random_seed=None):
    if not isinstance(n_samples, int) or not isinstance(n_features, int):
        raise TypeError("n_samples and n_features must be integers.")
    if n_samples < 2:
        raise ValueError("n_samples must be at least 2.")
    if n_features < 1:
        raise ValueError("n_features must be at least 1.")

    X, y = datasets.make_regression(
        n_samples=n_samples,
        n_features=n_features,
        n_informative=n_features,
        noise=0.1,
        random_state=random_seed,
    )

    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=0.2,
        random_state=random_seed,
    )

    model = LinearRegression()
    model.fit(X_train, y_train)

    predictions = model.predict(X_test)
    coefficients = model.coef_
    intercept = float(model.intercept_)
    mse = float(np.mean((y_test - predictions) ** 2))

    return predictions, coefficients, intercept, mse
```
