import pandas as pd
import numpy as np

def task_func(product_dict, product_keys):
    import matplotlib.pyplot as plt

    columns = ['Product', 'Quantity', 'Price', 'Profit', 'Average Price', 'Average Profit']

    def _to_number(value):
        try:
            num = pd.to_numeric(value)
        except Exception:
            return None
        if pd.isna(num):
            return None
        return num

    records = []

    if product_dict is None or product_keys is None:
        return pd.DataFrame(columns=columns), None

    for key in product_keys:
        if key not in product_dict:
            continue

        entry = product_dict[key]
        quantity = price = profit = None

        if hasattr(entry, "items") and not isinstance(entry, (list, tuple, np.ndarray)):
            items = list(entry.items())
            lower_map = {str(k).lower(): v for k, v in items}

            if 'quantity' in lower_map and 'price' in lower_map and 'profit' in lower_map:
                quantity = lower_map['quantity']
                price = lower_map['price']
                profit = lower_map['profit']
            else:
                vals = [v for _, v in items]
                if len(vals) >= 3:
                    quantity, price, profit = vals[:3]
                else:
                    continue
        elif isinstance(entry, (list, tuple, np.ndarray)) and len(entry) >= 3:
            quantity, price, profit = entry[:3]
        else:
            continue

        quantity = _to_number(quantity)
        price = _to_number(price)
        profit = _to_number(profit)

        if quantity is None or price is None or profit is None:
            continue

        records.append({
            'Product': key,
            'Quantity': quantity,
            'Price': price,
            'Profit': profit
        })

    if not records:
        return pd.DataFrame(columns=columns), None

    df = pd.DataFrame(records)
    df['Average Price'] = float(np.mean(df['Price']))
    df['Average Profit'] = float(np.mean(df['Profit']))
    df = df[columns]

    fig, ax = plt.subplots()
    ax.bar(df['Product'], df['Profit'])
    ax.set_xlabel('Product')
    ax.set_ylabel('Profit')
    ax.set_title('Profit by Product')

    return df, ax
