import numpy as np
import itertools

def task_func(data_list, file_name):
    if not data_list:
        with open(file_name, "w", encoding="utf-8"):
            pass
        return []

    def _to_nan(value):
        try:
            return float(value)
        except (TypeError, ValueError):
            return np.nan

    numeric_rows = []
    for row in data_list:
        values = []
        for item in row[1:]:
            values.append(_to_nan(item))
        numeric_rows.append(values)

    columns = list(itertools.zip_longest(*numeric_rows, fillvalue=np.nan))

    if not columns:
        with open(file_name, "w", encoding="utf-8"):
            pass
        return []

    means = []
    for col in columns:
        arr = np.asarray(col, dtype=float)
        if np.all(np.isnan(arr)):
            means.append(np.nan)
        else:
            means.append(float(np.nanmean(arr)))

    with open(file_name, "w", encoding="utf-8") as f:
        for idx, mean in enumerate(means, start=1):
            f.write(f"Position {idx}: {mean}\n")

    return means
