import pandas as pd
from sklearn.preprocessing import StandardScaler

def task_func(df):
    if df.empty:
        return df.copy()

    scaler = StandardScaler()
    df_standardized = pd.DataFrame(
        scaler.fit_transform(df),
        columns=df.columns,

    )
