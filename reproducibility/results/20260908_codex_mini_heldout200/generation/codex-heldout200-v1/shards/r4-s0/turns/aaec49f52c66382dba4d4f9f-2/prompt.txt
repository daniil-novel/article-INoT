import requests
import os
import zipfile

def task_func(url, destination_directory, headers=None):
    os.makedirs(destination_directory, exist_ok=True)

    # Choose a local filename for the downloaded ZIP and keep it after extraction.
    base_name = url.split("?")[0].split("#")[0].rstrip("/").split("/")[-1]
    if not base_name:
        base_name = "downloaded.zip"
    elif not base_name.lower().endswith(".zip"):
        base_name += ".zip"

    zip_path = os.path.join(destination_directory, base_name)

    # Download the ZIP file in streaming mode.
    response = requests.get(url, headers=headers, stream=True)
    response.raise_for_status()

    with open(zip_path, "wb") as f:
        for chunk in response.iter_content(chunk_size=8192):
            if chunk:
                f.write(chunk)

    extracted_files = []
    base_dir = os.path.abspath(destination_directory)

    # Validate and extract safely.
    with zipfile.ZipFile(zip_path, "r") as zf:
        for member in zf.infolist():
            member_name = member.filename

            # Skip directory entries.
            if member_name.endswith("/"):
                continue

            # Prevent path traversal / zip slip.
            target_path = os.path.abspath(os.path.join(destination_directory, member_name))
            if os.path.commonpath([base_dir, target_path]) != base_dir:
                raise ValueError("Unsafe ZIP entry path detected: {}".format(member_name))

            os.makedirs(os.path.dirname(target_path), exist_ok=True)

            with zf.open(member, "r") as source, open(target_path, "wb") as target:
                while True:
                    chunk = source.read(8192)
                    if not chunk:
                        break
                    target.write(chunk)

            extracted_files.append(os.path.relpath(target_path, destination_directory))

    return extracted_files
