import numpy as np
from scipy.stats import norm

def task_func(data: np.ndarray, threshold: float = 2.0) -> list:
    data = np.asarray(data, dtype=float)

    if data.size == 0:
        raise ValueError("data must be a non-empty 1D array")

    mean, std = norm.fit(data)
    var = float(std ** 2)

    if np.isclose(std, 0.0):
        return [], float(mean), var

    z_scores = (data - mean) / std
    outliers = np.where(np.abs(z_scores) > threshold)[0].tolist()

    return outliers, float(mean), var
