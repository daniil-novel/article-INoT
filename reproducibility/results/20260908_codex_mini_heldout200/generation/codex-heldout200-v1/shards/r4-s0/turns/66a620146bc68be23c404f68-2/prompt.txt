import re
import numpy as np
from scipy.stats import ttest_rel

def task_func(text1, text2):
    words1 = re.findall(r"\b\w+\b", text1)
    words2 = re.findall(r"\b\w+\b", text2)

    if len(words1) != len(words2) or len(words1) < 2:
        return np.nan, np.nan

    sample1 = np.asarray([len(word) for word in words1], dtype=float)
    sample2 = np.asarray([len(word) for word in words2], dtype=float)

    t_statistic, p_value = ttest_rel(sample1, sample2)
    return float(t_statistic), float(p_value)
