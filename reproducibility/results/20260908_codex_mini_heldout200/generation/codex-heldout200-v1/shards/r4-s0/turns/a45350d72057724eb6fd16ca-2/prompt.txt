```python
from datetime import datetime, timedelta
from random import randint
import matplotlib.pyplot as plt
import pandas as pd

TEMP_CATEGORIES = ['Cold', 'Normal', 'Hot']
FILE_PATH = 'custom_data.csv'


def task_func(hours, file_path=FILE_PATH):
    if not isinstance(hours, int):
        raise TypeError("hours must be an integer")
    if hours <= 0:
        raise ValueError("hours must be greater than 0")

    data = {
        'Time': [],
        'Temperature': [],
        'Category': []
    }

    start_time = datetime.now()

    for i in range(hours):
        temp_time = start_time + timedelta(hours=i)
        temperature = randint(-10, 40)

        if temperature <= 10:
            category = TEMP_CATEGORIES[0]
        elif temperature <= 25:
            category = TEMP_CATEGORIES[1]
        else:
            category = TEMP_CATEGORIES[2]

        data['Time'].append(temp_time)
        data['Temperature'].append(temperature)
        data['Category'].append(category)

    df = pd.DataFrame(data)
    df.to_csv(file_path, index=False)

    fig, ax = plt.subplots()
    ax.plot(df['Time'], df['Temperature'], marker='o', linestyle='-')
    ax.set_xlabel('Time')
    ax.set_ylabel('Temperature')
    ax.set_title('Temperature Over Time')
    fig.autofmt_xdate()

    return file_path, ax
```
