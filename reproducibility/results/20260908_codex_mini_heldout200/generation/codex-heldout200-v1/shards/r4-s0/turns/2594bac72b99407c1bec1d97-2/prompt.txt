import pandas as pd
import numpy as np
CATEGORIES = ["Electronics", "Clothing", "Home Decor", "Automotive", "Books"]

def task_func(s1, s2):
    import matplotlib.pyplot as plt

    # Normalize inputs to Series and align to the predefined category order
    s1 = pd.Series(s1).reindex(CATEGORIES)
    s2 = pd.Series(s2).reindex(CATEGORIES)

    # Coerce values to numeric; invalid entries become NaN
    s1 = pd.to_numeric(s1, errors="coerce")
    s2 = pd.to_numeric(s2, errors="coerce")

    # Filter categories where both stores exceed the threshold
    threshold = 200
    mask = (s1 > threshold) & (s2 > threshold)

    if not mask.any():
        return None, 0.0

    cats = np.array(CATEGORIES)[mask.to_numpy()]
    v1 = s1[mask].to_numpy(dtype=float)
    v2 = s2[mask].to_numpy(dtype=float)

    # Euclidean distance
    dist = float(np.sqrt(np.sum((v1 - v2) ** 2)))

    # Create grouped bar plot
    x = np.arange(len(cats))
    width = 0.35

    fig, ax = plt.subplots()
    ax.bar(x - width / 2, v1, width, label="Store 1")
    ax.bar(x + width / 2, v2, width, label="Store 2")

    ax.set_xticks(x)
    ax.set_xticklabels(cats)
    ax.set_xlabel("Category")
    ax.set_ylabel("Sales")
    ax.set_title("Sales comparison for categories with sales > 200")
    ax.legend()

    return ax, dist
