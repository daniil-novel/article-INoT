import pandas as pd
import matplotlib.pyplot as plt
import math

def task_func(data_dict):
    df = pd.DataFrame(data_dict)
    axes_list = []

    def _is_uniform_from_counts(counts):
        counts = [float(c) for c in counts if c is not None]
        total = sum(counts)
        k = len(counts)

        if total == 0:
            return True
        if k <= 1:
            return False

        # For very small samples, use a simple exact-ish heuristic.
        if total < 5 * k:
            return max(counts) - min(counts) <= 1

        expected = total / k
        chi2 = sum((c - expected) ** 2 / expected for c in counts)
        df_chi = k - 1
        if df_chi <= 0:
            return True

        # Wilson-Hilferty approximation for chi-square survival probability.
        z = ((chi2 / df_chi) ** (1 / 3) - (1 - 2 / (9 * df_chi))) / math.sqrt(2 / (9 * df_chi))
        p_value = 0.5 * math.erfc(z / math.sqrt(2))
        return p_value >= 0.05

    for column in df.columns:
        series = df[column]
        fig, ax = plt.subplots()

        non_uniform = False

        if pd.api.types.is_numeric_dtype(series):
            values = pd.to_numeric(series, errors="coerce")
            values = values[~values.isin([math.inf, -math.inf])].dropna()

            if values.empty:
                ax.set_title(f"Histogram of {column}")
                ax.set_xlabel(column)
                ax.set_ylabel("Count")
                axes_list.append(ax)
                continue

            unique_count = values.nunique(dropna=True)

            if unique_count <= 10:
                counts_series = values.value_counts().sort_index()
                x = range(len(counts_series))
                ax.bar(x, counts_series.values, edgecolor="black")
                if len(counts_series) <= 20:
                    ax.set_xticks(list(x))
                    ax.set_xticklabels([str(v) for v in counts_series.index], rotation=45, ha="right")
                non_uniform = not _is_uniform_from_counts(counts_series.values.tolist())
            else:
                bins = max(2, min(10, int(len(values) ** 0.5)))
                counts, _, _ = ax.hist(values, bins=bins, edgecolor="black")
                non_uniform = not _is_uniform_from_counts(counts.tolist())
        else:
            values = series.dropna()

            if values.empty:
                ax.set_title(f"Histogram of {column}")
                ax.set_xlabel(column)
                ax.set_ylabel("Count")
                axes_list.append(ax)
                continue

            counts_series = values.value_counts().sort_index()
            x = range(len(counts_series))
            ax.bar(x, counts_series.values, edgecolor="black")
            if len(counts_series) <= 20:
                ax.set_xticks(list(x))
                ax.set_xticklabels([str(v) for v in counts_series.index], rotation=45, ha="right")
            non_uniform = not _is_uniform_from_counts(counts_series.values.tolist())

        ax.set_title(f"Histogram of {column}")
        ax.set_xlabel(column)
        ax.set_ylabel("Count")
        fig.tight_layout()

        if non_uniform:
            print(f"Column '{column}' is not uniformly distributed.")

        axes_list.append(ax)

    return axes_list
