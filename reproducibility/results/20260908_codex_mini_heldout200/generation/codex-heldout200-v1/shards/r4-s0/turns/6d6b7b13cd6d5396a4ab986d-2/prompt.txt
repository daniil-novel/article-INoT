import pandas as pd
import seaborn as sns


def task_func(goals, penalties):
    def _as_sequence(value):
        if isinstance(value, (str, bytes)):
            return [value]
        try:
            return list(value)
        except TypeError:
            return [value]

    goals_list = _as_sequence(goals)
    penalties_list = _as_sequence(penalties)

    if len(goals_list) != len(penalties_list):
        raise ValueError("goals and penalties must have the same length")

    if len(goals_list) == 0:
        empty_df = pd.DataFrame(columns=["Team", "Goals", "Penalties"])
        return empty_df, None

    goals_num = pd.to_numeric(goals_list, errors="raise")
    penalties_num = pd.to_numeric(penalties_list, errors="raise")

    df = pd.DataFrame(
        {
            "Team": [f"Team {i + 1}" for i in range(len(goals_list))],
            "Goals": goals_num,
            "Penalties": penalties_num,
        }
    )

    plot_kwargs = {"alpha": 0.75, "s": 50}
    if len(df) <= 10:
        grid = sns.pairplot(
            df,
            vars=["Goals", "Penalties"],
            hue="Team",
            diag_kind="hist",
            corner=True,
            palette="tab10",
            plot_kws=plot_kwargs,
        )
    else:
        grid = sns.pairplot(
            df,
            vars=["Goals", "Penalties"],
            diag_kind="hist",
            corner=True,
            plot_kws=plot_kwargs,
        )

    return df, grid
