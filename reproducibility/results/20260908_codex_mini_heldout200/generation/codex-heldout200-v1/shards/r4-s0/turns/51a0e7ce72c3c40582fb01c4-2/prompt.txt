import zipfile
import os
import re
import shutil

def task_func(source_dir: str, target_dir: str, archive_name: str = 'archive.zip') -> str:
    if not os.path.isdir(source_dir):
        raise NotADirectoryError(f"Source directory does not exist or is not a directory: {source_dir}")

    os.makedirs(target_dir, exist_ok=True)

    safe_archive_name = os.path.basename(archive_name) if archive_name else "archive.zip"
    if not safe_archive_name:
        safe_archive_name = "archive.zip"

    archive_path = os.path.join(target_dir, safe_archive_name)
    source_root = os.path.abspath(source_dir)
    archive_abs = os.path.abspath(archive_path)

    processed_files = []
    processed_pattern = re.compile(r"_processed(?:\.[^./\\]+)?$")

    for root, _, files in os.walk(source_root):
        for filename in files:
            # Match files whose stem ends with "_processed"
            stem, _ = os.path.splitext(filename)
            if not processed_pattern.search(filename) and not stem.endswith("_processed"):
                continue

            file_path = os.path.abspath(os.path.join(root, filename))
            if file_path == archive_abs:
                continue

            processed_files.append(file_path)

    processed_files.sort()

    with zipfile.ZipFile(archive_path, mode="w", compression=zipfile.ZIP_DEFLATED) as zf:
        for file_path in processed_files:
            arcname = os.path.relpath(file_path, start=source_root)
            zf.write(file_path, arcname=arcname)

    return archive_path
