import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

CATEGORIES = ["Electronics", "Clothing", "Home Decor", "Automotive", "Books"]

def task_func(s1, s2):
    s1 = pd.Series(s1)
    s2 = pd.Series(s2)

    v1 = pd.to_numeric(s1.reindex(CATEGORIES), errors="coerce")
    v2 = pd.to_numeric(s2.reindex(CATEGORIES), errors="coerce")

    mask = (v1 > 200) & (v2 > 200)
    selected_categories = [cat for cat, keep in zip(CATEGORIES, mask) if bool(keep)]

    if not selected_categories:
        return None, 0.0

    vals1 = v1[mask].to_numpy(dtype=float)
    vals2 = v2[mask].to_numpy(dtype=float)
    distance = float(np.linalg.norm(vals1 - vals2))

    x = np.arange(len(selected_categories))
    width = 0.35

    fig, ax = plt.subplots()
    ax.bar(x - width / 2, vals1, width, label="Store 1")
    ax.bar(x + width / 2, vals2, width, label="Store 2")
    ax.set_xticks(x)
    ax.set_xticklabels(selected_categories, rotation=45, ha="right")
    ax.set_xlabel("Category")
    ax.set_ylabel("Sales")
    ax.set_title("Sales Comparison for Categories with Sales > 200")
    ax.legend()
    fig.tight_layout()

    return ax, distance
