```python
import subprocess
import os
import sys
import glob

def task_func(directory_path):
    """
    Find and run all .bat files in a given directory.

    Returns:
        list[tuple[str, int | None]]: (file_name, exit_code) for each batch file.
                                      exit_code is None if execution failed.
    """
    if not isinstance(directory_path, str) or not os.path.isdir(directory_path):
        return []

    try:
        entries = sorted(os.listdir(directory_path))
    except OSError:
        return []

    results = []
    for name in entries:
        if not name.lower().endswith(".bat"):
            continue

        full_path = os.path.join(directory_path, name)
        if not os.path.isfile(full_path):
            continue

        exit_code = None
        try:
            if os.name == "nt":
                completed = subprocess.run(
                    ["cmd", "/c", full_path],
                    stdout=subprocess.DEVNULL,
                    stderr=subprocess.DEVNULL,
                    check=False,
                )
            else:
                completed = subprocess.run(
                    [full_path],
                    stdout=subprocess.DEVNULL,
                    stderr=subprocess.DEVNULL,
                    check=False,
                )
            exit_code = completed.returncode
        except (OSError, subprocess.SubprocessError, ValueError):
            exit_code = None

        results.append((name, exit_code))

    return results
```
