import ctypes
import hashlib
import binascii
import os

def task_func(filepath):
    if filepath is None:
        raise ValueError("filepath must not be None")

    path = os.fspath(filepath)
    if not isinstance(path, str) or not path.strip():
        raise ValueError("filepath must be a non-empty string or path-like object")

    # Load the DLL; do not invoke any exported functions.
    ctypes.CDLL(path)

    md5_hash = hashlib.md5()
    sha256_hash = hashlib.sha256()

    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            md5_hash.update(chunk)
            sha256_hash.update(chunk)

    md5_hex = binascii.hexlify(md5_hash.digest()).decode("ascii")
    sha256_hex = binascii.hexlify(sha256_hash.digest()).decode("ascii")

    print(md5_hex)
    print(sha256_hex)

    return os.path.basename(path)
