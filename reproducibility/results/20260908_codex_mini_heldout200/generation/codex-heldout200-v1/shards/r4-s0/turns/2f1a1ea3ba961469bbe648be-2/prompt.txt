- Avoid assuming all images are BGR color images

import os
import cv2
import numpy as np
import matplotlib.pyplot as plt

def task_func(image_path, kernel_size):
    if isinstance(kernel_size, bool) or not isinstance(kernel_size, int) or kernel_size <= 0:
        raise ValueError("kernel_size must be a positive integer")

    if not isinstance(image_path, str) or not os.path.isfile(image_path):
        raise FileNotFoundError(f"Image file not found: {image_path}")

    image = cv2.imread(image_path, cv2.IMREAD_UNCHANGED)
    if image is None:
        raise FileNotFoundError(f"Image file not found: {image_path}")

    blurred_image = cv2.blur(image, (kernel_size, kernel_size))

    fig, (ax_original, ax_blurred) = plt.subplots(1, 2, figsize=(10, 5))

    def _prepare_for_plot(img):
        if img.ndim == 2:
            return img, "gray"
        if img.ndim == 3 and img.shape[2] == 4:
            return cv2.cvtColor(img, cv2.COLOR_BGRA2RGBA), None
        if img.ndim == 3 and img.shape[2] == 3:
            return cv2.cvtColor(img, cv2.COLOR_BGR2RGB), None
        return img, None

    original_plot, original_cmap = _prepare_for_plot(image)
    blurred_plot, blurred_cmap = _prepare_for_plot(blurred_image)

    ax_original.imshow(original_plot, cmap=original_cmap)
    ax_original.set_title("Original Image")
    ax_original.axis("off")

    ax_blurred.imshow(blurred_plot, cmap=blurred_cmap)
    ax_blurred.set_title("Blurred Image")
    ax_blurred.axis("off")

    fig.tight_layout()

    return blurred_image, ax_original, ax_blurred
