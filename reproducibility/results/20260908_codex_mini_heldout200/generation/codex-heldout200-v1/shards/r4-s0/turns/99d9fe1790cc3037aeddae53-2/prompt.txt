import pandas as pd
import os

def task_func(data_dir: str, csv_files: list) -> pd.DataFrame:
    if not csv_files:
        return pd.DataFrame()

    dfs = [pd.read_csv(os.path.join(data_dir, csv_file)) for csv_file in csv_files]
    return pd.concat(dfs, ignore_index=True)
