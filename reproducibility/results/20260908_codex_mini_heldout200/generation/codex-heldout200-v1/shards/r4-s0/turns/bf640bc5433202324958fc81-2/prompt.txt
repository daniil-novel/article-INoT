import time
from datetime import datetime
import random
import matplotlib.pyplot as plt

# Constants
DATE_FORMAT = "%Y-%m-%d %H:%M:%S"

def task_func(n, output_path=None):
    if not isinstance(n, int) or isinstance(n, bool):
        raise TypeError("n must be an integer")
    if n < 0:
        raise ValueError("n must be non-negative")

    now_ts = int(time.time())
    timestamps = [random.randint(0, now_ts) for _ in range(n)]
    formatted_timestamps = [
        datetime.utcfromtimestamp(ts).strftime(DATE_FORMAT) for ts in timestamps
    ]

    plt.figure()
    bins = min(50, max(1, int(n ** 0.5))) if n > 0 else 1
    plt.hist(timestamps, bins=bins, edgecolor="black")
    plt.title("Distribution of Generated Unix Timestamps")
    plt.xlabel("Unix timestamp")
    plt.ylabel("Frequency")
    plt.tight_layout()

    if output_path:
        plt.savefig(output_path)
    else:
        plt.show()

    plt.close()
    return formatted_timestamps
