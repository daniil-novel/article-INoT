import numpy as np
import matplotlib.pyplot as plt
from scipy.fft import fft

ANGLES = np.arange(0, 2*np.pi, 0.01)

def task_func(n_waves, seed=0):
    """
    Generate n sine waves with increasing frequency, compute the FFT of their
    mixed signal, and plot a histogram of FFT magnitudes.

    Parameters
    ----------
    n_waves : int
        Number of sine waves to generate.
    seed : int, optional
        Unused, kept for API compatibility.

    Returns
    -------
    list of np.ndarray
        The generated sine waves.
    np.ndarray
        FFT data of the mixed signal.
    matplotlib.axes.Axes or None
        Axes object containing the histogram plot, or None if n_waves < 1.
    """
    try:
        n_waves = int(n_waves)
    except (TypeError, ValueError):
        return [], np.array([]), None

    if n_waves < 1:
        return [], np.array([]), None

    np.random.seed(seed)

    waves = [np.sin((i + 1) * ANGLES) for i in range(n_waves)]
    mixed_signal = np.sum(np.vstack(waves), axis=0)

    fft_data = fft(mixed_signal)

    fig, ax = plt.subplots()
    ax.hist(np.abs(fft_data), bins="auto")
    ax.set_title("Histogram of FFT Magnitudes")
    ax.set_xlabel("Magnitude")
    ax.set_ylabel("Frequency")

    return waves, fft_data, ax
