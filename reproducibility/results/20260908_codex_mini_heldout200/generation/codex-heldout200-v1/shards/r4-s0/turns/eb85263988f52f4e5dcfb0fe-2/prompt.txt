import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

def task_func(num_groups=5, data_size=5, labels=None):
    """
    Generate random data, plot it as a stacked bar chart, save the figure,
    and return the figure, data, and filename.
    """
    # Validate num_groups
    if isinstance(num_groups, bool) or not isinstance(num_groups, (int, np.integer)):
        raise TypeError("num_groups must be an integer.")
    if num_groups <= 0:
        raise ValueError("num_groups must be greater than 0.")

    # Validate data_size
    if isinstance(data_size, bool) or not isinstance(data_size, (int, np.integer)):
        raise TypeError("data_size must be an integer.")
    if data_size <= 0:
        raise ValueError("data_size must be greater than 0.")

    num_groups = int(num_groups)
    data_size = int(data_size)

    # Validate / generate labels
    if labels is None:
        labels = [f"Group {i + 1}" for i in range(num_groups)]
    else:
        if isinstance(labels, (str, bytes)) or not hasattr(labels, "__len__"):
            raise TypeError("labels must be a list-like sequence of length num_groups.")
        if len(labels) != num_groups:
            raise ValueError("labels must have the same length as num_groups.")
        labels = list(labels)

    # Generate random data
    data = np.random.rand(data_size, num_groups)
    df = pd.DataFrame(data, columns=labels)

    # Create stacked bar chart
    fig, ax = plt.subplots(figsize=(max(6, num_groups * 1.2), max(4, data_size * 0.8)))
    df.plot(kind="bar", stacked=True, ax=ax)

    ax.set_title("Stacked Bar Chart of Random Data")
    ax.set_xlabel("Row Index")
    ax.set_ylabel("Value")
    ax.legend(title="Categories", bbox_to_anchor=(1.05, 1), loc="upper left")

    fig.tight_layout()

    # Save figure
    filename = "test_plot.png"
    fig.savefig(filename, bbox_inches="tight")

    return fig, df, filename
