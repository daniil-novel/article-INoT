import numpy as np
import pandas as pd

def task_func(matrix1, matrix2):
    arr1 = np.asarray(matrix1)
    arr2 = np.asarray(matrix2)

    if arr1.ndim != 2 or arr2.ndim != 2:
        raise ValueError("Both inputs must be 2D arrays.")

    if arr1.shape[0] != arr2.shape[0]:
        raise ValueError("The two matrices must have the same number of rows.")

    combined = np.concatenate((arr1, arr2), axis=1)
    df = pd.DataFrame(combined)
