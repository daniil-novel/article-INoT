import itertools
import math
from pandas import Series

def task_func(elements, subset_size, top_n=2):
    if subset_size == 0 or subset_size > len(elements) or subset_size < 0:
        return 1, Series([], dtype="int64")

    subset_sums = [sum(subset) for subset in itertools.combinations(elements, subset_size)]
    product_of_sums = math.prod(subset_sums) if subset_sums else 1

    if top_n <= 0:
        top_sums = []
    else:
        top_sums = sorted(subset_sums, reverse=True)[:top_n]

    return product_of_sums, Series(top_sums)
