```python
import pandas as pd
import statistics
import random

def task_func(students, subjects, seed=None):
    if seed is not None:
        random.seed(seed)

    columns = ["Student"] + list(subjects) + ["Average Grade"]
    rows = []

    for student in students:
        grades = [random.randint(0, 100) for _ in subjects]
        average = statistics.mean(grades) if grades else None
        rows.append([student] + grades + [average])

    return pd.DataFrame(rows, columns=columns)
```
