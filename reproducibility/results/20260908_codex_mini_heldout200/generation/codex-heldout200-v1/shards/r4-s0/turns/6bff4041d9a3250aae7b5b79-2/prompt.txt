import numpy as np
import pandas as pd
from sklearn.decomposition import PCA

def task_func(array, seed=None):
    arr = np.asarray(array)

    if arr.ndim != 2:
        raise ValueError("Input array must be 2D.")

    n_samples, n_features = arr.shape
    if n_samples == 0 or n_features == 0 or arr.size == 0:
        return pd.DataFrame(columns=["PC1", "PC2"])

    rng = np.random.default_rng(seed)
    perm = rng.permutation(n_features)
    shuffled = arr[:, perm]

    n_components = min(2, n_samples, n_features)
    transformed = PCA(n_components=n_components).fit_transform(shuffled)

    return pd.DataFrame(transformed, columns=["PC1", "PC2"][:n_components])
