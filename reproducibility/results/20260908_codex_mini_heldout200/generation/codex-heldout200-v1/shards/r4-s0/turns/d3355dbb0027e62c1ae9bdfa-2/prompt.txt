import pandas as pd
import matplotlib.pyplot as plt
from random import randint


def task_func(num_types=5, integer_range=(0, 100)):
    """
    Generate a DataFrame of random integers and plot it as a horizontal stacked bar chart.

    Parameters
    ----------
    num_types : int, optional
        Number of categories/rows to generate.
    integer_range : tuple, optional
        Inclusive lower and upper bounds for random integers.

    Returns
    -------
    tuple
        (fig, ax) matplotlib Figure and Axes objects.
    """
    # Validate num_types
    if not isinstance(num_types, int) or num_types <= 0:
        raise ValueError("num_types must be a positive integer.")

    # Validate integer_range
    if (
        not isinstance(integer_range, (tuple, list))
        or len(integer_range) != 2
    ):
        raise ValueError("integer_range must be a tuple or list of length 2.")

    low, high = integer_range
    if not isinstance(low, int) or not isinstance(high, int):
        raise ValueError("integer_range bounds must be integers.")

    if low > high:
        low, high = high, low

    # Generate category labels
    categories = [f"Type {i + 1}" for i in range(num_types)]

    # Create a DataFrame of random integers:
    # one row per category, one column per stacked segment
    data = {
        f"Segment {j + 1}": [randint(low, high) for _ in range(num_types)]
        for j in range(num_types)
    }
    df = pd.DataFrame(data, index=categories)

    # Plot horizontal stacked bar chart
    fig, ax = plt.subplots(figsize=(8, max(4, num_types * 0.6)))

    left = [0] * num_types
    y_positions = range(num_types)

    for column in df.columns:
        values = df[column].tolist()
        ax.barh(y_positions, values, left=left, label=column)
        left = [l + v for l, v in zip(left, values)]

    ax.set_yticks(list(y_positions))
    ax.set_yticklabels(categories)
    ax.set_xlabel("Values")
    ax.set_ylabel("Categories")
    ax.set_title("Horizontal Stacked Bar Chart")
    ax.legend(title="Segments", loc="best")

    fig.tight_layout()
    return fig, ax
