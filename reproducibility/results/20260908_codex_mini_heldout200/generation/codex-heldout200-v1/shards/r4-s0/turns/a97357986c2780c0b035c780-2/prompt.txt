import pandas as pd
import matplotlib.pyplot as plt
from random import randint

def task_func(num_types=5, integer_range=(0, 100)):
    """
    Generate a DataFrame of random integers and plot it as a horizontal
    stacked bar chart.

    Parameters
    ----------
    num_types : int, default=5
        Number of category rows to generate.
    integer_range : tuple[int, int], default=(0, 100)
        Inclusive range used for random integer generation.

    Returns
    -------
    tuple
        (fig, ax) where fig is a matplotlib Figure and ax is an Axes.
    """
    # Validate num_types
    if not isinstance(num_types, int) or isinstance(num_types, bool) or num_types <= 0:
        raise ValueError("num_types must be a positive integer.")

    # Validate integer_range
    if (
        not isinstance(integer_range, (tuple, list))
        or len(integer_range) != 2
        or not all(isinstance(v, int) and not isinstance(v, bool) for v in integer_range)
    ):
        raise ValueError("integer_range must be a 2-item tuple/list of integers.")

    low, high = integer_range
    if low > high:
        raise ValueError("integer_range must satisfy low <= high.")

    # Build category labels
    categories = [f"Category {i + 1}" for i in range(num_types)]

    # Choose a reasonable number of stacked segments
    num_segments = max(2, min(5, num_types))
    segment_labels = [f"Segment {i + 1}" for i in range(num_segments)]

    # Generate random data
    data = {
        label: [randint(low, high) for _ in range(num_types)]
        for label in segment_labels
    }
    df = pd.DataFrame(data, index=categories)

    # Plot horizontal stacked bar chart
    fig, ax = plt.subplots(figsize=(8, max(4, num_types * 0.6)))
    df.plot(kind="barh", stacked=True, ax=ax)

    # Polish the chart
    ax.set_xlabel("Value")
    ax.set_ylabel("Category")
    ax.set_title("Horizontal Stacked Bar Chart of Random Integer Values")
    ax.legend(title="Segments", bbox_to_anchor=(1.02, 1), loc="upper left")
    fig.tight_layout()

    return fig, ax
