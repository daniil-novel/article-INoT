import hashlib
import binascii

def task_func(salt, cursor):
    if not isinstance(salt, str):
        raise TypeError

    def quote_ident(name):
        return '"' + name.replace('"', '""') + '"'

    def hex_sha256(text):
        digest = hashlib.sha256(text.encode("utf-8")).digest()
        return binascii.hexlify(digest).decode("ascii")

    tables = []
    for row in cursor.execute(
        "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
    ).fetchall():
        tables.append(row[0])

    def table_priority(name):
        lname = name.lower()
        if lname in ("users", "user"):
            return (0, lname)
        if "user" in lname:
            return (1, lname)
        return (2, lname)

    for table in sorted(tables, key=table_priority):
        info = cursor.execute(f"PRAGMA table_info({quote_ident(table)})").fetchall()
        if not info:
            continue

        columns = [row[1] for row in info]
        password_col = None
        for candidate in ("password", "passwd", "pwd"):
            if candidate in columns:
                password_col = candidate
                break
        if password_col is None:
            continue

        pk_cols = [row[1] for row in info if row[5]]
        if pk_cols:
            select_cols = ", ".join(quote_ident(col) for col in pk_cols + [password_col])
            rows = cursor.execute(
                f"SELECT {select_cols} FROM {quote_ident(table)}"
            ).fetchall()

            where_clause = " AND ".join(f"{quote_ident(col)} = ?" for col in pk_cols)
            update_sql = (
                f"UPDATE {quote_ident(table)} "
                f"SET {quote_ident(password_col)} = ? "
                f"WHERE {where_clause}"
            )

            updated = 0
            for row in rows:
                old_password = row[-1]
                if old_password is None:
                    old_password = ""
                hashed = hex_sha256(salt + str(old_password))
                params = (hashed, *row[:-1])
                cursor.execute(update_sql, params)
                updated += 1
        else:
            rows = cursor.execute(
                f"SELECT rowid, {quote_ident(password_col)} FROM {quote_ident(table)}"
            ).fetchall()
            update_sql = (
                f"UPDATE {quote_ident(table)} "
                f"SET {quote_ident(password_col)} = ? "
                f"WHERE rowid = ?"
            )

            updated = 0
            for row in rows:
                rowid, old_password = row
                if old_password is None:
                    old_password = ""
                hashed = hex_sha256(salt + str(old_password))
                cursor.execute(update_sql, (hashed, rowid))
                updated += 1

        try:
            cursor.connection.commit()
        except Exception:
            pass

        return updated
