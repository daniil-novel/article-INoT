import pandas as pd
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

def task_func(df1, df2, features=["feature1", "feature2", "feature3"], target="target"):
    """
    Merge df1 and df2 on 'id', fit a linear regression model, and plot residuals.

    Returns
    -------
    dict
        {
            "coefficients": list,
            "intercept": float,
            "residuals_plot": matplotlib.axes.Axes
        }
    """
    if "id" not in df1.columns:
        raise KeyError("df1 must contain an 'id' column.")
    if "id" not in df2.columns:
        raise KeyError("df2 must contain an 'id' column.")
    if not features:
        raise ValueError("'features' must contain at least one column name.")

    merged = pd.merge(df1, df2, on="id", how="inner", suffixes=("_df1", "_df2"))

    if merged.empty:
        raise ValueError("No overlapping 'id' values were found after merging the dataframes.")

    def resolve_column(name):
        if name in merged.columns:
            return name

        c1 = f"{name}_df1"
        c2 = f"{name}_df2"
        has_c1 = c1 in merged.columns
        has_c2 = c2 in merged.columns

        if has_c1 and not has_c2:
            return c1
        if has_c2 and not has_c1:
            return c2
        if has_c1 and has_c2:
            raise ValueError(
                f"Column '{name}' is present in both dataframes and became ambiguous after merge."
            )
        raise KeyError(f"Required column '{name}' was not found after merging.")

    resolved_features = [resolve_column(col) for col in features]
    resolved_target = resolve_column(target)

    data = merged[resolved_features + [resolved_target]].copy()
    data = data.apply(pd.to_numeric, errors="coerce")
    data = data.dropna()

    if data.empty:
        raise ValueError("No valid rows remain after removing missing or non-numeric values.")
    if len(data) < 2:
        raise ValueError("At least 2 valid rows are required to fit a linear regression model.")

    X = data[resolved_features]
    y = data[resolved_target]

    model = LinearRegression()
    model.fit(X, y)

    predicted_values = model.predict(X)
    residuals = y - predicted_values

    fig, ax = plt.subplots()
    ax.scatter(predicted_values, residuals)
    ax.axhline(0, color="red", linestyle="--", linewidth=1)
    ax.set_title("Residuals Plot")
    ax.set_xlabel("Predicted Values")
    ax.set_ylabel("Residuals")

    return {
        "coefficients": model.coef_.tolist(),
        "intercept": float(model.intercept_),
        "residuals_plot": ax,
    }
