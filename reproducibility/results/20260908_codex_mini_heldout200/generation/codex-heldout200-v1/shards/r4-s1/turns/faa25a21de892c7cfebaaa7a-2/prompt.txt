import math
import statistics
import numpy as np

def task_func(input_list):
    # Normalize input to numeric values
    try:
        values = [float(x) for x in input_list]
    except Exception as exc:
        raise TypeError("input_list must contain only numeric values") from exc

    # Handle empty input safely
    if not values:
        return (0, 0, 0, 0, 0, 0)

    # Sort ascending
    sorted_values = sorted(values)

    # Statistics for the sorted list
    mean_sorted = round(statistics.mean(sorted_values))
    median_sorted = round(statistics.median(sorted_values))
    modes_sorted = statistics.multimode(sorted_values)
    mode_sorted = round(min(modes_sorted))

    # FFT magnitudes, rounded to nearest integer before statistics
    fft_magnitudes = np.abs(np.fft.fft(np.array(values, dtype=float)))
    fft_magnitudes_rounded = [round(x) for x in fft_magnitudes]

    # Statistics for FFT magnitudes
    mean_fft = round(statistics.mean(fft_magnitudes_rounded))
    median_fft = round(statistics.median(fft_magnitudes_rounded))
    modes_fft = statistics.multimode(fft_magnitudes_rounded)
    mode_fft = round(min(modes_fft))

    return (mean_sorted, median_sorted, mode_sorted, mean_fft, median_fft, mode_fft)
