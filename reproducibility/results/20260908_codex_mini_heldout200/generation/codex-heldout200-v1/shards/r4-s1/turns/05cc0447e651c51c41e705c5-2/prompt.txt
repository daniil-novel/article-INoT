import subprocess
import time
import json
import platform

LOGFILE_PATH = "logfile.log"


def _first_number(text):
    number = []
    started = False
    for ch in text:
        if ch.isdigit() or ch == ".":
            number.append(ch)
            started = True
        elif started:
            break
    if not number:
        return None
    try:
        return float("".join(number))
    except ValueError:
        return None


def _get_cpu_usage(system_name):
    try:
        if system_name == "Windows":
            result = subprocess.run(
                ["wmic", "cpu", "get", "loadpercentage", "/value"],
                capture_output=True,
                text=True,
                check=False,
            )
            values = []
            for line in result.stdout.splitlines():
                line = line.strip()
                if not line:
                    continue
                if "=" in line:
                    _, value = line.split("=", 1)
                    try:
                        values.append(float(value.strip()))
                    except ValueError:
                        pass
                else:
                    try:
                        values.append(float(line))
                    except ValueError:
                        pass
            if values:
                return round(sum(values) / len(values), 2)
            return 0.0

        # Unix/Linux/macOS fallback
        cmd = ["top", "-l", "1"] if system_name == "Darwin" else ["top", "-bn1"]
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            check=False,
        )

        for line in result.stdout.splitlines():
            lower = line.lower()
            if "cpu" not in lower:
                continue

            if "idle" in lower:
                for part in line.split(","):
                    if "idle" in part.lower():
                        idle = _first_number(part)
                        if idle is not None:
                            return round(max(0.0, min(100.0, 100.0 - idle)), 2)

            if " id" in lower or lower.strip().endswith("id"):
                for part in line.split(","):
                    if "id" in part.lower():
                        idle = _first_number(part)
                        if idle is not None:
                            return round(max(0.0, min(100.0, 100.0 - idle)), 2)

        return 0.0
    except Exception:
        return 0.0


def task_func(interval, duration):
    if interval <= 0 or duration <= 0:
        raise ValueError("interval and duration must be greater than zero")

    system_name = platform.system()
    records = []
    start = time.monotonic()
    next_sample_time = start

    while True:
        now = time.monotonic()
        if now < next_sample_time:
            time.sleep(next_sample_time - now)

        records.append(
            {
                "timestamp": time.time(),
                "cpu_usage": _get_cpu_usage(system_name),
            }
        )

        next_sample_time += interval
        if time.monotonic() - start >= duration:
            break

    try:
        with open(LOGFILE_PATH, "w", encoding="utf-8") as file_obj:
            json.dump(records, file_obj, ensure_ascii=False, indent=2)
    except OSError:
        return None

    return LOGFILE_PATH
