```python
import matplotlib.pyplot as plt
from collections import Counter

FRUITS = ['Apple', 'Banana', 'Cherry', 'Date', 'Elderberry', 'Fig', 'Grape', 'Honeydew', 'Indian Prune', 'Jackfruit']

def task_func(fruit_dict):
    counts = Counter(fruit_dict.values())
    result = {fruit: counts.get(fruit, 0) for fruit in FRUITS}

    fig, ax = plt.subplots()
    ax.bar(result.keys(), result.values())
    ax.set_xlabel("Fruit")
    ax.set_ylabel("Count")
    ax.set_title("Fruit Frequency")
    ax.tick_params(axis='x', rotation=45)
    fig.tight_layout()

    return result, ax
```
