import pandas as pd
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt

# Constants
FEATURE_NAMES = ["Feature 1", "Feature 2", "Feature 3", "Feature 4", "Feature 5"]

import numpy as np


def task_func(data_matrix):
    """
    Standardize a 2D data matrix, compute row means, and plot the distribution
    of the row means as a histogram.

    Parameters
    ----------
    data_matrix : array-like
        A 2D matrix-like structure with shape (n_samples, len(FEATURE_NAMES)).

    Returns
    -------
    tuple
        (dataframe, ax) where dataframe contains standardized features and
        a 'Mean' column, and ax is the matplotlib Axes for the histogram.
    """
    if data_matrix is None:
        raise ValueError("data_matrix must not be None.")

    arr = np.asarray(data_matrix)

    if arr.ndim != 2:
        raise ValueError("data_matrix must be a 2D matrix.")

    if arr.size == 0:
        raise ValueError("data_matrix must not be empty.")

    if arr.shape[1] != len(FEATURE_NAMES):
        raise ValueError(
            f"Each row must have exactly {len(FEATURE_NAMES)} columns "
            f"matching FEATURE_NAMES."
        )

    try:
        numeric_arr = arr.astype(float)
    except (TypeError, ValueError) as exc:
        raise ValueError("data_matrix must contain only numeric values.") from exc

    if not np.isfinite(numeric_arr).all():
        raise ValueError("data_matrix must not contain NaN or infinite values.")

    scaler = StandardScaler()
    standardized = scaler.fit_transform(numeric_arr)

    result_df = pd.DataFrame(standardized, columns=FEATURE_NAMES)
    result_df["Mean"] = result_df.mean(axis=1)

    fig, ax = plt.subplots()
    ax.hist(result_df["Mean"], bins="auto", edgecolor="black")
    ax.set_title("Distribution of Means")
    ax.set_xlabel("Mean")
    ax.set_ylabel("Frequency")

    return result_df, ax
