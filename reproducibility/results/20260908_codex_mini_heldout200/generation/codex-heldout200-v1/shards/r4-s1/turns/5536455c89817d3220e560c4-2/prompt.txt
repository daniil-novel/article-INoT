import pandas as pd
from sklearn.preprocessing import MinMaxScaler
import matplotlib.pyplot as plt

def task_func(data):
    df = data if isinstance(data, pd.DataFrame) else pd.DataFrame(data)

    if df.empty:
        raise ValueError("Input data is empty.")

    if df.isna().any().any():
        raise ValueError("Input data contains missing values.")

    if df.select_dtypes(include="number").shape[1] != df.shape[1]:
        raise ValueError("Input data must contain only numeric columns.")

    scaler = MinMaxScaler()
    scaled_values = scaler.fit_transform(df)

    result_df = pd.DataFrame(scaled_values, columns=df.columns, index=df.index)

    averages = result_df.mean(axis=1)
    result_df["Average"] = averages

    fig, ax = plt.subplots()
    ax.bar(result_df.index.astype(str), averages.values)
    ax.set_xlabel("Row")
    ax.set_ylabel("Average")
    ax.set_title("Average of Normalized Rows")
    plt.xticks(rotation=45)
    fig.tight_layout()

    return result_df, ax
