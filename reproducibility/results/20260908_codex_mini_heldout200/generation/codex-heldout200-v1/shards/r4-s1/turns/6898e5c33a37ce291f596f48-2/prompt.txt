import pandas as pd
import numpy as np
import itertools
from datetime import datetime, timedelta
import seaborn as sns
from typing import Tuple


def task_func(
    df,
    fruits=None,
    days=None,
    seed=None,
    sales_lower_bound=1,
    sales_upper_bound=50,
) -> Tuple[pd.DataFrame, sns.axisgrid.FacetGrid]:
    if not isinstance(df, pd.DataFrame):
        raise TypeError("'df' must be a pandas DataFrame.")
    if not df.empty:
        raise ValueError("'df' must be empty.")
    if sales_lower_bound >= sales_upper_bound:
        raise ValueError("'sales_lower_bound' must be less than 'sales_upper_bound'.")

    default_fruits = ["apple", "banana", "orange"]

    if fruits is None:
        fruits = default_fruits
    else:
        fruits = list(fruits)
        if not fruits:
            fruits = default_fruits

    if days is None:
        day_values = list(range(7))
    elif isinstance(days, int):
        day_values = list(range(days)) if days > 0 else list(range(7))
    else:
        day_values = list(days)
        if not day_values:
            day_values = list(range(7))

    rng = np.random.default_rng(seed)
    base_date = datetime(2020, 1, 1)

    records = []
    for fruit, day in itertools.product(fruits, day_values):
        if isinstance(day, (int, np.integer)):
            date_value = base_date + timedelta(days=int(day))
        elif isinstance(day, datetime):
            date_value = day
        else:
            date_value = day

        sales_value = float(rng.uniform(sales_lower_bound, sales_upper_bound))
        records.append({"fruit": fruit, "date": date_value, "sales": sales_value})

    generated_df = pd.DataFrame(records, columns=["fruit", "date", "sales"])
    updated_df = pd.concat([df.copy(), generated_df], ignore_index=True)

    plot = sns.catplot(data=updated_df, x="fruit", y="sales", kind="box")

    return updated_df, plot
