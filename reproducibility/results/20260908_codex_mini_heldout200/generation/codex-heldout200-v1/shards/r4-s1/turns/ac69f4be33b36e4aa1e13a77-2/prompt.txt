import math
import statistics
import numpy as np

from collections import Counter


def _extract_degree_value(item):
    if hasattr(item, "degree"):
        item = item.degree
    if isinstance(item, complex):
        if item.imag != 0:
            raise TypeError("Degree values must be real numbers.")
        item = item.real
    return float(item)


def _deterministic_mode(values):
    counts = Counter(values)
    max_count = max(counts.values())
    return min(value for value, count in counts.items() if count == max_count)


def task_func(input_list):
    if not input_list:
        return (0, 0, 0, 0, 0, 0)

    degree_values = [_extract_degree_value(x) for x in input_list]
    sorted_values = sorted(degree_values)

    mean_sorted = statistics.mean(sorted_values)
    median_sorted = statistics.median(sorted_values)
    mode_sorted = _deterministic_mode(sorted_values)

    fft_magnitudes = np.abs(np.fft.fft(np.asarray(sorted_values, dtype=float)))
    fft_magnitudes = [float(round(x)) for x in fft_magnitudes]

    mean_fft = statistics.mean(fft_magnitudes)
    median_fft = statistics.median(fft_magnitudes)
    mode_fft = _deterministic_mode(fft_magnitudes)

    return tuple(
        int(round(x))
        for x in (
            mean_sorted,
            median_sorted,
            mode_sorted,
            mean_fft,
            median_fft,
            mode_fft,
        )
    )
