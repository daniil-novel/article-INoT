import numpy as np
from scipy.stats import mode

def task_func(list_of_lists):
    merged = []

    for sublist in list_of_lists:
        if sublist is None:
            continue
        arr = np.asarray(sublist).ravel()
        if arr.size > 0:
            merged.extend(arr.tolist())

    merged_array = np.asarray(merged)

    if merged_array.size == 0:
        return np.array([]), 0

    result = mode(merged_array, keepdims=False)
    mode_value = np.asarray(result.mode)
    mode_count = int(np.asarray(result.count).item())

    return mode_value, mode_count
