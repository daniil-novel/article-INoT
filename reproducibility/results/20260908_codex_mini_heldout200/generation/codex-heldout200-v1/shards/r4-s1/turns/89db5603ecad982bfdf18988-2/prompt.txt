import matplotlib.pyplot as plt
from collections import Counter

FRUITS = ['Apple', 'Banana', 'Cherry', 'Date', 'Elderberry', 'Fig', 'Grape', 'Honeydew', 'Indian Prune', 'Jackfruit']

def task_func(fruit_dict):
    # Count only valid fruit names from the predefined list
    values = [fruit for fruit in fruit_dict.values() if isinstance(fruit, str)]
    counts = Counter(values)

    fruit_counts = {fruit: counts.get(fruit, 0) for fruit in FRUITS}
    y = [fruit_counts[fruit] for fruit in FRUITS]

    fig, ax = plt.subplots()
    ax.bar(FRUITS, y)
    ax.set_xlabel("Fruit")
    ax.set_ylabel("Count")
    ax.set_title("Fruit Frequency")
    ax.tick_params(axis='x', rotation=45)

    return fruit_counts, ax
