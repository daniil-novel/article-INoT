import math
import matplotlib.pyplot as plt
import numpy as np
import random
# Constants
RANGE = 10000
SIZE = 1000
PI = np.pi

def task_func(size=SIZE, frequency=1):
    """
    Create and plot random sinusoidal values.

    Returns
    -------
    matplotlib.axes.Axes
        Axes object containing the sinusoidal wave plot.
    """
    if not isinstance(size, int):
        size = int(size)

    if size <= 0:
        raise ValueError("size must be a positive integer")

    if frequency <= 0:
        raise ValueError("frequency must be a positive number")

    # Randomize sinusoid characteristics
    amplitude = random.uniform(0.5, 2.0)
    phase = random.uniform(0.0, 2.0 * PI)
    noise_level = random.uniform(0.0, 0.25)

    # Generate x-values and sinusoidal y-values with random noise
    x = np.linspace(0, RANGE, size)
    noise = np.array([random.uniform(-noise_level, noise_level) for _ in range(size)])
    y = amplitude * np.sin(2.0 * PI * frequency * x / RANGE + phase) + noise

    # Plot
    fig, ax = plt.subplots()
    ax.plot(x, y, linewidth=1.5)
    ax.set_xlabel("x")
    ax.set_ylabel("sinusoidal value")
    ax.set_title("Random Sinusoidal Wave")
    ax.grid(True, alpha=0.3)

    return ax
