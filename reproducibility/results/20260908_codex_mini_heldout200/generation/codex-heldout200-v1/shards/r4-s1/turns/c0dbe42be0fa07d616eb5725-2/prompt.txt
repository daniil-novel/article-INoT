```python
import collections
import itertools
import matplotlib.pyplot as plt

# Constants
ITEMS = ['apple', 'banana']


def task_func(a, b, items=ITEMS):
    combined = itertools.chain(a, b)
    counts = collections.Counter(combined)
    frequencies = [counts[item] for item in items]

    fig, ax = plt.subplots()
    ax.bar(items, frequencies)
    ax.set_xlabel('Items')
    ax.set_ylabel('Frequency')
    ax.set_title('Frequency of predefined items in combined list')
    return ax
```
