import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


def task_func(rows=5, cols=5):
    categories = ["A", "B", "C", "D", "E"]

    if not isinstance(rows, int) or not isinstance(cols, int):
        raise ValueError("rows and cols must be integers.")
    if rows <= 0 or cols <= 0:
        raise ValueError("rows and cols must be positive integers.")
    if cols > len(categories):
        raise ValueError("Number of columns exceeds the number of available categories.")

    data = np.random.rand(rows, cols)
    df = pd.DataFrame(
        data,
        columns=categories[:cols],
        index=[f"Row {i + 1}" for i in range(rows)],
    )

    ax = df.plot(kind="bar", stacked=True, figsize=(8, 5))
    ax.set_xlabel("Rows")
    ax.set_ylabel("Value")
    ax.set_title("Stacked Bar Chart of Random Numerical Data")
    return ax
