import os
import csv
from openpyxl import load_workbook


def task_func(file_name, excel_file_path, csv_file_path) -> str:
    """
    Convert the active sheet of an Excel workbook to a CSV file.

    Args:
        file_name: Desired output CSV file name (or base name).
        excel_file_path: Path to the source Excel file.
        csv_file_path: Destination directory or full CSV file path.

    Returns:
        The name of the created CSV file.
    """
    if not isinstance(excel_file_path, str) or not excel_file_path:
        raise ValueError("excel_file_path must be a non-empty string")
    if not os.path.isfile(excel_file_path):
        raise FileNotFoundError(f"Excel file not found: {excel_file_path}")

    excel_ext = os.path.splitext(excel_file_path)[1].lower()
    if excel_ext == ".xls":
        raise ValueError(
            "Legacy .xls files are not supported by openpyxl; please convert the file to .xlsx first."
        )

    # Derive a CSV file name.
    if isinstance(file_name, str) and file_name.strip():
        output_name = os.path.basename(file_name.strip())
    else:
        output_name = os.path.splitext(os.path.basename(excel_file_path))[0]

    output_root, output_ext = os.path.splitext(output_name)
    if output_ext.lower() in {".csv", ".xls", ".xlsx", ".xlsm", ".xltx", ".xltm"}:
        output_name = output_root + ".csv"
    else:
        output_name = output_name + ".csv"

    # Determine final output path.
    if not isinstance(csv_file_path, str) or not csv_file_path:
        raise ValueError("csv_file_path must be a non-empty string")

    csv_ext = os.path.splitext(csv_file_path)[1].lower()
    if os.path.isdir(csv_file_path) or csv_file_path.endswith(os.sep) or csv_ext == "":
        output_csv_path = os.path.join(csv_file_path, output_name)
    else:
        output_csv_path = csv_file_path
        if os.path.splitext(output_csv_path)[1].lower() != ".csv":
            output_csv_path = os.path.splitext(output_csv_path)[0] + ".csv"

    output_dir = os.path.dirname(output_csv_path)
    if output_dir:
        os.makedirs(output_dir, exist_ok=True)

    workbook = load_workbook(excel_file_path, data_only=True)
    try:
        worksheet = workbook.active

        with open(output_csv_path, mode="w", newline="", encoding="utf-8") as csv_file:
            writer = csv.writer(csv_file)
            for row in worksheet.iter_rows(values_only=True):
                writer.writerow(["" if cell is None else cell for cell in row])
    finally:
        try:
            workbook.close()
        except Exception:
            pass

    return os.path.basename(output_csv_path)
