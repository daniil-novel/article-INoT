```python
import os
import glob
import subprocess
import tarfile


def task_func(directory, backup_dir='/path/to/backup'):
    if not os.path.isdir(directory):
        raise FileNotFoundError(f"Specified directory does not exist: {directory}")

    os.makedirs(backup_dir, exist_ok=True)

    log_files = glob.glob(os.path.join(directory, '*.log'))
    if not log_files:
        return 'No logs found to backup'

    backup_path = os.path.join(backup_dir, 'logs_backup.tar.gz')

    with tarfile.open(backup_path, 'w:gz') as tar:
        for log_file in log_files:
            tar.add(log_file, arcname=os.path.basename(log_file))

    for log_file in log_files:
        os.remove(log_file)

    return backup_path
```
