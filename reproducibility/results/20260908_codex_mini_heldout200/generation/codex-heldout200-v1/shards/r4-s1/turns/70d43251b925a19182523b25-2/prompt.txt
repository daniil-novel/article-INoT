import re
import nltk
nltk.download('stopwords')
from nltk.corpus import stopwords
from collections import Counter

# Constants
STOPWORDS = set(stopwords.words('english'))

def task_func(text, n=2):
    """
    Remove stopwords and duplicate tokens from text, then count n-grams.

    Args:
        text (str): Input text.
        n (int): Size of the n-grams. Defaults to 2.

    Returns:
        dict: A dictionary mapping n-gram tuples to their counts.
    """
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    if not isinstance(n, int):
        raise TypeError("n must be an integer")
    if n <= 0:
        raise ValueError("n must be a positive integer")

    # Tokenize and normalize
    tokens = re.findall(r"[A-Za-z]+(?:'[A-Za-z]+)?", text.lower())

    # Remove stopwords
    filtered_tokens = [token for token in tokens if token not in STOPWORDS]

    # Remove duplicate tokens while preserving order
    seen = set()
    unique_tokens = []
    for token in filtered_tokens:
        if token not in seen:
            seen.add(token)
            unique_tokens.append(token)

    # Generate n-grams
    if len(unique_tokens) < n:
        return {}

    ngrams = [tuple(unique_tokens[i:i + n]) for i in range(len(unique_tokens) - n + 1)]

    # Count n-grams
    return dict(Counter(ngrams))
