import pandas as pd
import numpy as np
import itertools
from datetime import datetime, timedelta
import seaborn as sns

def task_func(df, fruits=None, days=None, seed=None, sales_lower_bound=1, sales_upper_bound=50):
    if not isinstance(df, pd.DataFrame):
        raise TypeError("'df' must be a pandas DataFrame.")
    if not df.empty:
        raise ValueError("'df' must be empty.")

    try:
        if not sales_lower_bound < sales_upper_bound:
            raise ValueError
    except TypeError as exc:
        raise ValueError("'sales_lower_bound' must be less than 'sales_upper_bound'.") from exc
    except ValueError as exc:
        raise ValueError("'sales_lower_bound' must be less than 'sales_upper_bound'.") from exc

    if fruits is None or isinstance(fruits, str):
        raise ValueError("'fruits' must be a non-empty iterable of fruit names.")
    try:
        fruits_list = list(fruits)
    except TypeError as exc:
        raise ValueError("'fruits' must be a non-empty iterable of fruit names.") from exc
    if not fruits_list:
        raise ValueError("'fruits' must be a non-empty iterable of fruit names.")

    unique_fruits = []
    for fruit in fruits_list:
        if fruit not in unique_fruits:
            unique_fruits.append(fruit)

    if not isinstance(days, int) or isinstance(days, bool) or days <= 0:
        raise ValueError("'days' must be a positive integer.")

    rng = np.random.default_rng(seed)
    start_day = datetime(2020, 1, 1)
    day_sequence = [start_day + timedelta(days=i) for i in range(days)]

    records = []
    for fruit, day in itertools.product(unique_fruits, day_sequence):
        records.append(
            {
                "fruit": fruit,
                "day": day,
                "sales": int(rng.integers(sales_lower_bound, sales_upper_bound)),
            }
        )

    updated_df = pd.concat([df, pd.DataFrame(records)], ignore_index=True)
    boxplot = sns.catplot(data=updated_df, x="fruit", y="sales", kind="box", order=unique_fruits)

    return updated_df, boxplot
