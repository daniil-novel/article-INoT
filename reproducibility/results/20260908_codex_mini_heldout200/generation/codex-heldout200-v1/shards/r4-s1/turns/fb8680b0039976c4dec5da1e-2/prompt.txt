```python
import json
import os
import hashlib
import base64
import time


def task_func(file_path, unknown_key):
    """
    Read a JSON file, extract:
        A -> [unknown_key] -> maindata -> [0] -> Info

    Then SHA256-hash the extracted value, base64-encode the hash,
    write it to a timestamped file, and return the absolute path
    to the created file.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        data = json.load(f)

    try:
        value = data["A"][unknown_key]["maindata"][0]["Info"]
    except (KeyError, TypeError, IndexError) as e:
        raise KeyError(
            "JSON does not contain the required structure: "
            "A -> [unknown_key] -> maindata -> [0] -> Info"
        ) from e

    value_bytes = str(value).encode("utf-8")
    digest = hashlib.sha256(value_bytes).digest()
    encoded_hash = base64.b64encode(digest).decode("utf-8")

    timestamp = time.strftime("%Y%m%d_%H%M%S", time.localtime())
    microseconds = f"{int((time.time() % 1) * 1_000_000):06d}"
    output_name = f"hash_{timestamp}_{microseconds}.txt"

    output_dir = os.path.dirname(os.path.abspath(file_path))
    output_path = os.path.abspath(os.path.join(output_dir, output_name))

    with open(output_path, "w", encoding="utf-8") as f:
        f.write(encoded_hash)

    return output_path
```
