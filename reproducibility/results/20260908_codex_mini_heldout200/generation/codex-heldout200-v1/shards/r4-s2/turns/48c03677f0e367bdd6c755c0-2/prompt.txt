import pandas as pd
import random
from sklearn.model_selection import train_test_split

def task_func(n_data_points=10000, min_value=0.0, max_value=10.0, test_size=0.2):
    def truncate_to_3_decimals(value):
        return int(value * 1000) / 1000.0

    values = [
        truncate_to_3_decimals(random.uniform(min_value, max_value))
        for _ in range(n_data_points)
    ]

    df = pd.DataFrame({"Value": values})

    if n_data_points == 0:
        empty_df = pd.DataFrame({"Value": []})
        return empty_df.copy(), empty_df.copy()

    train_df, test_df = train_test_split(df, test_size=test_size)

    return train_df.reset_index(drop=True), test_df.reset_index(drop=True)
