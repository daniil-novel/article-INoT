import json
import csv
import os
import base64

def task_func(raw_string, filename, output_dir):
    if not raw_string or not filename or not output_dir:
        raise ValueError("raw_string, filename, and output_dir are required")

    raw_string = raw_string.strip()
    filename = filename.strip()

    if not filename:
        raise ValueError("filename must not be empty")

    if not filename.lower().endswith(".csv"):
        filename += ".csv"

    os.makedirs(output_dir, exist_ok=True)
    file_path = os.path.join(output_dir, filename)

    try:
        decoded_bytes = base64.b64decode(raw_string, validate=True)
    except Exception as exc:
        raise ValueError("Invalid base64 input") from exc

    try:
        decoded_text = decoded_bytes.decode("utf-8")
    except UnicodeDecodeError as exc:
        raise ValueError("Decoded data must be UTF-8 text") from exc

    try:
        data = json.loads(decoded_text)
    except json.JSONDecodeError as exc:
        raise ValueError("Decoded string is not valid JSON") from exc

    if data is None:
        raise ValueError("Empty JSON data is not supported")

    def serialize_value(value):
        if isinstance(value, (dict, list)):
            return json.dumps(value, ensure_ascii=False)
        return value

    rows = []
    fieldnames = []

    if isinstance(data, list):
        if not data:
            raise ValueError("Empty JSON list is not supported")

        if all(isinstance(item, dict) for item in data):
            seen = set()
            for item in data:
                for key in item.keys():
                    if key not in seen:
                        seen.add(key)
                        fieldnames.append(key)

            if not fieldnames:
                raise ValueError("Empty JSON objects are not supported")

            for item in data:
                row = {key: serialize_value(item.get(key, "")) for key in fieldnames}
                rows.append(row)

        elif all(not isinstance(item, (dict, list)) for item in data):
            fieldnames = ["value"]
            rows = [{"value": serialize_value(item)} for item in data]

        else:
            fieldnames = ["value"]
            rows = [{"value": serialize_value(item)} for item in data]

    elif isinstance(data, dict):
        if not data:
            raise ValueError("Empty JSON object is not supported")

        fieldnames = list(data.keys())
        rows = [{key: serialize_value(data.get(key, "")) for key in fieldnames}]

    else:
        fieldnames = ["value"]
        rows = [{"value": serialize_value(data)}]

    with open(file_path, mode="w", newline="", encoding="utf-8") as csv_file:
        writer = csv.DictWriter(csv_file, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)

    return file_path
