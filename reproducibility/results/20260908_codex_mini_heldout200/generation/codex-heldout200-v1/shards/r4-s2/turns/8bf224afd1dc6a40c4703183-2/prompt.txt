import pandas as pd
import random
from sklearn.model_selection import train_test_split
import math

def task_func(n_data_points=10000, min_value=0.0, max_value=10.0, test_size=0.2):
    if not isinstance(n_data_points, int) or isinstance(n_data_points, bool) or n_data_points < 0:
        raise ValueError("n_data_points must be a non-negative integer.")
    
    if not isinstance(min_value, (int, float)) or isinstance(min_value, bool):
        raise ValueError("min_value must be numeric.")
    if not isinstance(max_value, (int, float)) or isinstance(max_value, bool):
        raise ValueError("max_value must be numeric.")
    if min_value > max_value:
        raise ValueError("min_value must be less than or equal to max_value.")
    
    if not isinstance(test_size, (int, float)) or isinstance(test_size, bool):
        raise ValueError("test_size must be numeric.")
    
    if n_data_points == 0:
        empty_df = pd.DataFrame(columns=["Value"])
        return empty_df.copy(), empty_df.copy()
    
    values = [
        math.trunc(random.uniform(min_value, max_value) * 1000) / 1000.0
        for _ in range(n_data_points)
    ]
    data = pd.DataFrame({"Value": values})
    
    if n_data_points == 1:
        # A one-row dataset cannot be split into two non-empty sets.
        return data.iloc[0:0].reset_index(drop=True), data.reset_index(drop=True)
    
    if isinstance(test_size, float):
        if not (0 < test_size < 1):
            raise ValueError("test_size as a float must be between 0 and 1.")
        # Ensure both splits are non-empty for small datasets.
        n_test = max(1, min(n_data_points - 1, math.ceil(n_data_points * test_size)))
        train_df, test_df = train_test_split(data, test_size=n_test, shuffle=True)
    else:
        if test_size <= 0 or test_size >= n_data_points:
            raise ValueError("test_size as an int must be between 1 and n_data_points - 1.")
        train_df, test_df = train_test_split(data, test_size=test_size, shuffle=True)
    
    return train_df.reset_index(drop=True), test_df.reset_index(drop=True)
