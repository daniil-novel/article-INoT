import pandas as pd
import numpy as np
# Constants
COLUMNS = ['column1', 'column2', 'column3', 'column4', 'column5']
def task_func(df, dct):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("df must be a pandas DataFrame")
    
    replaced_df = df.copy().replace(dct)
    replaced_df = replaced_df.apply(pd.to_numeric, errors='coerce')
    
    return replaced_df.corr(method='pearson')
