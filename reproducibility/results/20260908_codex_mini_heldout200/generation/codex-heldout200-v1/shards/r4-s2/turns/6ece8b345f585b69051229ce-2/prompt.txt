import json
import csv
import os
import base64

def task_func(raw_string, filename, output_dir):
    """
    Decode a base64-encoded JSON string, save its contents to a CSV file,
    and return the output file path.
    """
    if not filename or not isinstance(filename, str):
        raise ValueError("filename must be a non-empty string")
    if not output_dir or not isinstance(output_dir, str):
        raise ValueError("output_dir must be a non-empty string")

    os.makedirs(output_dir, exist_ok=True)

    if not filename.lower().endswith(".csv"):
        filename = filename + ".csv"
    file_path = os.path.join(output_dir, filename)

    # Normalize and decode base64 input
    try:
        if isinstance(raw_string, str):
            raw_bytes = raw_string.encode("utf-8")
        else:
            raw_bytes = bytes(raw_string)
        decoded_bytes = base64.b64decode(raw_bytes, validate=True)
    except Exception as exc:
        raise ValueError("Invalid base64 input") from exc

    try:
        decoded_text = decoded_bytes.decode("utf-8")
    except Exception as exc:
        raise ValueError("Decoded content is not valid UTF-8") from exc

    try:
        data = json.loads(decoded_text)
    except Exception as exc:
        raise ValueError("Decoded content is not valid JSON") from exc

    def _serialize(value):
        if value is None:
            return ""
        if isinstance(value, (dict, list)):
            return json.dumps(value, ensure_ascii=False)
        return str(value)

    with open(file_path, "w", newline="", encoding="utf-8") as csvfile:
        # List handling
        if isinstance(data, list):
            if len(data) == 0:
                # Write an empty CSV file
                pass
            elif all(isinstance(item, dict) for item in data):
                # Union of keys preserving first-seen order
                headers = []
                seen = set()
                for item in data:
                    for key in item.keys():
                        if key not in seen:
                            seen.add(key)
                            headers.append(key)

                writer = csv.DictWriter(csvfile, fieldnames=headers)
                writer.writeheader()
                for item in data:
                    row = {key: _serialize(item.get(key)) for key in headers}
                    writer.writerow(row)
            else:
                # Mixed or scalar list: store each element in one column
                writer = csv.writer(csvfile)
                writer.writerow(["value"])
                for item in data:
                    writer.writerow([_serialize(item)])

        # Dictionary handling
        elif isinstance(data, dict):
            writer = csv.writer(csvfile)
            writer.writerow(["key", "value"])
            for key, value in data.items():
                writer.writerow([str(key), _serialize(value)])

        # Scalar handling
        else:
            writer = csv.writer(csvfile)
            writer.writerow(["value"])
            writer.writerow([_serialize(data)])

    return file_path
