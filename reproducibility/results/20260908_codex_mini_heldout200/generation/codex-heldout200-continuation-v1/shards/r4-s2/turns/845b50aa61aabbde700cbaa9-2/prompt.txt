import seaborn as sns
import matplotlib.pyplot as plt
def task_func(df, target_values=[1, 3, 4]):
    # Work on a copy so the original dataframe is not modified
    work_df = df.copy()

    target_set = set(target_values)

    # Replace every value not in target_values with 0
    if hasattr(work_df, "columns"):
        for col in work_df.columns:
            mask = ~work_df[col].isin(target_set)
            work_df.loc[mask, col] = 0

    # Handle empty/no-column dataframes gracefully
    if not hasattr(work_df, "columns") or len(work_df.columns) == 0:
        fig, ax = plt.subplots(figsize=(6, 4))
        ax.set_axis_off()
        return ax

    n_cols = len(work_df.columns)
    fig, axes = plt.subplots(n_cols, 1, figsize=(8, 4 * n_cols))

    if n_cols == 1:
        axes = [axes]

    for ax, col in zip(axes, work_df.columns):
        # Plot the distribution after replacement
        sns.countplot(x=work_df[col].astype(str), ax=ax, color="steelblue")
        ax.set_title(col)
        ax.set_xlabel(col)
        ax.set_ylabel("Count")
        ax.tick_params(axis="x", rotation=45)

    plt.tight_layout()
    return axes[0]
