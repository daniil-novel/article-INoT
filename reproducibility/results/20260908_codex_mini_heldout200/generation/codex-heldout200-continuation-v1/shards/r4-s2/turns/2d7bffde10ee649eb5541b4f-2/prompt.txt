import socket
from ipaddress import IPv4Network
from threading import Thread

def task_func(ip_range, port):
    network = ip_range if isinstance(ip_range, IPv4Network) else IPv4Network(ip_range)
    port = int(port)
    if port < 1 or port > 65535:
        raise ValueError("port must be in range 1..65535")

    results = {}

    def check_ip(ip):
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                sock.settimeout(0.8)
                is_open = sock.connect_ex((str(ip), port)) == 0
        except OSError:
            is_open = False
        results[str(ip)] = is_open

    threads = []
    for ip in network.hosts():
        thread = Thread(target=check_ip, args=(ip,))
        thread.start()
        threads.append(thread)

    for thread in threads:
        thread.join()

    return results
