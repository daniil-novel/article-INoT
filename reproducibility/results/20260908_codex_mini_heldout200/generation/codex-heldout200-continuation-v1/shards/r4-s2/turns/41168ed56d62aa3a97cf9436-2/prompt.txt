import pandas as pd
import matplotlib.pyplot as plt
from collections import Counter

def task_func(student_grades, possible_grades=["A", "B", "C", "D", "F"]):
    # Build a case-insensitive lookup for valid grades while preserving
    # the first canonical display form and the original order.
    canonical_grades = []
    normalized_to_canonical = {}
    for grade in possible_grades:
        if not isinstance(grade, str):
            grade = str(grade)
        normalized = grade.casefold()
        if normalized not in normalized_to_canonical:
            normalized_to_canonical[normalized] = grade
            canonical_grades.append(grade)

    # Count only valid grades from student_grades, case-insensitively.
    counts = Counter()
    for grade in student_grades:
        if not isinstance(grade, str):
            continue
        normalized = grade.casefold()
        if normalized in normalized_to_canonical:
            counts[normalized_to_canonical[normalized]] += 1

    # Ensure all possible grades appear, even with zero count.
    data = {
        "Count": [int(counts.get(grade, 0)) for grade in canonical_grades]
    }
    df = pd.DataFrame(data, index=canonical_grades)
    df.index.name = "Grade"

    # Create the bar chart.
    fig, ax = plt.subplots()
    ax.bar(df.index.tolist(), df["Count"].tolist())
    ax.set_title("Grade Distribution")
    ax.set_xlabel("Grade")
    ax.set_ylabel("Number of Students")

    return df, ax
