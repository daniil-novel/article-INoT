import subprocess
import os
import signal
import time

def task_func(process_name: str) -> int:
    target = (process_name or "").strip()
    if not target:
        return 0

    current_pid = os.getpid()

    def normalize(name: str) -> str:
        base = os.path.basename(name.strip())
        return base.lower() if os.name == "nt" else base

    def matches(candidate: str) -> bool:
        cand_base = os.path.basename(candidate.strip())
        if not cand_base:
            return False

        cand_norm = normalize(cand_base)
        target_norm = normalize(target)

        if cand_norm == target_norm:
            return True

        if os.name == "nt":
            cand_stem = os.path.splitext(cand_base)[0].lower()
            target_stem = os.path.splitext(os.path.basename(target))[0].lower()
            return cand_stem == target_stem

        return False

    pids = []

    try:
        if os.name == "nt":
            result = subprocess.run(
                ["tasklist", "/fo", "csv", "/nh"],
                capture_output=True,
                text=True,
                check=False,
            )
            for line in result.stdout.splitlines():
                line = line.strip()
                if not line or not line.startswith('"'):
                    continue
                parts = line.split('","')
                if len(parts) < 2:
                    continue
                image_name = parts[0].strip('"')
                pid_part = parts[1].strip('"')
                try:
                    pid = int(pid_part)
                except ValueError:
                    continue
                if pid != current_pid and matches(image_name):
                    pids.append(pid)
        else:
            result = subprocess.run(
                ["ps", "-eo", "pid=,comm="],
                capture_output=True,
                text=True,
                check=False,
            )
            for line in result.stdout.splitlines():
                line = line.strip()
                if not line:
                    continue
                parts = line.split(None, 1)
                if len(parts) != 2:
                    continue
                pid_part, name = parts
                try:
                    pid = int(pid_part)
                except ValueError:
                    continue
                if pid != current_pid and matches(name):
                    pids.append(pid)
    except Exception:
        return 0

    if not pids:
        return 0

    stopped = 0
    seen = set()
    for pid in pids:
        if pid in seen:
            continue
        seen.add(pid)
        try:
            os.kill(pid, signal.SIGTERM)
            stopped += 1
        except (ProcessLookupError, PermissionError, OSError):
            pass

    time.sleep(1)
    return stopped
