import pandas as pd
import seaborn as sns
import numpy as np
import ast

def task_func(csv_file):
    """
    Load email data from a CSV file, compute per-row statistics for the list column,
    plot a histogram of mean values, and return the DataFrame plus Axes.
    """
    df = pd.read_csv(csv_file)

    required_columns = {"email", "list"}
    missing = required_columns - set(df.columns)
    if missing:
        raise ValueError(f"CSV file must contain columns: {sorted(required_columns)}. Missing: {sorted(missing)}")

    def parse_list_value(value):
        if pd.isna(value):
            raise ValueError("Encountered missing value in 'list' column.")
        if isinstance(value, str):
            parsed = ast.literal_eval(value)
        else:
            parsed = value

        if isinstance(parsed, np.ndarray):
            parsed = parsed.tolist()
        elif not isinstance(parsed, (list, tuple)):
            parsed = [parsed]

        return list(parsed)

    def compute_stats(values):
        if len(values) == 0:
            return 0.0, np.nan, np.nan

        numeric = pd.to_numeric(pd.Series(values), errors="raise").astype(float)
        valid = numeric.dropna()

        if valid.empty:
            return np.nan, np.nan, np.nan

        return valid.sum(), valid.mean(), valid.std(ddof=1)

    parsed_lists = []
    sums = []
    means = []
    stds = []

    for item in df["list"]:
        lst = parse_list_value(item)
        s, m, st = compute_stats(lst)
        parsed_lists.append(lst)
        sums.append(s)
        means.append(m)
        stds.append(st)

    result_df = pd.DataFrame(
        {
            "email": df["email"],
            "list": parsed_lists,
            "sum": sums,
            "mean": means,
            "std": stds,
        }
    )

    ax = sns.histplot(data=result_df, x="mean")
    ax.set_title("Histogram of Mean Values")
    ax.set_xlabel("Mean")
    ax.set_ylabel("Count")

    return result_df, ax
