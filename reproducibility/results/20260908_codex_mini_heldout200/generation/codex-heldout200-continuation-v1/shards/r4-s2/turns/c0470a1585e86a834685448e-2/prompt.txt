import urllib.request
from bs4 import BeautifulSoup
import csv
import os

# Constants
CSV_FILE_PATH = 'scraped_data.csv'


def task_func(url):
    def normalize_text(value):
        return " ".join(str(value).split())

    def pad_row(row, length):
        row = list(row)
        if len(row) < length:
            row.extend([""] * (length - len(row)))
        return row[:length]

    def write_csv(headers, rows):
        max_len = max([len(headers)] + [len(r) for r in rows]) if (headers or rows) else 1
        if not headers:
            headers = [f"column_{i + 1}" for i in range(max_len)]
        else:
            headers = pad_row(headers, max_len)
        normalized_rows = [pad_row(r, max_len) for r in rows]

        with open(CSV_FILE_PATH, "w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            writer.writerow(headers)
            writer.writerows(normalized_rows)

    try:
        if not url or not str(url).strip():
            write_csv(["error"], [["Empty URL provided"]])
            return CSV_FILE_PATH

        request = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Mozilla/5.0 (compatible; CodexScraper/1.0)"
            },
        )

        with urllib.request.urlopen(request, timeout=15) as response:
            charset = None
            if hasattr(response.headers, "get_content_charset"):
                charset = response.headers.get_content_charset()
            html = response.read().decode(charset or "utf-8", errors="replace")

        soup = BeautifulSoup(html, "html.parser")

        # 1) Try to scrape the first non-empty table
        table_rows = []
        for table in soup.find_all("table"):
            rows = []
            for tr in table.find_all("tr"):
                cells = tr.find_all(["th", "td"])
                if not cells:
                    continue
                values = [normalize_text(cell.get_text(" ", strip=True)) for cell in cells]
                if any(values):
                    rows.append(values)

            if rows:
                table_rows = rows
                break

        if table_rows:
            first_row = table_rows[0]
            first_tr = None
            # Re-find the first row element to check whether it looks like a header
            for table in soup.find_all("table"):
                first_tr = table.find("tr")
                if first_tr and first_tr.find_all(["th", "td"]):
                    break

            has_header_cells = bool(first_tr and first_tr.find_all("th"))

            if has_header_cells:
                headers = first_row
                data_rows = table_rows[1:]
            else:
                max_len = max(len(r) for r in table_rows)
                headers = [f"column_{i + 1}" for i in range(max_len)]
                data_rows = table_rows

            write_csv(headers, data_rows)
            return CSV_FILE_PATH

        # 2) Fallback: scrape visible text blocks
        block_tags = ["h1", "h2", "h3", "h4", "h5", "h6", "p", "li", "blockquote", "pre"]
        seen = set()
        text_rows = []

        for tag in soup.find_all(block_tags):
            text = normalize_text(tag.get_text(" ", strip=True))
            if text and text not in seen:
                seen.add(text)
                text_rows.append([text])

        if not text_rows:
            title = normalize_text(soup.title.get_text(" ", strip=True)) if soup.title and soup.title.get_text(strip=True) else ""
            if title:
                text_rows = [[title]]
            else:
                page_text = normalize_text(soup.get_text(" ", strip=True))
                if page_text:
                    text_rows = [[page_text]]
                else:
                    text_rows = [[""]]

        write_csv(["text"], text_rows)
        return CSV_FILE_PATH

    except Exception as e:
        # Ensure the function always produces the requested CSV file
        try:
            with open(CSV_FILE_PATH, "w", newline="", encoding="utf-8") as f:
                writer = csv.writer(f)
                writer.writerow(["error"])
                writer.writerow([normalize_text(e)])
        except Exception:
            pass
        return CSV_FILE_PATH
