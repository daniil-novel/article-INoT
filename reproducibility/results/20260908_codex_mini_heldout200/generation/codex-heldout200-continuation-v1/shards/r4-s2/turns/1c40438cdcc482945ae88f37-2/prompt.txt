from bs4 import BeautifulSoup
import requests
# Constants
URL = "http://example.com"
def task_func(url=URL, from_encoding="cp1251", use_lxml=False):
    if url is None:
        return None

    if isinstance(url, str):
        url = url.strip()
        if not url:
            return None

    try:
        response = requests.get(url, timeout=10)
        if response.status_code != 200:
            return None

        decoded_html = response.content.decode(from_encoding)
        parser = "lxml" if use_lxml else "html.parser"
        soup = BeautifulSoup(decoded_html, parser)
        return soup
    except (requests.exceptions.RequestException, UnicodeDecodeError, LookupError, TypeError, ValueError):
        return None
