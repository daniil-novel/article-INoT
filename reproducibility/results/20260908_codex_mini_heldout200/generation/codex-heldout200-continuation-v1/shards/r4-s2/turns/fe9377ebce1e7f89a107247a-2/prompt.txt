import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(data=None):
    """
    Convert string-formatted weights to floats and plot weight vs height scatter plot.

    Parameters
    ----------
    data : dict, optional
        Dictionary with keys:
        - 'Weight_String': list of weights as strings
        - 'Height': list of numeric heights

    Returns
    -------
    ax : matplotlib.axes._axes.Axes
        Scatter plot axis with title "Weight vs Height".

    Raises
    ------
    ValueError
        If input is invalid or any value in 'Weight_String' is not a string.
    """
    if data is None:
        data = {
            "Weight_String": ["50.0", "60.5", "70.2", "80.1", "90.4"],
            "Height": [150, 160, 170, 180, 190],
        }

    if not isinstance(data, dict):
        raise ValueError("data must be a dictionary with keys 'Weight_String' and 'Height'.")

    if "Weight_String" not in data or "Height" not in data:
        raise ValueError("data must contain both 'Weight_String' and 'Height' keys.")

    weight_strings = data["Weight_String"]
    heights = data["Height"]

    if len(weight_strings) != len(heights):
        raise ValueError("'Weight_String' and 'Height' must have the same length.")

    if any(not isinstance(w, str) for w in weight_strings):
        raise ValueError("All values in 'Weight_String' must be strings.")

    try:
        weights = [float(w) for w in weight_strings]
    except ValueError as e:
        raise ValueError("All values in 'Weight_String' must be convertible to float.") from e

    try:
        heights_numeric = pd.to_numeric(pd.Series(heights), errors="raise").tolist()
    except Exception as e:
        raise ValueError("All values in 'Height' must be numeric.") from e

    df = pd.DataFrame(
        {
            "Weight": weights,
            "Height": heights_numeric,
        }
    )

    fig, ax = plt.subplots()
    sns.scatterplot(data=df, x="Weight", y="Height", ax=ax)
    ax.set_title("Weight vs Height")
    ax.set_xlabel("Weight")
    ax.set_ylabel("Height")

    return ax
