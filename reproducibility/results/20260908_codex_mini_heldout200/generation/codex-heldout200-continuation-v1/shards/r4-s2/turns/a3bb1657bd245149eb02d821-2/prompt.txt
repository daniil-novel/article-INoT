import numpy as np
import pandas as pd

def task_func(rows, columns=["A", "B", "C", "D", "E"], seed=0) -> pd.DataFrame:
    if isinstance(rows, bool) or not isinstance(rows, (int, np.integer)):
        raise TypeError("rows must be a non-negative integer")
    if rows < 0:
        raise ValueError("rows must be non-negative")

    unique_columns = sorted(set(columns))
    rng = np.random.default_rng(seed)

    data = rng.random((rows, len(unique_columns)))
    if len(unique_columns) > 0:
        perm = rng.permutation(len(unique_columns))
        data = data[:, perm]
        shuffled_columns = [unique_columns[i] for i in perm]
    else:
        shuffled_columns = []

    return pd.DataFrame(data, columns=shuffled_columns)
