import urllib.request
from bs4 import BeautifulSoup
import csv
import os
# Constants
CSV_FILE_PATH = 'scraped_data.csv'

def task_func(url):
    if not isinstance(url, str) or not url.strip():
        raise ValueError("url must be a non-empty string")

    url = url.strip()
    if not url.startswith(("http://", "https://")):
        raise ValueError("url must start with http:// or https://")

    def _clean_text(node):
        if node is None:
            return ""
        return " ".join(node.get_text(" ", strip=True).split())

    def _unique_headers(headers):
        seen = {}
        result = []
        for header in headers:
            name = header.strip() if header else ""
            if not name:
                name = "column"
            base = name
            count = seen.get(base, 0)
            if count:
                name = f"{base}_{count + 1}"
            seen[base] = count + 1
            result.append(name)
        return result

    html_text = ""
    error_message = None

    try:
        request = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(request, timeout=20) as response:
            raw = response.read()
            charset = response.headers.get_content_charset() or "utf-8"
            html_text = raw.decode(charset, errors="replace")
    except Exception as exc:
        error_message = str(exc)

    soup = BeautifulSoup(html_text, "html.parser") if html_text else None
    records = []

    if soup is not None:
        tables = soup.find_all("table")
        for table_index, table in enumerate(tables):
            rows = table.find_all("tr")
            if not rows:
                continue

            header_row_index = None
            headers = []

            for i, row in enumerate(rows):
                th_cells = row.find_all("th")
                if th_cells:
                    headers = [_clean_text(cell) for cell in th_cells]
                    header_row_index = i
                    break

            if headers:
                headers = _unique_headers(headers)
            else:
                max_cols = 0
                for row in rows:
                    cell_count = len(row.find_all(["td", "th"]))
                    if cell_count > max_cols:
                        max_cols = cell_count
                if max_cols == 0:
                    continue
                headers = [f"col_{i + 1}" for i in range(max_cols)]
                header_row_index = -1  # include all rows as data

            data_row_number = 0
            for i, row in enumerate(rows):
                if i == header_row_index:
                    continue

                cells = row.find_all(["td", "th"])
                if not cells:
                    continue

                values = [_clean_text(cell) for cell in cells]

                if len(values) > len(headers):
                    for extra_index in range(len(headers), len(values)):
                        headers.append(f"col_{extra_index + 1}")
                elif len(values) < len(headers):
                    values.extend([""] * (len(headers) - len(values)))

                data_row_number += 1
                record = {"table_index": table_index, "row_index": data_row_number}
                for header, value in zip(headers, values):
                    record[header] = value
                records.append(record)

        if not records:
            fallback_records = []

            title = _clean_text(soup.title) if soup.title else ""
            if title:
                fallback_records.append({"content_type": "title", "text": title, "href": ""})

            for tag_name in ["h1", "h2", "h3", "h4", "h5", "h6", "p", "li"]:
                for tag in soup.find_all(tag_name):
                    text = _clean_text(tag)
                    if text:
                        fallback_records.append(
                            {"content_type": tag_name, "text": text, "href": ""}
                        )

            for link in soup.find_all("a", href=True):
                text = _clean_text(link)
                href = link.get("href", "").strip()
                if text or href:
                    fallback_records.append(
                        {"content_type": "link", "text": text, "href": href}
                    )

            records = fallback_records

    if not records:
        if error_message:
            records = [{"content_type": "error", "text": error_message, "href": ""}]
        else:
            records = [{"content_type": "empty", "text": "", "href": ""}]

    directory = os.path.dirname(CSV_FILE_PATH)
    if directory:
        os.makedirs(directory, exist_ok=True)

    fieldnames = []
    seen_fields = set()
    for record in records:
        for key in record.keys():
            if key not in seen_fields:
                seen_fields.add(key)
                fieldnames.append(key)

    with open(CSV_FILE_PATH, "w", newline="", encoding="utf-8") as csvfile:
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(records)

    return CSV_FILE_PATH
