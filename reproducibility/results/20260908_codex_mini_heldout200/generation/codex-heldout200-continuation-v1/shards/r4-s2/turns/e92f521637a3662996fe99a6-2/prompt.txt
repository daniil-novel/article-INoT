import pandas as pd
from random import randint, seed

def task_func(dictionary, item, sample_size=None, random_seed=None):
    """
    Convert a dictionary to a DataFrame, find positions of `item`,
    optionally sample the positions, and return (positions, dataframe).

    Also records a frequency distribution internally as:
    task_func.last_frequency_distribution = {
        "total": int,
        "by_row": dict,
        "by_column": dict,
    }
    """
    df = pd.DataFrame(dictionary)

    positions = []
    row_counts = {}
    column_counts = {}

    # Handle missing-value searches (NaN/None/pd.NA)
    try:
        target_is_missing = bool(pd.isna(item))
    except Exception:
        target_is_missing = False

    for col_name, series in df.items():
        for row_index, value in series.items():
            if target_is_missing:
                match = pd.isna(value)
            else:
                try:
                    match = value == item
                except Exception:
                    match = False

            if match:
                pos = (row_index, col_name)
                positions.append(pos)

                row_counts[row_index] = row_counts.get(row_index, 0) + 1
                column_counts[col_name] = column_counts.get(col_name, 0) + 1

    # Record frequency distribution internally
    task_func.last_frequency_distribution = {
        "total": len(positions),
        "by_row": row_counts,
        "by_column": column_counts,
    }

    # Optional random sampling
    if sample_size is not None:
        if isinstance(sample_size, bool) or not isinstance(sample_size, int):
            raise TypeError("sample_size must be an integer or None.")
        if sample_size < 0:
            raise ValueError("sample_size must be non-negative.")

        if random_seed is not None:
            seed(random_seed)

        if sample_size >= len(positions):
            return positions, df

        pool = positions[:]
        sampled = []
        while len(sampled) < sample_size:
            idx = randint(0, len(pool) - 1)
            sampled.append(pool.pop(idx))
        positions = sampled

    return positions, df
