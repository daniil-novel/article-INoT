import random
import string
import pandas as pd

def task_func(data_list, seed=0):
    random.seed(seed)
    records = []

    for s in data_list:
        parts = s.split(",")
        idx = random.randrange(len(parts))
        replacement = "".join(random.choice(string.ascii_lowercase) for _ in range(len(parts[idx])))
        parts[idx] = replacement
        modified = ",".join(parts)
        records.append({
            "Original String": s,
            "Modified String": modified
        })

    return pd.DataFrame(records, columns=["Original String", "Modified String"])
