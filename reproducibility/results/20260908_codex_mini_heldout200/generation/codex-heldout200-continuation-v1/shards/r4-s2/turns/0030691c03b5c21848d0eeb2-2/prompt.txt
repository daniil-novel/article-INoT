import pandas as pd
from scipy import stats
import matplotlib.pyplot as plt

def task_func(df):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("input df is not a DataFrame")

    filtered_df = df.drop_duplicates(subset="name", keep="first").copy()

    filtered_df["age"] = pd.to_numeric(filtered_df["age"], errors="coerce")
    filtered_df["score"] = pd.to_numeric(filtered_df["score"], errors="coerce")
    filtered_df = filtered_df.dropna(subset=["age", "score"])

    if len(filtered_df) < 2:
        raise ValueError("Not enough data points for linear regression")

    regression = stats.linregress(filtered_df["age"], filtered_df["score"])

    fig, ax = plt.subplots()
    ax.scatter(filtered_df["age"], filtered_df["score"])

    x_line = [filtered_df["age"].min(), filtered_df["age"].max()]
    y_line = [regression.slope * x + regression.intercept for x in x_line]
    ax.plot(x_line, y_line)

    ax.set_title("Linear Regression")
    ax.set_xlabel("Age")
    ax.set_ylabel("Score")

    return plt, ax
