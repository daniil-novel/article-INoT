import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

def task_func(df):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("df must be a pandas DataFrame.")
    if df.empty:
        raise ValueError("Input dataframe is empty.")

    data = df.copy()

    def _find_datetime_series(frame: pd.DataFrame) -> pd.Series:
        # Prefer a datetime-like index
        if isinstance(frame.index, pd.DatetimeIndex):
            return pd.Series(frame.index, index=frame.index)

        # Prefer columns with date/time-like names
        for col in frame.columns:
            col_name = str(col).lower()
            if any(token in col_name for token in ("date", "time", "timestamp", "datetime")):
                parsed = pd.to_datetime(frame[col], errors="coerce")
                if parsed.notna().any():
                    return parsed

        # Fallback: any non-numeric column that can mostly be parsed as datetime
        for col in frame.columns:
            if pd.api.types.is_numeric_dtype(frame[col]):
                continue
            parsed = pd.to_datetime(frame[col], errors="coerce")
            if parsed.notna().sum() >= max(2, int(0.8 * len(frame))):
                return parsed

        # Fallback: parse a non-numeric, non-RangeIndex index
        if not isinstance(frame.index, pd.RangeIndex) and not pd.api.types.is_numeric_dtype(frame.index):
            parsed_index = pd.to_datetime(frame.index, errors="coerce")
            if parsed_index.notna().sum() >= max(2, int(0.8 * len(frame))):
                return pd.Series(parsed_index, index=frame.index)

        raise ValueError("Could not determine a datetime column or datetime-like index.")

    def _find_close_series(frame: pd.DataFrame, excluded=None) -> pd.Series:
        excluded = set(excluded or [])

        # Priority to columns that clearly look like closing prices
        priority_keywords = ("close", "adj close", "adj_close", "closing", "price")
        for col in frame.columns:
            if col in excluded:
                continue
            col_name = str(col).lower()
            if any(keyword in col_name for keyword in priority_keywords):
                series = pd.to_numeric(frame[col], errors="coerce")
                if series.notna().any():
                    return series

        # Otherwise choose the numeric-convertible column with the most valid values
        best_col = None
        best_series = None
        best_valid = -1
        for col in frame.columns:
            if col in excluded:
                continue
            series = pd.to_numeric(frame[col], errors="coerce")
            valid = int(series.notna().sum())
            if valid > best_valid:
                best_valid = valid
                best_col = col
                best_series = series

        if best_series is None or best_valid == 0:
            raise ValueError("Could not determine a closing price column.")

        return best_series

    date_series = _find_datetime_series(data)
    close_series = _find_close_series(data)

    clean = pd.DataFrame({"date": date_series, "close": close_series})
    clean = clean.dropna(subset=["date", "close"]).sort_values("date").reset_index(drop=True)

    if len(clean) < 2:
        raise ValueError("At least 2 valid historical observations are required.")

    # Use elapsed seconds from the first timestamp for numerical stability
    origin = clean["date"].iloc[0]
    x = (clean["date"] - origin).dt.total_seconds().to_numpy().reshape(-1, 1)
    y = clean["close"].astype(float).to_numpy()

    model = LinearRegression()
    model.fit(x, y)

    step = 24 * 60 * 60  # constant 1-day step in seconds
    last_offset = float((clean["date"].iloc[-1] - origin).total_seconds())
    future_offsets = (last_offset + step * np.arange(1, 8)).reshape(-1, 1)

    future_dates = [clean["date"].iloc[-1] + pd.to_timedelta(step * i, unit="s") for i in range(1, 8)]
    predicted_prices = model.predict(future_offsets).tolist()

    fig, ax = plt.subplots(figsize=(10, 5))
    ax.plot(clean["date"], clean["close"], label="Historical Close", color="tab:blue")
    ax.plot(
        future_dates,
        predicted_prices,
        label="Predicted Next 7 Days",
        color="tab:orange",
        linestyle="--",
        marker="o",
    )
    ax.set_xlabel("Date")
    ax.set_ylabel("Closing Price")
    ax.set_title("Stock Closing Price Forecast")
    ax.legend()
    ax.grid(True, alpha=0.3)
    fig.autofmt_xdate()

    return predicted_prices, ax
