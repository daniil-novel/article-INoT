import pandas as pd
from sklearn.cluster import DBSCAN

def task_func(data, cols):
    """
    Perform DBSCAN clustering on the specified columns and store labels in a new
    'Cluster' column.

    Parameters
    ----------
    data : pandas.DataFrame or array-like
        Input data.
    cols : list-like
        Column names to use for clustering.

    Returns
    -------
    df : pandas.DataFrame
        The DataFrame with a new 'Cluster' column.
    """
    df = data.copy() if isinstance(data, pd.DataFrame) else pd.DataFrame(data)

    # Always return a DataFrame with the required output column for empty input
    if df.empty:
        df["Cluster"] = pd.Series(dtype=int)
        return df

    if cols is None:
        raise ValueError("'cols' must be provided and contain at least one column name.")

    cols = list(cols)
    if len(cols) == 0:
        raise ValueError("'cols' must contain at least one column name.")

    missing = [col for col in cols if col not in df.columns]
    if missing:
        raise KeyError(f"The following columns are missing from the DataFrame: {missing}")

    X = df[cols].apply(pd.to_numeric, errors="raise")

    if X.isna().any().any():
        raise ValueError("Selected columns contain missing values; DBSCAN requires complete numeric data.")

    model = DBSCAN(eps=3, min_samples=2)
    df["Cluster"] = model.fit_predict(X)

    return df
