import socket
from ipaddress import IPv4Network
from threading import Thread

def task_func(ip_range, port):
    """
    Scan an IPv4 network range and report whether a TCP port is open on each host.

    Args:
        ip_range: An IPv4Network instance or CIDR string.
        port: TCP port number (1..65535).

    Returns:
        dict: Mapping of IP address strings to booleans indicating port status.
    """
    if isinstance(ip_range, IPv4Network):
        network = ip_range
    else:
        try:
            network = IPv4Network(ip_range, strict=False)
        except Exception as exc:
            raise ValueError("ip_range must be a valid IPv4 network") from exc

    if not isinstance(port, int) or isinstance(port, bool) or not (1 <= port <= 65535):
        raise ValueError("port must be an integer between 1 and 65535")

    results = {}

    def probe(host):
        status = False
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                sock.settimeout(1.0)
                status = sock.connect_ex((str(host), port)) == 0
        except OSError:
            status = False
        results[str(host)] = status

    threads = []
    for host in network.hosts():
        thread = Thread(target=probe, args=(host,))
        thread.daemon = True
        thread.start()
        threads.append(thread)

    for thread in threads:
        thread.join()

    return results
