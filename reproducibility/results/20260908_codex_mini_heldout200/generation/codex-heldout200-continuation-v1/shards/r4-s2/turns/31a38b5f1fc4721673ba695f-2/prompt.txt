from datetime import datetime
import pandas as pd
import matplotlib.pyplot as plt

# Constants
DATE_FORMAT = "%Y-%m-%d %H:%M:%S"


def task_func(timestamps):
    if not timestamps:
        raise ValueError("Input list of timestamps is empty")

    datetimes = [datetime.fromtimestamp(ts) for ts in timestamps]
    df = pd.DataFrame({
        "Timestamp": timestamps,
        "Datetime": datetimes,
    })

    fig, ax = plt.subplots()
    date_numbers = pd.to_datetime(df["Datetime"]).map(pd.Timestamp.toordinal)
    ax.hist(date_numbers, bins=10)

    # Format the x-axis to display datetimes using DATE_FORMAT
    ax.xaxis.set_major_formatter(
        plt.matplotlib.dates.DateFormatter(DATE_FORMAT)
    )
    fig.autofmt_xdate()

    return df, ax
