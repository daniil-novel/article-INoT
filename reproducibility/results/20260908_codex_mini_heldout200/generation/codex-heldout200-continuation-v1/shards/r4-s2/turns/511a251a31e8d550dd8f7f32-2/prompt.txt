import numpy as np
import itertools

def task_func(matrix):
    arr = np.asarray(matrix)

    if arr.ndim != 2:
        raise ValueError("Input must be a 2D numpy array or 2D array-like structure.")

    sorted_arr = np.sort(arr, axis=None)

    unique_pairs = []
    seen = set()
    for pair in itertools.combinations(sorted_arr.tolist(), 2):
        if pair not in seen:
            seen.add(pair)
            unique_pairs.append(pair)

    return sorted_arr, unique_pairs
