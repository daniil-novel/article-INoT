```python
import random
import string
import pandas as pd

def task_func(data_list, seed=0):
    rng = random.Random(seed)

    originals = []
    modified = []

    for s in data_list:
        parts = s.split(',')
        idx = rng.randrange(len(parts))
        segment = parts[idx]
        replacement = ''.join(rng.choices(string.ascii_lowercase, k=len(segment)))
        parts[idx] = replacement

        originals.append(s)
        modified.append(','.join(parts))

    return pd.DataFrame({
        'Original String': originals,
        'Modified String': modified
    })
```
