import pandas as pd
import seaborn as sns
import numpy as np
import ast

def task_func(csv_file):
    import matplotlib.pyplot as plt

    try:
        df = pd.read_csv(csv_file)
    except pd.errors.EmptyDataError:
        df = pd.DataFrame(columns=["email", "list"])

    required_cols = {"email", "list"}
    missing = required_cols - set(df.columns)
    if missing:
        raise KeyError(f"Missing required column(s): {', '.join(sorted(missing))}")

    def _parse_list(value):
        if isinstance(value, list):
            return value
        if isinstance(value, (tuple, np.ndarray, pd.Series)):
            return list(value)

        try:
            if pd.isna(value):
                return []
        except Exception:
            pass

        if isinstance(value, str):
            text = value.strip()
            if not text or text.lower() in {"nan", "none", "null"}:
                return []
            try:
                parsed = ast.literal_eval(text)
            except (ValueError, SyntaxError):
                return []
            if isinstance(parsed, list):
                return parsed
            if isinstance(parsed, (tuple, np.ndarray, pd.Series)):
                return list(parsed)
            return [parsed]

        return [value]

    def _stats(lst):
        if len(lst) == 0:
            return 0.0, np.nan, np.nan

        numeric = pd.to_numeric(pd.Series(lst), errors="coerce")
        if numeric.isna().any():
            return np.nan, np.nan, np.nan

        arr = numeric.to_numpy(dtype=float)
        total = float(np.sum(arr))
        mean = float(np.mean(arr))
        std = float(np.std(arr, ddof=1)) if arr.size > 1 else np.nan
        return total, mean, std

    result_df = df.loc[:, ["email", "list"]].copy()
    result_df["list"] = result_df["list"].apply(_parse_list)

    stats = result_df["list"].apply(_stats)
    result_df[["sum", "mean", "std"]] = pd.DataFrame(stats.tolist(), index=result_df.index)
    result_df = result_df[["email", "list", "sum", "mean", "std"]]

    fig, ax = plt.subplots()
    sns.histplot(x=result_df["mean"].dropna(), ax=ax)

    return result_df, ax
