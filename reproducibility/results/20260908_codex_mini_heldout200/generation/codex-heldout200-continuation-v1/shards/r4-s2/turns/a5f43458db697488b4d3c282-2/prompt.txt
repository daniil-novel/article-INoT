from datetime import datetime
import pytz
from dateutil.parser import parse

def task_func(date_str, tz_str):
    tz = pytz.timezone(tz_str)
    dt = parse(date_str)

    if dt.tzinfo is None:
        local_dt = tz.localize(dt, is_dst=None)
    else:
        local_dt = dt.astimezone(tz)

    next_year_new_year = tz.localize(
        datetime(local_dt.year + 1, 1, 1, 0, 0, 0),
        is_dst=None
    )

    return int((next_year_new_year - local_dt).total_seconds())
