import pandas as pd
import random
import re

def task_func(data_list, seed=None):
    if seed is not None:
        random.seed(seed)

    original_strings = []
    modified_strings = []

    for s in data_list:
        if not isinstance(s, str):
            s = str(s)

        parts = s.split(',')
        op = random.choice(['remove', 'replace', 'shuffle', 'randomize'])

        if op == 'remove':
            if len(parts) > 1:
                idx = random.randrange(len(parts))
                parts = parts[:idx] + parts[idx + 1:]
            else:
                op = random.choice(['replace', 'shuffle', 'randomize'])

        if op == 'replace':
            idx = random.randrange(len(parts))
            parts[idx] = 'random_string'
        elif op == 'shuffle':
            random.shuffle(parts)
        elif op == 'randomize':
            parts = random.sample(parts, len(parts))

        original_strings.append(s)
        modified_strings.append(','.join(parts))

    df = pd.DataFrame({
        'Original String': original_strings,
        'Modified String': modified_strings
    })
    return df
