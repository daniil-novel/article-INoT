```python
from PIL import Image
import matplotlib.pyplot as plt
import numpy as np
import os

def task_func(img_path, angle):
    if not os.path.exists(img_path):
        raise FileNotFoundError(f"No such file: {img_path}")

    with Image.open(img_path) as img:
        original_image = img.copy()
        original_array = np.array(original_image)

        rotated_image = original_image.rotate(angle, expand=True)
        rotated_array = np.array(rotated_image)

    fig, axes = plt.subplots(1, 2, figsize=(10, 5))

    def _show(ax, arr, title):
        if arr.ndim == 2:
            ax.imshow(arr, cmap="gray")
        elif arr.ndim == 3 and arr.shape[2] == 1:
            ax.imshow(arr.squeeze(), cmap="gray")
        else:
            ax.imshow(arr)
        ax.set_title(title)
        ax.axis("off")

    _show(axes[0], original_array, "Original Image")
    _show(axes[1], rotated_array, f"Rotated Image ({angle}°)")
    plt.tight_layout()
    plt.show()

    return original_array, rotated_array
```
