import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns
import numpy as np

def task_func(data_url="http://lib.stat.cmu.edu/datasets/boston", seed=42, save_path=None):
    fig = None
    try:
        np.random.seed(seed)

        raw_df = pd.read_csv(data_url, sep=r"\s+", skiprows=22, header=None)
        raw_df = raw_df.apply(pd.to_numeric, errors="coerce")

        if raw_df.empty or raw_df.shape[0] < 2 or raw_df.shape[1] < 3:
            raise ValueError("Boston Housing dataset could not be parsed.")

        if raw_df.shape[0] % 2 != 0:
            raise ValueError("Unexpected Boston Housing dataset shape.")

        if raw_df.isnull().any().any():
            raise ValueError("Boston Housing dataset contains non-numeric or missing values.")

        values = raw_df.to_numpy(dtype=float)
        data = np.hstack([values[::2, :], values[1::2, :2]])
        target = values[1::2, 2]

        if data.shape[0] == 0:
            raise ValueError("Boston Housing dataset is empty after parsing.")

        columns = [
            "CRIM", "ZN", "INDUS", "CHAS", "NOX", "RM", "AGE",
            "DIS", "RAD", "TAX", "PTRATIO", "B", "LSTAT", "MEDV"
        ]
        df = pd.DataFrame(np.column_stack([data, target]), columns=columns)

        corr = df.corr()

        fig, ax = plt.subplots(figsize=(12, 10))
        sns.heatmap(
            corr,
            ax=ax,
            cmap="coolwarm",
            square=True,
            linewidths=0.5,
            cbar_kws={"shrink": 0.8},
        )
        ax.set_title("Boston Housing Correlation Heatmap")

        if save_path is not None:
            fig.savefig(save_path, bbox_inches="tight")

        return ax

    except Exception as exc:
        if fig is not None:
            plt.close(fig)
        raise ValueError("An error occurred while generating or saving the plot.") from exc
