## Edge cases to handle

### Input/path issues
- `directory` does not exist
- `directory` is not a directory
- `directory` is empty
- path is unreadable or inaccessible due to permissions

### Backup creation issues
- backup root `/tmp/backup` does not exist
- backup root cannot be created
- destination path already exists
- filesystem errors during copy
- files are locked or in use

### Cleanup issues
- some files cannot be deleted due to permissions
- read-only files
- nested subdirectories fail to remove
- cleanup partially succeeds

### Special file types
- symbolic links
- sockets, FIFOs, device files
- very deep directory trees
- large files

### Consistency concerns
- source changes while backup is in progress
- backup succeeds but cleanup fails
- cleanup starts only after a successful backup
- partial failures should be reported, not silently ignored

## Error-handling strategy

- Use a list of human-readable error strings.
- Catch exceptions around:
  - backup directory creation
  - recursive copy
  - file removal
  - directory removal
- Prefer continuing cleanup after individual file errors, so one bad file does not stop the whole operation.
- Do not let a single failure crash the function unless the backup itself cannot be created at all.

import os
import shutil
import uuid

# Constants
BACKUP_DIR = '/tmp/backup'

def get_unique_backup_dir():
    return os.path.join(BACKUP_DIR, f"backup-{uuid.uuid4().hex}")

def _is_within(path, parent):
    try:
        path = os.path.abspath(path)
        parent = os.path.abspath(parent)
        return os.path.commonpath([path, parent]) == parent
    except Exception:
        return False

def _safe_makedirs(path, errors):
    try:
        os.makedirs(path, exist_ok=True)
        return True
    except Exception as exc:
        errors.append(f"Failed to create backup directory '{path}': {exc}")
        return False

def _copy_tree(src, dst, errors):
    try:
        os.makedirs(dst, exist_ok=True)
    except Exception as exc:
        errors.append(f"Failed to create destination '{dst}': {exc}")
        return

    try:
        entries = list(os.scandir(src))
    except Exception as exc:
        errors.append(f"Failed to list source directory '{src}': {exc}")
        return

    for entry in entries:
        src_path = entry.path
        dst_path = os.path.join(dst, entry.name)

        # Avoid recursively copying the backup root if it is inside the source tree.
        if _is_within(src_path, BACKUP_DIR):
            continue

        try:
            if entry.is_symlink():
                target = os.readlink(src_path)
                os.symlink(target, dst_path)
            elif entry.is_dir(follow_symlinks=False):
                _copy_tree(src_path, dst_path, errors)
                try:
                    shutil.copystat(src_path, dst_path, follow_symlinks=False)
                except Exception as exc:
                    errors.append(f"Failed to preserve directory metadata for '{src_path}': {exc}")
            else:
                shutil.copy2(src_path, dst_path, follow_symlinks=False)
        except Exception as exc:
            errors.append(f"Failed to copy '{src_path}' to '{dst_path}': {exc}")

    try:
        shutil.copystat(src, dst, follow_symlinks=False)
    except Exception as exc:
        errors.append(f"Failed to preserve metadata for '{src}': {exc}")

def _remove_path(path, errors):
    try:
        if os.path.islink(path) or os.path.isfile(path):
            os.unlink(path)
            return
        if os.path.isdir(path):
            try:
                entries = list(os.scandir(path))
            except Exception as exc:
                errors.append(f"Failed to list directory '{path}' for cleanup: {exc}")
                return

            for entry in entries:
                child_path = entry.path

                # Never delete the backup root if it is nested inside the source tree.
                if _is_within(child_path, BACKUP_DIR):
                    continue

                _remove_path(child_path, errors)

            try:
                os.rmdir(path)
            except OSError:
                # Directory may not be empty due to skipped backup content or an error.
                pass
            return

        # Fallback for unusual filesystem entries
        os.unlink(path)
    except FileNotFoundError:
        return
    except Exception as exc:
        errors.append(f"Failed to remove '{path}': {exc}")

def task_func(directory):
    errors = []
    backup_dir = get_unique_backup_dir()

    try:
        src = os.fspath(directory)
    except Exception as exc:
        errors.append(f"Invalid directory path: {exc}")
        return backup_dir, errors

    src = os.path.abspath(src)

    if not os.path.exists(src):
        errors.append(f"Source directory does not exist: '{src}'")
        return backup_dir, errors

    if not os.path.isdir(src):
        errors.append(f"Source path is not a directory: '{src}'")
        return backup_dir, errors

    if not _safe_makedirs(BACKUP_DIR, errors):
        return backup_dir, errors

    # Create the backup first.
    _copy_tree(src, backup_dir, errors)

    # Only clean the source if the backup path was created and no copy errors occurred.
    if not errors:
        try:
            entries = list(os.scandir(src))
        except Exception as exc:
            errors.append(f"Failed to list source directory '{src}' for cleanup: {exc}")
            return backup_dir, errors

        for entry in entries:
            _remove_path(entry.path, errors)

    return backup_dir, errors
