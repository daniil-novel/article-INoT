import numpy as np
from sklearn.decomposition import PCA

def task_func(tuples_list, n_components):
    """
    Perform PCA on a list of tuples and return the transformed data.
    """
    if not isinstance(n_components, (int, np.integer)) or isinstance(n_components, bool):
        raise TypeError("n_components must be a positive integer.")
    if n_components <= 0:
        raise ValueError("n_components must be a positive integer.")

    tuples_list = list(tuples_list)
    if len(tuples_list) == 0:
        return np.empty((0, 0), dtype=float)

    data = np.asarray(tuples_list, dtype=float)

    if data.ndim != 2:
        raise ValueError("tuples_list must be a 2D sequence of equal-length tuples.")

    n_samples, n_features = data.shape
    if n_samples == 0 or n_features == 0:
        return np.empty((n_samples, 0), dtype=float)

    n_components = min(n_components, n_samples, n_features)

    pca = PCA(n_components=n_components)
    transformed_data = pca.fit_transform(data)
    return transformed_data
