from collections import Counter
import random
import string

# Constants
LETTERS = string.ascii_letters


def task_func(list_of_lists):
    """
    Replace each nested sublist with a random letter and return a count of
    the generated letters as a dictionary.
    """
    counts = Counter()

    def traverse(obj, is_root=False):
        if not isinstance(obj, list):
            return

        for item in obj:
            traverse(item, is_root=False)

        if not is_root:
            counts[random.choice(LETTERS)] += 1

    traverse(list_of_lists, is_root=True)
    return dict(counts)
