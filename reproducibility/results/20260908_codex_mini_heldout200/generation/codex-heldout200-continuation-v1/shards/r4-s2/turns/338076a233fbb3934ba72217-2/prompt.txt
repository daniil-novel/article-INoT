import pandas as pd
from sklearn.cluster import DBSCAN
import numpy as np

def task_func(data, cols):
    df = data.copy() if isinstance(data, pd.DataFrame) else pd.DataFrame(data)

    if cols is None:
        cols = list(df.columns)
    elif isinstance(cols, str):
        cols = [cols]
    else:
        cols = list(cols)

    if len(cols) == 0:
        raise ValueError("cols must contain at least one column name.")

    missing_cols = [c for c in cols if c not in df.columns]
    if missing_cols:
        raise KeyError(f"Missing columns in data: {missing_cols}")

    features = df[cols].apply(pd.to_numeric, errors="coerce")
    valid_mask = features.notna().all(axis=1)

    labels = np.full(len(df), -1, dtype=int)
    if valid_mask.any():
        clustering = DBSCAN(eps=3, min_samples=2)
        labels[valid_mask.to_numpy()] = clustering.fit_predict(features.loc[valid_mask])

    df["Cluster"] = labels
    return df
