import math
from random import randint
import pandas as pd

def task_func(cities_list):
    cities = list(cities_list)
    populations = [math.ceil(randint(1, 1_000_000) / 1000) * 1000 for _ in cities]
    return pd.DataFrame({

        'Population': populations
    })
