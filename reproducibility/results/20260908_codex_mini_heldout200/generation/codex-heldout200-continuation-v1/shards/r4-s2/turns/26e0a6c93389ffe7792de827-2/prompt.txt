import numpy as np
from scipy import stats

def task_func(df, column, alpha):
    if not (0 < alpha < 1):
        return False

    try:
        values = np.asarray(df[column], dtype=float)
    except Exception:
        return False

    values = values[np.isfinite(values)]

    if values.size < 3:
        return False

    if np.all(values == values[0]):
        return False

    try:
        _, p_value = stats.shapiro(values)
    except Exception:
        return False

    if np.isnan(p_value):
        return False

    return bool(p_value > alpha)
