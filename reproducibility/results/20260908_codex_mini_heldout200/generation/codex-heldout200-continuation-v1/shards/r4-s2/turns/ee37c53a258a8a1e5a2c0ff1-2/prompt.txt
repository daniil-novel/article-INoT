import pandas as pd
from sklearn.preprocessing import LabelEncoder

def task_func(data):
    """
    Encode categorical values using sklearn's LabelEncoder and return a
    mapping DataFrame with columns: Category, Encoded.
    """
    if data is None:
        return pd.DataFrame(columns=["Category", "Encoded"])

    if isinstance(data, pd.DataFrame):
        raise ValueError("data must be one-dimensional, not a DataFrame")

    if isinstance(data, pd.Series):
        series = data.copy()
    elif isinstance(data, (str, bytes)):
        series = pd.Series([data])
    else:
        try:
            series = pd.Series(list(data))
        except TypeError:
            series = pd.Series([data])

    series = series.dropna()

    if series.empty:
        return pd.DataFrame(columns=["Category", "Encoded"])

    le = LabelEncoder()
    try:
        le.fit(series)
    except Exception as exc:
        raise ValueError(f"Unable to encode categorical data: {exc}") from exc

    return pd.DataFrame(
        {
            "Category": le.classes_,
            "Encoded": le.transform(le.classes_),
        }
    )
