import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns
import numpy as np

def task_func(data_url="http://lib.stat.cmu.edu/datasets/boston", seed=42):
    import io
    import urllib.request

    try:
        np.random.seed(seed)

        # Load text from URL or local path
        if isinstance(data_url, str) and data_url.startswith(("http://", "https://")):
            with urllib.request.urlopen(data_url) as response:
                text = response.read().decode("utf-8", errors="ignore")
        else:
            with open(data_url, "r", encoding="utf-8", errors="ignore") as f:
                text = f.read()

        # Parse the original Boston Housing dataset format
        raw = np.loadtxt(io.StringIO(text), skiprows=22)

        if raw.ndim != 2 or raw.shape[0] < 2 or raw.shape[0] % 2 != 0 or raw.shape[1] < 3:
            raise ValueError("Unexpected Boston dataset format.")

        # Standard Boston dataset reconstruction
        data = np.hstack([raw[::2, :], raw[1::2, :2]])
        target = raw[1::2, 2]

        feature_names = [
            "CRIM", "ZN", "INDUS", "CHAS", "NOX", "RM", "AGE",
            "DIS", "RAD", "TAX", "PTRATIO", "B", "LSTAT"
        ]

        if data.shape[1] != len(feature_names):
            raise ValueError("Boston dataset feature count mismatch.")

        df = pd.DataFrame(data, columns=feature_names)
        df["MEDV"] = target

        corr = df.corr(numeric_only=True)

        fig, ax = plt.subplots(figsize=(12, 10))
        sns.heatmap(
            corr,
            ax=ax,
            cmap="coolwarm",
            annot=False,
            square=True,
            linewidths=0.5,
            cbar=True
        )
        ax.set_title("Boston Housing Dataset Correlation Heatmap")
        ax.tick_params(axis="x", rotation=45)
        ax.tick_params(axis="y", rotation=0)
        fig.tight_layout()

        # Optional save path: set either task_func.save_path or SAVE_PATH before calling
        save_path = getattr(task_func, "save_path", None) or globals().get("SAVE_PATH", None)
        if save_path:
            fig.savefig(save_path, bbox_inches="tight")

        return ax

    except Exception as exc:
        raise ValueError(f"Failed to generate or save the correlation heatmap: {exc}") from exc
