import os
import shutil
import uuid

# Constants
BACKUP_DIR = '/tmp/backup'


def get_unique_backup_dir():
    return os.path.join(BACKUP_DIR, f"backup_{uuid.uuid4().hex}")


def _is_within_directory(parent, child):
    parent = os.path.abspath(parent)
    child = os.path.abspath(child)
    try:
        return os.path.commonpath([parent, child]) == parent
    except ValueError:
        return False


def _make_unique_path(path):
    candidate = os.path.abspath(path)
    while os.path.exists(candidate):
        candidate = f"{candidate}_{uuid.uuid4().hex}"
    return candidate


def _copy_contents(src_dir, dst_dir, errors):
    try:
        with os.scandir(src_dir) as it:
            for entry in it:
                src_path = entry.path
                dst_path = os.path.join(dst_dir, entry.name)

                try:
                    if entry.is_symlink():
                        target = os.readlink(src_path)
                        try:
                            os.symlink(target, dst_path)
                        except (OSError, NotImplementedError) as exc:
                            errors.append(
                                f"Failed to copy symlink '{src_path}' to '{dst_path}': {exc}"
                            )
                    elif entry.is_dir(follow_symlinks=False):
                        try:
                            os.makedirs(dst_path, exist_ok=True)
                        except OSError as exc:
                            errors.append(
                                f"Failed to create directory '{dst_path}': {exc}"
                            )
                            continue
                        _copy_contents(src_path, dst_path, errors)
                    elif entry.is_file(follow_symlinks=False):
                        shutil.copy2(src_path, dst_path)
                    else:
                        errors.append(f"Skipped unsupported file type: {src_path}")
                except OSError as exc:
                    errors.append(f"Failed to copy '{src_path}' to '{dst_path}': {exc}")
    except OSError as exc:
        errors.append(f"Failed to read directory '{src_dir}': {exc}")


def _remove_path(path, errors, delete_root=True):
    try:
        if os.path.islink(path) or os.path.isfile(path):
            os.unlink(path)
            return

        if os.path.isdir(path):
            with os.scandir(path) as it:
                for entry in it:
                    _remove_path(entry.path, errors, delete_root=True)
            if delete_root:
                os.rmdir(path)
            return

        os.unlink(path)
    except FileNotFoundError:
        return
    except OSError as exc:
        errors.append(f"Failed to remove '{path}': {exc}")


def _remove_tree_contents(directory, errors):
    try:
        with os.scandir(directory) as it:
            for entry in it:
                _remove_path(entry.path, errors, delete_root=True)
    except OSError as exc:
        errors.append(f"Failed to read directory '{directory}' during cleanup: {exc}")


def task_func(directory):
    errors = []

    backup_dir = _make_unique_path(get_unique_backup_dir())

    try:
        source_dir = os.fspath(directory)
    except TypeError as exc:
        errors.append(f"Invalid directory path: {exc}")
        return backup_dir, errors

    source_dir = os.path.abspath(source_dir)

    if not os.path.exists(source_dir):
        errors.append(f"Source directory does not exist: {source_dir}")
        return backup_dir, errors

    if not os.path.isdir(source_dir):
        errors.append(f"Source path is not a directory: {source_dir}")
        return backup_dir, errors

    if _is_within_directory(source_dir, backup_dir):
        alt_base = os.path.join(
            os.path.dirname(source_dir.rstrip(os.sep)) or os.sep,
            f"backup_{uuid.uuid4().hex}",
        )
        backup_dir = _make_unique_path(alt_base)

        if _is_within_directory(source_dir, backup_dir):
            errors.append("Unable to create a backup directory outside the source directory.")
            return backup_dir, errors

    try:
        os.makedirs(backup_dir, exist_ok=False)
    except FileExistsError:
        backup_dir = _make_unique_path(backup_dir)
        try:
            os.makedirs(backup_dir, exist_ok=False)
        except OSError as exc:
            errors.append(f"Failed to create backup directory '{backup_dir}': {exc}")
            return backup_dir, errors
    except OSError as exc:
        errors.append(f"Failed to create backup directory '{backup_dir}': {exc}")
        return backup_dir, errors

    _copy_contents(source_dir, backup_dir, errors)
    _remove_tree_contents(source_dir, errors)

    return backup_dir, errors
