import pandas as pd
from scipy import stats
import matplotlib.pyplot as plt

def task_func(df):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("input df must be a pandas DataFrame")

    required_columns = {"name", "age", "score"}
    missing_columns = required_columns - set(df.columns)
    if missing_columns:
        raise ValueError(f"missing required columns: {', '.join(sorted(missing_columns))}")

    clean_df = df.copy()

    # Exclude all rows whose name appears more than once
    clean_df = clean_df[~clean_df["name"].duplicated(keep=False)].copy()

    # Keep only valid numeric age/score rows
    clean_df["age"] = pd.to_numeric(clean_df["age"], errors="coerce")
    clean_df["score"] = pd.to_numeric(clean_df["score"], errors="coerce")
    clean_df = clean_df.dropna(subset=["age", "score"])

    if len(clean_df) < 2 or clean_df["age"].nunique() < 2:
        raise ValueError("insufficient data for linear regression")

    result = stats.linregress(clean_df["age"], clean_df["score"])

    fig, ax = plt.subplots()
    ax.scatter(clean_df["age"], clean_df["score"])

    x_vals = clean_df["age"].sort_values()
    y_vals = result.intercept + result.slope * x_vals
    ax.plot(x_vals, y_vals)

    ax.set_title("Linear Regression")
    ax.set_xlabel("Age")
    ax.set_ylabel("Score")

    return plt, ax
