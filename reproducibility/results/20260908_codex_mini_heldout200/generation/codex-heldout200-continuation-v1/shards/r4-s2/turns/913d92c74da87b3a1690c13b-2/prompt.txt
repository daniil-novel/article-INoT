import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

def task_func(df):
    step = 24 * 60 * 60  # 24 hours in seconds
    fig, ax = plt.subplots()

    if df is None:
        preds = [np.nan] * 7
        ax.set_title("No data available")
        return preds, ax

    if not isinstance(df, pd.DataFrame):
        df = pd.DataFrame(df)

    if df.empty:
        preds = [np.nan] * 7
        ax.set_title("No data available")
        return preds, ax

    data = df.copy()

    # --- Detect close price column ---
    close_col = None
    lower_map = {str(col).strip().lower(): col for col in data.columns}

    for key in ["close", "closing price", "adj close", "adj_close", "adjclose"]:
        if key in lower_map:
            close_col = lower_map[key]
            break

    if close_col is None:
        for col in data.columns:
            if "close" in str(col).strip().lower():
                close_col = col
                break

    if close_col is None:
        numeric_cols = data.select_dtypes(include=[np.number]).columns.tolist()
        if numeric_cols:
            close_col = numeric_cols[0]
        else:
            close_col = data.columns[-1]

    close_series = pd.to_numeric(data[close_col], errors="coerce")

    # --- Detect / construct time values ---
    time_series = None

    # 1) DatetimeIndex
    if isinstance(data.index, pd.DatetimeIndex):
        time_series = pd.Series(data.index, index=data.index)

    # 2) Columns that look like time/date/timestamp
    if time_series is None:
        time_candidates = []
        for col in data.columns:
            if col == close_col:
                continue
            col_lower = str(col).strip().lower()
            if any(k in col_lower for k in ["date", "datetime", "timestamp", "time"]):
                time_candidates.append(col)

        # Try preferred candidates first, then any other non-close columns
        candidate_order = time_candidates + [c for c in data.columns if c != close_col and c not in time_candidates]

        for col in candidate_order:
            s = data[col]

            parsed = pd.to_datetime(s, errors="coerce")
            valid_count = parsed.notna().sum()

            # If numeric data was not parsed well, try interpreting as Unix seconds
            if valid_count == 0 and pd.api.types.is_numeric_dtype(s):
                parsed = pd.to_datetime(s, errors="coerce", unit="s")
                valid_count = parsed.notna().sum()

            if valid_count > 0:
                time_series = pd.Series(parsed, index=data.index)
                break

    # 3) Fallback: synthetic equally spaced timestamps in seconds
    if time_series is None:
        time_series = pd.Series(np.arange(len(data), dtype=float) * step, index=data.index)

    # Convert datetime-like timestamps to Unix seconds
    if pd.api.types.is_datetime64_any_dtype(time_series):
        ts = pd.Series(time_series)
        if getattr(ts.dt, "tz", None) is not None:
            ts = ts.dt.tz_convert(None)
        time_seconds = ts.astype("int64") // 10**9
    else:
        # Already numeric synthetic/epoch-like values
        time_seconds = pd.to_numeric(time_series, errors="coerce")

    work = pd.DataFrame({"time": time_seconds, "close": close_series}).dropna()

    if work.empty:
        preds = [np.nan] * 7
        ax.set_title("Insufficient data for regression")
        return preds, ax

    # Sort and aggregate duplicate timestamps
    work = work.sort_values("time")
    work = work.groupby("time", as_index=False)["close"].mean()

    if len(work) < 2 or work["time"].nunique() < 2:
        last_price = float(work["close"].iloc[-1])
        preds = [last_price] * 7

        hist_x = pd.to_datetime(work["time"], unit="s", errors="coerce")
        future_times = work["time"].iloc[-1] + step * np.arange(1, 8)
        future_x = pd.to_datetime(future_times, unit="s", errors="coerce")

        ax.plot(hist_x, work["close"], label="Historical Close", color="tab:blue")
        ax.plot(future_x, preds, "--o", label="Predicted Close", color="tab:orange")
        ax.axvline(hist_x.iloc[-1] if isinstance(hist_x, pd.Series) else hist_x[-1],
                   color="gray", linestyle=":", alpha=0.7)
        ax.set_xlabel("Date")
        ax.set_ylabel("Close Price")
        ax.set_title("Stock Closing Price Forecast")
        ax.legend()
        fig.autofmt_xdate()
        return preds, ax

    # --- Fit simple linear regression ---
    base_time = float(work["time"].iloc[0])
    X = (work["time"] - base_time).to_numpy().reshape(-1, 1)
    y = work["close"].to_numpy()

    model = LinearRegression()
    model.fit(X, y)

    # --- Predict next 7 days using fixed 24*60*60 second step ---
    future_times = work["time"].iloc[-1] + step * np.arange(1, 8)
    future_X = (future_times - base_time).reshape(-1, 1)
    preds = model.predict(future_X).tolist()
    preds = [float(p) for p in preds]

    # --- Plot historical and predicted data ---
    hist_x = pd.to_datetime(work["time"], unit="s", errors="coerce")
    future_x = pd.to_datetime(future_times, unit="s", errors="coerce")

    ax.plot(hist_x, work["close"], label="Historical Close", color="tab:blue")
    ax.plot(future_x, preds, "--o", label="Predicted Close", color="tab:orange")
    ax.axvline(hist_x.iloc[-1] if isinstance(hist_x, pd.Series) else hist_x[-1],
               color="gray", linestyle=":", alpha=0.7)

    ax.set_xlabel("Date")
    ax.set_ylabel("Close Price")
    ax.set_title("Stock Closing Price Forecast")
    ax.legend()
    fig.autofmt_xdate()

    return preds, ax
