import requests
from PIL import Image
import io

def task_func(url):
    try:
        response = requests.get(url, timeout=5, allow_redirects=False)
        if not (200 <= response.status_code < 300):
            raise ValueError(f"HTTP request failed with status code {response.status_code}")

        image = Image.open(io.BytesIO(response.content))
        image.load()  # Force full decoding/validation
        return image

    except requests.exceptions.RequestException as e:
        raise ValueError(f"Invalid URL or request failed: {e}") from e
    except (OSError, ValueError) as e:
        raise ValueError(f"Invalid image data or HTTP error: {e}") from e
