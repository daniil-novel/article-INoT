import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

def task_func(df, column, bins=30, density=True, alpha=0.6, color="g", seed=None):
    if column not in df.columns:
        raise KeyError(f"Column '{column}' not found in DataFrame.")

    series = df[column].dropna()
    try:
        data = np.asarray(series, dtype=float)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Column '{column}' must contain numeric data.") from exc

    if data.size == 0:
        raise ValueError(f"Column '{column}' has no valid numeric data to plot.")

    fig, ax = plt.subplots()

    # Plot histogram
    ax.hist(data, bins=bins, density=density, alpha=alpha, color=color, label="Data")

    mu, sigma = norm.fit(data)

    # Overlay fitted normal distribution curve
    if sigma > 0:
        xmin, xmax = float(np.min(data)), float(np.max(data))
        if xmin == xmax:
            pad = 1.0 if xmin == 0 else max(1.0, abs(xmin) * 0.1)
            xmin -= pad
            xmax += pad

        x = np.linspace(xmin, xmax, 200)
        y = norm.pdf(x, mu, sigma)

        if not density:
            bin_edges = np.histogram_bin_edges(data, bins=bins)
            if len(bin_edges) > 1:
                bin_width = float(np.mean(np.diff(bin_edges)))
                y = y * len(data) * bin_width

        ax.plot(x, y, "r-", linewidth=2, label="Fitted normal")
    else:
        # Degenerate case: constant data
        ax.axvline(mu, color="r", linewidth=2, label="Fitted normal")

    ax.set_title(f"Histogram of {column} with fitted normal curve")
    ax.set_xlabel(column)
    ax.set_ylabel("Density" if density else "Count")
    ax.legend()
    ax.grid(True, alpha=0.3)

    return ax
