from sklearn.cluster import KMeans
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

def task_func(df1, df2, column1="feature1", column2="feature2"):
    if column1 not in df1.columns:
        raise ValueError(f"'{column1}' not found in df1.")
    if column2 not in df2.columns:
        raise ValueError(f"'{column2}' not found in df2.")

    common_keys = [c for c in df1.columns if c in df2.columns and c not in {column1, column2}]
    if not common_keys:
        raise ValueError("No common id column found to merge df1 and df2.")

    left = df1[common_keys + [column1]].rename(columns={column1: "__feature1__"})
    right = df2[common_keys + [column2]].rename(columns={column2: "__feature2__"})

    merged = left.merge(right, on=common_keys, how="inner")
    if merged.empty:
        raise ValueError("Merge produced no rows; cannot perform clustering.")

    x = pd.to_numeric(merged["__feature1__"], errors="coerce").to_numpy(dtype=float)
    y = pd.to_numeric(merged["__feature2__"], errors="coerce").to_numpy(dtype=float)

    features = np.column_stack((x, y))
    valid_mask = np.isfinite(features).all(axis=1)
    features = features[valid_mask]

    if features.shape[0] < 2:
        raise ValueError("At least 2 valid merged rows are required for KMeans clustering.")

    model = KMeans(n_clusters=2, n_init=10)
    labels = model.fit_predict(features).astype(np.int32, copy=False)

    fig, ax = plt.subplots()
    ax.scatter(features[:, 0], features[:, 1], c=labels, cmap="viridis")
    ax.set_xlabel(column1)
    ax.set_ylabel(column2)

    return labels, ax
