import nltk
from string import punctuation
import seaborn as sns
import matplotlib.pyplot as plt

# Constants
PUNCTUATION = set(punctuation)

from collections import Counter

def task_func(text):
    tokens = text.split()
    valid_tokens = [
        token for token in tokens
        if token.startswith("$") and not set(token).issubset(PUNCTUATION)
    ]

    if not valid_tokens:
        return None

    counts = Counter(valid_tokens)
    words = list(counts.keys())
    freqs = list(counts.values())

    fig, ax = plt.subplots(figsize=(max(6, len(words) * 0.8), 4))
    sns.barplot(x=words, y=freqs, ax=ax, color="steelblue")

    ax.set_xlabel("Words")
    ax.set_ylabel("Frequency")
    ax.tick_params(axis="x", rotation=45)

    plt.tight_layout()
    return ax
