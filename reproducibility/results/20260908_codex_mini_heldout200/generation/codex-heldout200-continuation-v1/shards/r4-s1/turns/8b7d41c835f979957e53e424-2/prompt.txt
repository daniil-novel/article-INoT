import pandas as pd
import matplotlib.pyplot as plt

def task_func(df, items=None, locations=None):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("df must be a pandas DataFrame")

    required_columns = {"Item", "Location"}
    if not required_columns.issubset(df.columns):
        raise ValueError("df must contain 'Item' and 'Location' columns")

    # Default to the unique values present in the dataframe, preserving order of appearance.
    if items is None:
        items = pd.Index(df["Item"].dropna().tolist()).unique().tolist()
    else:
        items = list(items)

    if locations is None:
        locations = pd.Index(df["Location"].dropna().tolist()).unique().tolist()
    else:
        locations = list(locations)

    # Filter to requested items and locations.
    filtered = df[df["Item"].isin(items) & df["Location"].isin(locations)].copy()

    # Count occurrences for each Location / Item pair.
    counts = (
        filtered.groupby(["Location", "Item"])
        .size()
        .unstack(fill_value=0)
    )

    # Preserve requested ordering and include missing combinations as zeros.
    counts = counts.reindex(index=locations, columns=items, fill_value=0)

    fig, ax = plt.subplots()

    if not counts.empty and len(counts.columns) > 0:
        counts.plot(kind="bar", ax=ax)
    else:
        # Keep a valid axes object even when there is no data to plot.
        ax.set_xticks([])
        ax.set_yticks([])

    ax.set_title("Distribution of Items Across Locations")
    ax.set_xlabel("Location")
    ax.set_ylabel("Count")
    if len(items) > 0:
        ax.legend(title="Item")

    plt.tight_layout()
    return ax
