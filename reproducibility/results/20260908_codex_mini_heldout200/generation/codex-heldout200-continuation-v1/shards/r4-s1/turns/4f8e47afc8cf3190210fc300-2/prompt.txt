import string
import random

def task_func(length, random_seed=None):
    BRACKETS = "(){}[]"
    alphabet = string.ascii_lowercase + BRACKETS
    rng = random.Random(random_seed)
    return "".join(rng.choice(alphabet) for _ in range(length))
