import numpy as np
import random
# Constants
COLORS = ['b', 'g', 'r', 'c', 'm', 'y', 'k']

def task_func(ax):
    # Randomize the radial label position
    ax.set_rlabel_position(random.randint(0, 359))

    # Generate a smooth polar domain
    theta = np.linspace(0, 2 * np.pi, 300)

    # Random sine wave parameters
    freq = random.randint(1, 6)
    phase = random.uniform(0, 2 * np.pi)
    amplitude = random.uniform(0.5, 2.0)
    offset = random.uniform(1.0, 3.0)

    # Build the radial values
    r = offset + amplitude * np.sin(freq * theta + phase)

    # Choose a random color and plot
    color = random.choice(COLORS)
    ax.plot(theta, r, color=color)

    return color
