import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm
import pandas as pd

def task_func(df, column, bins=30, density=True, alpha=0.6, color="g", seed=None):
    if seed is not None:
        np.random.seed(seed)

    try:
        series = df[column]
    except Exception as exc:
        raise KeyError(f"Column {column!r} not found in DataFrame.") from exc

    values = pd.to_numeric(series, errors="coerce").to_numpy(dtype=float)
    values = values[np.isfinite(values)]

    if values.size == 0:
        raise ValueError(f"Column {column!r} contains no finite numeric values.")

    fig, ax = plt.subplots()
    ax.hist(values, bins=bins, density=density, alpha=alpha, color=color, label="Histogram")

    mu, sigma = norm.fit(values)
    if not np.isfinite(sigma) or sigma <= 0:
        sigma = np.finfo(float).eps

    data_min = float(np.min(values))
    data_max = float(np.max(values))

    if data_min == data_max:
        pad = max(1.0, abs(data_min) * 0.1)
        x_min, x_max = data_min - pad, data_max + pad
    else:
        pad = 0.1 * (data_max - data_min)
        x_min, x_max = data_min - pad, data_max + pad

    x = np.linspace(x_min, x_max, 401)
    y = norm.pdf(x, loc=mu, scale=sigma)

    if not density:
        _, edges = np.histogram(values, bins=bins)
        bin_widths = np.diff(edges)
        bin_width = float(np.mean(bin_widths)) if bin_widths.size else 1.0
        y = y * values.size * bin_width

    ax.plot(x, y, "r-", linewidth=2, label="Fitted normal PDF")
    ax.set_xlabel(str(column))
    ax.set_ylabel("Density" if density else "Count")
    ax.set_title(f"Histogram of {column} with fitted normal curve")
    ax.legend()

    return ax
