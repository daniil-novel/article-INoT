from sklearn.cluster import KMeans
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


def task_func(df1, df2, column1="feature1", column2="feature2"):
    if column1 not in df1.columns:
        raise KeyError(f"Column '{column1}' not found in df1.")
    if column2 not in df2.columns:
        raise KeyError(f"Column '{column2}' not found in df2.")

    x = pd.to_numeric(df1[column1], errors="coerce").reset_index(drop=True)
    y = pd.to_numeric(df2[column2], errors="coerce").reset_index(drop=True)

    merged = pd.concat([x, y], axis=1)
    merged.columns = [column1, column2]
    merged = merged.dropna()

    if len(merged) < 2:
        raise ValueError("KMeans with k=2 requires at least 2 valid paired samples.")

    data = merged[[column1, column2]].to_numpy()

    kmeans = KMeans(n_clusters=2, n_init=10, random_state=0)
    labels = kmeans.fit_predict(data).astype(np.int32, copy=False)

    fig, ax = plt.subplots()
    ax.scatter(data[:, 0], data[:, 1], c=labels, cmap="viridis", s=40)
    ax.set_xlabel(column1)
    ax.set_ylabel(column2)

    return labels, ax
