```python
import seaborn as sns
from scipy.stats import chi2_contingency

import pandas as pd
import matplotlib.pyplot as plt


def task_func(df1, df2, column1="feature1", column2="feature2"):
    """
    Merge two dataframes on 'id', run a chi-square independence test on
    the contingency table built from `column1` and `column2`, and plot a heatmap.

    Returns
    -------
    tuple
        (p_value, heatmap_axes)
    """
    if "id" not in df1.columns:
        raise KeyError("df1 must contain an 'id' column.")
    if "id" not in df2.columns:
        raise KeyError("df2 must contain an 'id' column.")

    merged = df1.merge(df2, on="id", how="inner")

    if column1 not in merged.columns:
        raise KeyError(f"'{column1}' not found after merging the dataframes.")
    if column2 not in merged.columns:
        raise KeyError(f"'{column2}' not found after merging the dataframes.")

    data = merged[[column1, column2]].dropna()

    if data.empty:
        raise ValueError("No valid rows remain after removing missing values.")

    contingency_table = pd.crosstab(data[column1], data[column2])

    if contingency_table.shape[0] < 2 or contingency_table.shape[1] < 2:
        raise ValueError(
            "Chi-square test requires at least 2 categories in each variable."
        )

    _, p, _, _ = chi2_contingency(contingency_table)

    fig, ax = plt.subplots(figsize=(8, 6))
    sns.heatmap(contingency_table, annot=True, fmt="d", cmap="YlGnBu", ax=ax)
    ax.set_xlabel(column2)
    ax.set_ylabel(column1)
    ax.set_title("Contingency Table Heatmap")

    return p, ax
```
