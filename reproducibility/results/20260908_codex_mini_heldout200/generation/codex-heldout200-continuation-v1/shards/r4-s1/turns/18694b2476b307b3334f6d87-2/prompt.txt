import itertools
import numpy as np
import matplotlib.pyplot as plt

def task_func(elements, subset_size):
    """
    Generate all subsets of a given size from `elements`, plot a histogram
    of their sums, and return the Axes object along with the combinations
    and their sums.

    Parameters
    ----------
    elements : tuple
        Input tuple of elements.
    subset_size : int
        Size of each subset to generate.

    Returns
    -------
    matplotlib.axes.Axes
        Axes object containing the histogram.
    list
        List of all combinations of subsets.
    list
        List of sums for each subset.
    """
    if isinstance(subset_size, bool) or not isinstance(subset_size, (int, np.integer)):
        raise TypeError("subset_size must be an integer")
    if subset_size < 0:
        raise ValueError("subset_size must be non-negative")

    if subset_size > len(elements):
        combinations_list = []
        subset_sums = []
    else:
        combinations_list = list(itertools.combinations(elements, subset_size))
        subset_sums = [sum(combination) for combination in combinations_list]

    fig, ax = plt.subplots()
    ax.hist(subset_sums, bins="auto", edgecolor="black")
    ax.set_xlabel("Sum of subset")
    ax.set_ylabel("Frequency")
    ax.set_title("Histogram of subset sums")

    return ax, combinations_list, subset_sums
