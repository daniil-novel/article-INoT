import pandas as pd
from sklearn.feature_selection import SelectKBest, f_classif
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df1, df2):
    if not isinstance(df1, pd.DataFrame):
        df1 = pd.DataFrame(df1)

    if isinstance(df2, pd.DataFrame):
        if df2.shape[1] == 0:
            raise ValueError("df2 must contain at least one target column.")
        y = df2.iloc[:, 0].copy()
    elif isinstance(df2, pd.Series):
        y = df2.copy()
    else:
        y = pd.Series(df2)
        if len(y) != len(df1):
            raise ValueError("df2 must have the same length as df1 when provided as an array-like object.")
        y.index = df1.index

    X = df1.select_dtypes(include="number").copy()
    if X.shape[1] < 2:
        raise ValueError("At least 2 numeric feature columns are required for SelectKBest(k=2).")

    data = pd.concat([X, y.rename("__target__")], axis=1).dropna()
    if data.empty:
        raise ValueError("No rows remain after dropping missing values.")

    X_clean = data[X.columns]
    y_clean = data["__target__"]

    if X_clean.shape[1] < 2:
        raise ValueError("At least 2 numeric feature columns are required after cleaning.")
    if y_clean.nunique() < 2:
        raise ValueError("Target must contain at least two classes.")

    selector = SelectKBest(score_func=f_classif, k=2)
    selector.fit(X_clean, y_clean)

    selected_features = X_clean.columns[selector.get_support(indices=True)].tolist()

    corr = data[selected_features].corr()

    fig, ax = plt.subplots()
    sns.heatmap(corr, annot=True, cmap="coolwarm", vmin=-1, vmax=1, ax=ax)

    return selected_features, ax
