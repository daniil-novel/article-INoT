import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

def task_func(length):
    MU = 0
    SIGMA = 1

    try:
        length = int(length)
    except (TypeError, ValueError) as exc:
        raise ValueError("length must be an integer-like value") from exc

    if length < 0:
        raise ValueError("length must be non-negative")

    distribution = np.random.normal(loc=MU, scale=SIGMA, size=length)

    fig, ax = plt.subplots()

    if length > 0:
        bins = min(30, max(1, int(np.sqrt(length))))
        ax.hist(distribution, bins=bins, density=True, alpha=0.6, edgecolor="black", label="Histogram")

        x_min = float(np.min(distribution))
        x_max = float(np.max(distribution))
        if np.isclose(x_min, x_max):
            pad = 1.0
        else:
            pad = 0.1 * (x_max - x_min)
        x = np.linspace(x_min - pad, x_max + pad, 400)
    else:
        ax.hist(distribution, bins=30, density=True, alpha=0.6, edgecolor="black", label="Histogram")
        x = np.linspace(-4, 4, 400)

    ax.plot(x, norm.pdf(x, loc=MU, scale=SIGMA), color="red", linewidth=2, label="N(0,1) PDF")
    ax.set_xlabel("Value")
    ax.set_ylabel("Density")
    ax.set_title("Normal Distribution Histogram and PDF")
    ax.legend()

    return distribution, ax
