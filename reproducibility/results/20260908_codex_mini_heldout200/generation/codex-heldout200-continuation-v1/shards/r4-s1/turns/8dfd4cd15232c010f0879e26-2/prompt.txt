import pandas as pd
import matplotlib.pyplot as plt

def task_func(df, items=None, locations=None):
    if not isinstance(df, pd.DataFrame):
        raise ValueError("'df' must be a pandas DataFrame.")
    if "Item" not in df.columns or "Location" not in df.columns:
        raise ValueError("DataFrame must contain 'Item' and 'Location' columns.")

    def _normalize चयन(values, default_values):
        if values is None:
            return list(default_values)
        if isinstance(values, (str, bytes)):
            values = [values]
        else:
            try:
                values = list(values)
            except TypeError:
                values = [values]
        # Preserve order while removing duplicates
        return list(dict.fromkeys(values))

    default_items = pd.unique(df["Item"].dropna()).tolist()
    default_locations = pd.unique(df["Location"].dropna()).tolist()

    selected_items = _normalize चयन(items, default_items)
    selected_locations = _normalize चयन(locations, default_locations)

    filtered = df.dropna(subset=["Item", "Location"])
    if selected_items:
        filtered = filtered[filtered["Item"].isin(selected_items)]
    else:
        filtered = filtered.iloc[0:0]
    if selected_locations:
        filtered = filtered[filtered["Location"].isin(selected_locations)]
    else:
        filtered = filtered.iloc[0:0]

    counts = (
        filtered.groupby(["Location", "Item"])
        .size()
        .unstack(fill_value=0)
        .reindex(index=selected_locations, columns=selected_items, fill_value=0)
    )

    fig, ax = plt.subplots()

    x_positions = list(range(len(selected_locations)))
    if selected_items and selected_locations:
        width = 0.8 / max(len(selected_items), 1)
        center_offset = (len(selected_items) - 1) / 2
        for idx, item in enumerate(selected_items):
            offsets = [x + (idx - center_offset) * width for x in x_positions]
            ax.bar(offsets, counts[item].tolist(), width=width, label=str(item))
        ax.set_xticks(x_positions)
        ax.set_xticklabels([str(loc) for loc in selected_locations], rotation=45, ha="right")
        ax.legend()
    else:
        ax.set_xticks([])
        ax.set_xticklabels([])

    ax.set_title("Distribution of Items Across Locations")
    ax.set_xlabel("Location")
    ax.set_ylabel("Count")
    ax.margins(x=0.02)
    fig.tight_layout()

    return ax
