```python
import numpy as np
import matplotlib.pyplot as plt

# Constants
CITIES = ['New York', 'London', 'Beijing', 'Tokyo', 'Sydney', 'Paris', 'Berlin', 'Moscow', 'Madrid', 'Rome']

def task_func(city_dict, max_range=1000000, seed=0):
    np.random.seed(seed)

    favorite_cities = set(city_dict.values())
    populations = {}

    for city in CITIES:
        if city in favorite_cities:
            populations[city] = int(np.random.randint(1, max_range + 1))
        else:
            populations[city] = -1

    fig, ax = plt.subplots()
    ax.bar(list(populations.keys()), list(populations.values()))
    ax.set_xlabel("City")
    ax.set_ylabel("Population")
    ax.set_title("City Populations")
    plt.xticks(rotation=45, ha="right")

    return populations, ax
```
