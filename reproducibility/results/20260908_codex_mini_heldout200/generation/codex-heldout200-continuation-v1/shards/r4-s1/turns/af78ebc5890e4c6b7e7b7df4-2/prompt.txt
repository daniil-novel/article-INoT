from datetime import datetime
import pytz
import re
from faker import Faker

def task_func(epoch_milliseconds, seed=0, timezones=["UTC"]):
    fake = Faker()
    fake.seed_instance(seed)

    # Convert epoch milliseconds to a datetime in the current system timezone
    local_dt = datetime.fromtimestamp(epoch_milliseconds / 1000).astimezone()

    utc_offset_pattern = re.compile(r"^UTC([+-])(0\d|1\d|2[0-3]):([0-5]\d)$")

    if timezones is None:
        candidates = []
    elif isinstance(timezones, str):
        candidates = [timezones]
    else:
        try:
            candidates = list(timezones)
        except TypeError:
            candidates = []

    valid_timezones = []
    for tz in candidates:
        if not isinstance(tz, str):
            continue
        if tz == "UTC" or tz in pytz.all_timezones or utc_offset_pattern.match(tz):
            if tz not in valid_timezones:
                valid_timezones.append(tz)

    if not valid_timezones:
        selected_timezone = "UTC"
    else:
        selected_timezone = fake.random_element(elements=tuple(valid_timezones))

    if selected_timezone == "UTC":
        tzinfo = pytz.UTC
    else:
        match = utc_offset_pattern.match(selected_timezone)
        if match:
            sign = 1 if match.group(1) == "+" else -1
            hours = int(match.group(2))
            minutes = int(match.group(3))
            total_minutes = sign * (hours * 60 + minutes)
            tzinfo = pytz.FixedOffset(total_minutes)
        else:
            tzinfo = pytz.timezone(selected_timezone)

    event_dt = local_dt.astimezone(tzinfo)
    event_name = fake.catch_phrase()

    return {
        event_name: [
            {
                "date": event_dt.strftime("%Y-%m-%d"),
                "time": event_dt.strftime("%H:%M:%S"),
                "timezone": selected_timezone,
            }
        ]
    }
