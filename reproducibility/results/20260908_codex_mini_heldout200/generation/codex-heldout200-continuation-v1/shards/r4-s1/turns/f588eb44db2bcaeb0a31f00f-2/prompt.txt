import json
import numpy as np
from collections import defaultdict
import matplotlib.pyplot as plt

def task_func(input_file):
    with open(input_file, "r", encoding="utf-8") as f:
        data = json.load(f)

    if not isinstance(data, list):
        raise ValueError("Input JSON must be a list of dictionaries.")

    grouped = defaultdict(list)

    for item in data:
        if not isinstance(item, dict):
            raise ValueError("Each element in the JSON list must be a dictionary.")
        for key, value in item.items():
            grouped[key].append(value)

    result = {}
    plots = []

    numeric_types = (int, float, np.integer, np.floating)

    for key, values in grouped.items():
        numeric_values = [
            float(v)
            for v in values
            if isinstance(v, numeric_types) and not isinstance(v, bool) and np.isfinite(v)
        ]

        if numeric_values:
            arr = np.asarray(numeric_values, dtype=float)
            mean_val = float(np.mean(arr))
            median_val = float(np.median(arr))
        else:
            mean_val = np.nan
            median_val = np.nan

        result[key] = {"mean": mean_val, "median": median_val}

        fig, ax = plt.subplots(figsize=(5, 4))
        ax.bar(["mean", "median"], [mean_val, median_val], color=["tab:blue", "tab:orange"])
        ax.set_title(str(key))
        ax.set_ylabel("value")
        ax.grid(axis="y", linestyle="--", alpha=0.3)
        plots.append(ax)

    return result, plots
