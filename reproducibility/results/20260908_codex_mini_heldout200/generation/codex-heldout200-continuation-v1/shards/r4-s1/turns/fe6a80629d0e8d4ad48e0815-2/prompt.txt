from datetime import datetime
import pytz
import re
from faker import Faker


def task_func(epoch_milliseconds, seed=0, timezones=["UTC"]):
    fake = Faker()
    fake.seed_instance(seed)

    utc_dt = datetime.fromtimestamp(epoch_milliseconds / 1000, tz=pytz.UTC)
    local_dt = utc_dt.astimezone()

    if timezones is None:
        candidates = []
    elif isinstance(timezones, str):
        candidates = [timezones]
    else:
        candidates = list(timezones)

    utc_offset_pattern = re.compile(
        r"^UTC([+-])((?:0?\d)|(?:1\d)|(?:2[0-3])):([0-5]\d)$"
    )

    valid_timezones = []
    seen = set()
    for tz in candidates:
        if not isinstance(tz, str):
            continue
        if tz in pytz.all_timezones or utc_offset_pattern.match(tz):
            if tz not in seen:
                seen.add(tz)
                valid_timezones.append(tz)

    selected_timezone = "UTC" if not valid_timezones else fake.random_element(valid_timezones)
    event_name = fake.catch_phrase()

    return {
        event_name: [
            {
                "date": local_dt.strftime("%Y-%m-%d"),
                "time": local_dt.strftime("%H:%M:%S"),
                "timezone": selected_timezone,
            }
        ]
    }
