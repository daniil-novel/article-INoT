import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.axes import Axes
from statsmodels.tsa.arima.model import ARIMA
from typing import List, Tuple


def task_func(df: pd.DataFrame) -> Tuple[List[float], Axes]:
    """
    Forecast the next 7 days of share closing prices using an ARIMA model.

    Parameters
    ----------
    df : pd.DataFrame
        Input data containing a closing-price column and, optionally, a date column
        or datetime index.

    Returns
    -------
    Tuple[List[float], Axes]
        A list of 7 forecasted prices and a Matplotlib Axes object with the plot.
    """
    if df is None or df.empty:
        raise ValueError("Input DataFrame is empty.")

    # --- Identify closing-price column ---
    close_candidates = []
    for col in df.columns:
        col_lower = str(col).strip().lower()
        if col_lower == "close":
            close_candidates = [col]
            break
        if "close" in col_lower:
            close_candidates.append(col)

    if not close_candidates:
        raise ValueError("No closing price column found. Expected a column like 'Close'.")

    close_col = close_candidates[0]

    # --- Prepare and clean the series ---
    data = df.copy()

    # Sort by date if a date-like column or datetime index exists
    date_col = None
    for col in data.columns:
        if col == close_col:
            continue
        col_series = data[col]
        if pd.api.types.is_datetime64_any_dtype(col_series):
            date_col = col
            break

    if date_col is not None:
        data = data.sort_values(by=date_col)
    elif isinstance(data.index, pd.DatetimeIndex):
        data = data.sort_index()

    close_series = pd.to_numeric(data[close_col], errors="coerce").dropna()

    if len(close_series) < 3:
        raise ValueError("No valid closing price data available or too few observations for ARIMA.")

    close_series = close_series.astype(float)

    # --- Fit ARIMA with fallbacks ---
    candidate_orders = [(5, 1, 0), (3, 1, 0), (1, 1, 0), (0, 1, 0), (1, 0, 0)]
    model_fit = None

    for order in candidate_orders:
        try:
            model = ARIMA(close_series, order=order)
            model_fit = model.fit()
            break
        except Exception:
            continue

    if model_fit is None:
        raise ValueError("ARIMA model could not be fit to the provided closing price data.")

    # --- Forecast next 7 days ---
    forecast_result = model_fit.forecast(steps=7)
    forecast_values = [float(x) for x in forecast_result.tolist()]

    # --- Build x-axis values for plotting ---
    if isinstance(data.index, pd.DatetimeIndex):
        hist_x = data.index[-len(close_series):]
        last_date = hist_x[-1]
        future_x = pd.bdate_range(start=last_date + pd.Timedelta(days=1), periods=7)
    elif date_col is not None and pd.api.types.is_datetime64_any_dtype(data[date_col]):
        hist_x = data[date_col].iloc[-len(close_series):]
        last_date = hist_x.iloc[-1]
        future_x = pd.bdate_range(start=last_date + pd.Timedelta(days=1), periods=7)
    else:
        hist_x = range(len(close_series))
        future_x = range(len(close_series), len(close_series) + 7)

    # --- Plot historical data and forecast ---
    fig, ax = plt.subplots(figsize=(10, 5))
    ax.plot(hist_x, close_series.values, label="Historical Close", color="tab:blue")
    ax.plot(future_x, forecast_values, label="7-Day Forecast", color="tab:orange", linestyle="--", marker="o")

    ax.set_title("ARIMA Forecast of Share Closing Prices")
    ax.set_xlabel("Date" if not isinstance(hist_x, range) else "Time Step")
    ax.set_ylabel("Closing Price")
    ax.legend()
    ax.grid(True, alpha=0.3)

    plt.tight_layout()
    return forecast_values, ax
