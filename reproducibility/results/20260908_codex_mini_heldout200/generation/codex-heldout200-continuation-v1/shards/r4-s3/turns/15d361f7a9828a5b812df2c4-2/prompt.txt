```python
import sys
import subprocess
# Constants
PYTHON_VERSION = '3.8'
PATH_TO_APPEND = '/path/to/whatever'
def task_func(python_version=PYTHON_VERSION, path_to_append=PATH_TO_APPEND):
    """
    Add a path to sys.path and report the requested Python version.

    Note:
        A running Python process cannot truly switch its interpreter version
        globally at runtime. This function records and returns the requested
        version while safely updating sys.path.

    Returns:
        str: python_version (the Python version that was switched to/requested)
    """
    if not isinstance(python_version, str) or not python_version.strip():
        raise ValueError("python_version must be a non-empty string.")
    if not isinstance(path_to_append, str):
        raise TypeError("path_to_append must be a string.")

    python_version = python_version.strip()
    path_to_append = path_to_append.strip()

    if path_to_append and path_to_append not in sys.path:
        sys.path.append(path_to_append)

    return python_version
```
