import pandas as pd
import seaborn as sns
from sklearn.preprocessing import StandardScaler

def task_func(df1, df2):
    if not isinstance(df1, pd.DataFrame) or not isinstance(df2, pd.DataFrame):
        raise TypeError("Both df1 and df2 must be pandas DataFrame objects.")
    if "id" not in df1.columns or "id" not in df2.columns:
        raise ValueError("Both df1 and df2 must contain an 'id' column.")

    df1_scaled = df1.copy()
    numeric_cols = df1_scaled.select_dtypes(include="number").columns.tolist()
    numeric_cols = [col for col in numeric_cols if col != "id"]

    if numeric_cols:
        if df1_scaled[numeric_cols].isna().any().any():
            raise ValueError(
                "df1 contains missing values in numeric columns; please impute or drop them before scaling."
            )
        scaler = StandardScaler()
        df1_scaled[numeric_cols] = scaler.fit_transform(df1_scaled[numeric_cols])
        pair_plot = sns.pairplot(df1_scaled[numeric_cols])
    else:
        pair_plot = sns.PairGrid(pd.DataFrame({"placeholder": [0]}))

    merged_df = pd.merge(df1_scaled, df2.copy(), how="outer", on="id")
    return merged_df, pair_plot
