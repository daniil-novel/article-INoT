import random
import string
import hashlib
import time

def task_func(data_dict: dict, seed=0) -> dict:
    data_dict = dict(data_dict)
    data_dict["a"] = 1

    rng = random.Random(seed)
    salt = "".join(rng.choice(string.ascii_lowercase) for _ in range(5))

    for key, value in list(data_dict.items()):
        hashed_value = hashlib.sha256((str(value) + salt).encode("utf-8")).hexdigest()
        data_dict[key] = hashed_value

    data_dict["timestamp"] = int(time.time())
    return data_dict
