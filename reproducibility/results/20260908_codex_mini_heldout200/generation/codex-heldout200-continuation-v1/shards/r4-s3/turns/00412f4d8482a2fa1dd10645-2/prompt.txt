import numpy as np
import seaborn as sns

def task_func(P, T):
    """
    Multiply a 2D matrix P by a 3D tensor T, then reduce the tensor contraction
    to a 2D array for heatmap visualization.

    Returns
    -------
    tuple
        (result: numpy.ndarray, ax: matplotlib.axes.Axes)
    """
    # Validate input types
    if not isinstance(P, np.ndarray) or not isinstance(T, np.ndarray):
        raise TypeError("P and T must both be numpy.ndarray instances.")

    # Validate numeric, real-valued data
    if not np.issubdtype(P.dtype, np.number) or not np.issubdtype(T.dtype, np.number):
        raise TypeError("P and T must contain numeric values.")
    if np.iscomplexobj(P) or np.iscomplexobj(T):
        raise TypeError("P and T must be real-valued arrays.")

    # Validate shapes
    if P.ndim != 2:
        raise ValueError("P must be a 2D matrix.")
    if T.ndim != 3:
        raise ValueError("T must be a 3D tensor.")
    if P.shape[1] != T.shape[0]:
        raise ValueError(
            f"Incompatible shapes: P.shape={P.shape} and T.shape={T.shape}. "
            "Expected P.shape[1] == T.shape[0]."
        )

    # Matrix-tensor multiplication: (m, n) x (n, p, q) -> (m, p, q)
    contracted = np.tensordot(P, T, axes=([1], [0]))

    # Reduce to 2D for heatmap visualization
    result = np.sum(contracted, axis=-1)

    if result.size == 0:
        raise ValueError("The resulting array is empty and cannot be plotted as a heatmap.")

    import matplotlib.pyplot as plt

    fig, ax = plt.subplots()
    sns.heatmap(result, ax=ax)
    ax.set_title("Matrix-Tensor Product Heatmap")

    return result, ax
