import numpy as np
import random
def task_func(length=10000, seed=0):
    if length < 0:
        raise ValueError("length must be non-negative")

    random.seed(seed)

    walk = [0]
    position = 0

    for _ in range(length):
        step = random.choice([-1, 1])
        position += step
        walk.append(position)

    return np.array(walk)
