import re
import pandas as pd

def task_func(df: pd.DataFrame) -> int:
    if not isinstance(df, pd.DataFrame):
        raise TypeError
    pattern = r'[(){}[\]]'
    return int(sum(len(re.findall(pattern, str(value))) for value in df.to_numpy().ravel()))
