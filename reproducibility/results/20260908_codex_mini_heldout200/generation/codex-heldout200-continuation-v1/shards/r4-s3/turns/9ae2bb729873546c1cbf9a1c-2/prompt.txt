import matplotlib.pyplot as plt
import numpy as np

def task_func(ax, radius):
    if not hasattr(ax, "name") or ax.name != "polar":
        raise TypeError("ax must be a polar Axes")
    if radius < 0:
        raise ValueError("radius must be non-negative")

    theta = np.linspace(0, 2 * np.pi, 361)
    r = np.full_like(theta, radius, dtype=float)
    ax.plot(theta, r)

    else:
        rticks = np.unique(np.linspace(0, radius, 4))
        rticks = rticks.tolist()
