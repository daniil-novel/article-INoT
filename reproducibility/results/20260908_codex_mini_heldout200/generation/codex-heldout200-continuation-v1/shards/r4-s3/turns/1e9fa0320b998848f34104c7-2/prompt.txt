import os
import glob
import shutil
import time

# Constants
FILE_EXTENSIONS = ['.txt', '.csv', '.xlsx', '.docx', '.pdf']


def task_func(my_path: str, days_old: int) -> str:
    if days_old < 0:
        raise ValueError("days_old must be non-negative")

    if not os.path.isdir(my_path):
        raise NotADirectoryError(f"Not a valid directory: {my_path}")

    archive_path = os.path.join(my_path, "archive")
    os.makedirs(archive_path, exist_ok=True)

    cutoff_time = time.time() - (days_old * 24 * 60 * 60)

    def _unique_destination(dest_dir: str, filename: str) -> str:
        base_name, ext = os.path.splitext(filename)
        candidate = os.path.join(dest_dir, filename)
        counter = 1
        while os.path.exists(candidate):
            candidate = os.path.join(dest_dir, f"{base_name}_{counter}{ext}")
            counter += 1
        return candidate

    for entry in os.scandir(my_path):
        if not entry.is_file():
            continue

        _, ext = os.path.splitext(entry.name)
        if ext.lower() not in FILE_EXTENSIONS:
            continue

        try:
            mtime = os.path.getmtime(entry.path)
        except OSError:
            continue

        if mtime < cutoff_time:
            destination = _unique_destination(archive_path, entry.name)
            shutil.move(entry.path, destination)

    return archive_path
