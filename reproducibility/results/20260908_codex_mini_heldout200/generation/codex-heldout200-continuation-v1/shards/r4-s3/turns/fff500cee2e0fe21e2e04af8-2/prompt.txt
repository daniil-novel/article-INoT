import os
import pandas as pd
from dateutil.parser import parse
import matplotlib.pyplot as plt

OUTPUT_DIR = './output'

def task_func(csv_path=os.path.join(OUTPUT_DIR, 'data.csv'), date_column='date'):
    df = pd.read_csv(csv_path)

    if date_column not in df.columns:
        raise KeyError(f"Column {date_column!r} not found in CSV.")

    def _safe_parse(value):
        if pd.isna(value):
            return pd.NaT
        try:
            return parse(str(value))
        except (ValueError, TypeError, OverflowError):
            return pd.NaT

    # Convert the date column to datetime objects
    df[date_column] = df[date_column].apply(_safe_parse)

    # Extract years from valid datetimes
    years = df[date_column].dropna().map(lambda x: x.year)

    # Plot histogram of year distribution
    fig, ax = plt.subplots()
    if not years.empty:
        min_year = int(years.min())
        max_year = int(years.max())
        ax.hist(years, bins=range(min_year, max_year + 2), edgecolor='black')
    else:
        ax.hist([], bins=1, edgecolor='black')

    ax.set_xlabel('Year')
    ax.set_ylabel('Count')
    ax.set_title('Year Distribution of Dates')

    return ax
