```python
import subprocess
from ipaddress import IPv4Network
import os


def task_func(ip_range):
    network = ip_range if isinstance(ip_range, IPv4Network) else IPv4Network(ip_range, strict=False)

    result = {}
    ping_args = ["ping", "-n", "1"] if os.name == "nt" else ["ping", "-c", "1"]

    for ip in network:
        try:
            completed = subprocess.run(
                ping_args + [str(ip)],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
                check=False,
            )
            result[str(ip)] = completed.returncode == 0
        except subprocess.CalledProcessError:
            raise

    return result
```
