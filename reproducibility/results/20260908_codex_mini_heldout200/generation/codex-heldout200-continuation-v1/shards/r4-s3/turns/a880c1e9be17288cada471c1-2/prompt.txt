import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler

def task_func(df):
    """
    Standardize numeric columns in a DataFrame and plot their correlation matrix.

    Parameters
    ----------
    df : pandas.DataFrame
        Input DataFrame.

    Returns
    -------
    tuple[pandas.DataFrame, matplotlib.figure.Figure]
        Standardized DataFrame and the heatmap figure.
    """
    if df.empty:
        raise ValueError("The DataFrame is empty.")

    numeric_cols = df.select_dtypes(include=[np.number]).columns
    if len(numeric_cols) == 0:
        raise ValueError("No numeric columns are present in the DataFrame.")

    standardized_df = df.copy(deep=True)

    scaler = StandardScaler()
    standardized_values = scaler.fit_transform(standardized_df[numeric_cols])
    standardized_df[numeric_cols] = standardized_values

    corr_matrix = standardized_df[numeric_cols].corr()

    size = max(6, 0.8 * len(numeric_cols))
    fig, ax = plt.subplots(figsize=(size, size))
    sns.heatmap(
        corr_matrix,
        ax=ax,
        cmap="coolwarm",
        center=0,
        square=True,
        linewidths=0.5,
        cbar=True
    )
    ax.set_title("Correlation Matrix of Numeric Columns")
    plt.tight_layout()

    return standardized_df, fig
