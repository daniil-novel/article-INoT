import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler

# Constants
PLOT_TITLE = "Scaled Values"


def task_func(data_dict):
    if not isinstance(data_dict, dict):
        raise TypeError("data_dict must be a dictionary.")

    fig, ax = plt.subplots()
    ax.set_title(PLOT_TITLE)
    ax.set_xlabel("Key")
    ax.set_ylabel("Scaled Value")

    if not data_dict:
        return pd.DataFrame(), ax

    scalar_like = all(pd.api.types.is_scalar(v) for v in data_dict.values())

    if scalar_like:
        df = pd.DataFrame({"value": pd.Series(data_dict)})
    else:
        df = pd.DataFrame(data_dict)

    try:
        df = df.apply(pd.to_numeric, errors="raise")
    except Exception as exc:
        raise ValueError("data_dict must contain only numeric values.") from exc

    if df.isna().any().any():
        raise ValueError("data_dict contains missing or irregular values.")

    scaler = MinMaxScaler()
    scaled_values = scaler.fit_transform(df)

    scaled_df = pd.DataFrame(scaled_values, index=df.index, columns=df.columns)

    if scaled_df.shape[1] == 1:
        ax.bar(scaled_df.index.astype(str), scaled_df.iloc[:, 0].values)
    else:
        for column in scaled_df.columns:
            ax.plot(scaled_df.index.astype(str), scaled_df[column].values, marker="o", label=str(column))
        ax.legend()

    return scaled_df, ax
