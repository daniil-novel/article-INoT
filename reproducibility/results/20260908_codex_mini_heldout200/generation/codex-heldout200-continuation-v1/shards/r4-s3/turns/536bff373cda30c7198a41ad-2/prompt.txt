```python
import datetime
import numpy as np
import matplotlib.pyplot as plt

# Constants
TIME_FORMAT = "%d/%m/%y %H:%M:%S.%f"

def task_func(time_strings):
    """
    Compute integer-second differences between consecutive datetime strings
    and plot them as a bar chart.

    Parameters
    ----------
    time_strings : list[str]
        Datetime strings formatted according to TIME_FORMAT.

    Returns
    -------
    matplotlib.axes.Axes
        The axes object containing the bar chart.
    """
    parsed_times = [datetime.datetime.strptime(ts, TIME_FORMAT) for ts in time_strings]
    diffs = [
        int((parsed_times[i + 1] - parsed_times[i]).total_seconds())
        for i in range(len(parsed_times) - 1)
    ]

    fig, ax = plt.subplots()
    x = np.arange(len(diffs))
    ax.bar(x, diffs)

    ax.set_xlabel("Interval")
    ax.set_ylabel("Difference (seconds)")
    ax.set_title("Differences Between Consecutive Datetimes")
    ax.set_xticks(x)
    ax.set_xticklabels([str(i) for i in x])

    return ax
```
