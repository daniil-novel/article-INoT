import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.axes import Axes
from statsmodels.tsa.arima.model import ARIMA
from typing import List, Tuple


def task_func(df: pd.DataFrame) -> Tuple[List[float], Axes]:
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input must be a pandas DataFrame.")
    if df.empty:
        raise ValueError("Input DataFrame is empty.")

    normalized_cols = {col: str(col).strip().lower() for col in df.columns}

    close_col = None
    exact_close_names = {
        "close",
        "closing_price",
        "closing price",
        "adj close",
        "adj_close",
        "adjclose",
    }

    for col, name in normalized_cols.items():
        if name in exact_close_names:
            close_col = col
            break

    if close_col is None:
        for col, name in normalized_cols.items():
            if "close" in name:
                close_col = col
                break

    if close_col is None:
        raise ValueError("Could not find a closing-price column in the input DataFrame.")

    working = pd.DataFrame({"close": pd.to_numeric(df[close_col], errors="coerce")})

    date_col = None
    if isinstance(df.index, pd.DatetimeIndex):
        working["date"] = pd.to_datetime(df.index, errors="coerce")
        date_col = "date"
    else:
        for col in df.columns:
            if col == close_col:
                continue
            col_name = normalized_cols[col]
            if pd.api.types.is_datetime64_any_dtype(df[col]):
                working["date"] = pd.to_datetime(df[col], errors="coerce")
                date_col = "date"
                break
            if col_name in {"date", "datetime", "time", "timestamp"}:
                parsed = pd.to_datetime(df[col], errors="coerce")
                if parsed.notna().any():
                    working["date"] = parsed
                    date_col = "date"
                    break

    working["close"] = working["close"].replace([float("inf"), float("-inf")], pd.NA)

    if date_col is not None:
        working = working.dropna(subset=["close", "date"]).sort_values("date")
    else:
        working = working.dropna(subset=["close"])

    if working.empty:
        raise ValueError("No valid numeric closing-price observations were found.")

    series = working["close"].astype(float)

    if len(series) < 5 or series.nunique(dropna=True) <= 1:
        forecast_values = [float(series.iloc[-1])] * 7
    else:
        try:
            model = ARIMA(series, order=(1, 1, 1))
            fitted = model.fit()
            forecast = fitted.forecast(steps=7)
            forecast_values = [float(v) for v in forecast.tolist()]
        except Exception:
            forecast_values = [float(series.iloc[-1])] * 7

    fig, ax = plt.subplots(figsize=(10, 5))

    if date_col is not None:
        hist_x = working["date"].tolist()
        ax.plot(hist_x, series.tolist(), label="Historical Close", color="tab:blue")

        last_date = pd.to_datetime(hist_x[-1])
        future_x = pd.date_range(start=last_date, periods=8, freq="D")[1:]
        ax.plot(
            future_x,
            forecast_values,
            label="7-Day Forecast",
            color="tab:orange",
            linestyle="--",
            marker="o",
        )
        ax.set_xlabel("Date")
    else:
        hist_x = list(range(len(series)))
        future_x = list(range(len(series), len(series) + 7))
        ax.plot(hist_x, series.tolist(), label="Historical Close", color="tab:blue")
        ax.plot(
            future_x,
            forecast_values,
            label="7-Day Forecast",
            color="tab:orange",
            linestyle="--",
            marker="o",
        )
        ax.set_xlabel("Time")

    ax.set_ylabel("Closing Price")
    ax.set_title("ARIMA Forecast for Next 7 Days")
    ax.grid(True, alpha=0.3)
    ax.legend()

    return forecast_values, ax
