import math
import itertools
from functools import reduce

def task_func(numbers):
    nums = list(numbers)
    if not nums:
        return 0.0

    total = 0.0
    for r in range(1, len(nums) + 1):
        for combo in itertools.combinations(nums, r):
            product = reduce(lambda a, b: a * b, combo, 1)
            total += math.log(product)
    return total
