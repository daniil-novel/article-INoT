import pandas as pd
import re
import random

def task_func(s: str, seed: int = 0) -> pd.DataFrame:
    columns = ["ID", "Quantity", "Code", "Price", "Product", "Description"]
    if not s or not s.strip():
        return pd.DataFrame(columns=columns)

    rng = random.Random(seed)
    product_names = ["Apple", "Banana", "Orange", "Pear", "Grape"]
    code_to_product = {}
    rows = []

    for line in s.splitlines():
        line = line.strip()
        if not line:
            continue

        parts = re.split(r"\s+", line, maxsplit=4)
        if len(parts) < 5:
            raise ValueError(f"Malformed line: {line!r}")

        id_str, quantity_str, code, price_str, description = parts
        code = code.strip()
        description = description.strip()

        try:
            row_id = int(id_str.strip())
            quantity = int(quantity_str.strip())
            price = int(price_str.strip())
        except ValueError as e:
            raise ValueError(f"Invalid numeric value in line: {line!r}") from e

        if code not in code_to_product:
            code_to_product[code] = rng.choice(product_names)

        rows.append(
            {
                "ID": row_id,
                "Quantity": quantity,
                "Code": code,
                "Price": price,
                "Product": code_to_product[code],
                "Description": description,
            }
        )

    return pd.DataFrame(rows, columns=columns)
