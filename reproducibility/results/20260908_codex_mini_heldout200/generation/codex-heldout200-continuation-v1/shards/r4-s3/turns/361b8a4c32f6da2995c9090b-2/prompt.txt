import pandas as pd
import numpy as np
from collections import Counter
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame) -> (Counter, plt.Axes):
    if not isinstance(df, pd.DataFrame) or df.empty:
        raise ValueError("DataFrame must be non-empty.")
    if "name" not in df.columns or "age" not in df.columns:
        raise ValueError("DataFrame must contain 'name' and 'age' columns.")

    ages = pd.to_numeric(df["age"], errors="coerce")
    if ages.isna().any():
        raise ValueError("Age values must be numeric.")

    age_values = ages.to_numpy(dtype=float)
    if not np.isfinite(age_values).all():
        raise ValueError("Age values must be finite.")

    age_values = np.floor(age_values)
    if np.any(age_values < 0):
        raise ValueError("Age must not be negative.")

    work_df = df.copy()
    work_df["_age"] = age_values.astype(int)

    dup_mask = work_df["name"].duplicated(keep=False)
    dup_ages = work_df.loc[dup_mask, "_age"]

    if dup_ages.empty:
        return Counter(), None

    age_distribution = Counter(dup_ages.tolist())

    min_age = int(dup_ages.min())
    max_age = int(dup_ages.max())
    bins = np.arange(min_age - 0.5, max_age + 0.5 + 1e-9, 1.0)

    fig, ax = plt.subplots()
    ax.hist(dup_ages.to_numpy(), bins=bins, edgecolor="black")
    ax.set_xlabel("age")
    ax.set_ylabel("count")

    return age_distribution, ax
