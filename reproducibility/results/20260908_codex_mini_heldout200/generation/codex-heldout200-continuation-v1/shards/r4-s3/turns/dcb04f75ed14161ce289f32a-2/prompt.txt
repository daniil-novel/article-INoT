import random
import bisect
import statistics
import matplotlib.pyplot as plt

def task_func(n, value):
    if not isinstance(n, int):
        raise TypeError("n must be an integer")
    if n < 0:
        raise ValueError("n must be non-negative")

    if n == 0:
        sorted_nums = []
        greater_than_avg = []
        count_ge_value = 0
    else:
        nums = [random.random() for _ in range(n)]
        avg = statistics.mean(nums)
        greater_than_avg = [x for x in nums if x > avg]
        sorted_nums = sorted(nums)
        count_ge_value = len(sorted_nums) - bisect.bisect_left(sorted_nums, value)

    plt.plot(sorted_nums)
    plt.xlabel("Index")
    plt.ylabel("Value")
    plt.title("Sorted Random Numbers")
    plt.show()

    return greater_than_avg, count_ge_value
