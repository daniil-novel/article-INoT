import json
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd

def task_func(json_data: str, key_path: list):
    if not isinstance(json_data, str) or not json_data.strip():
        raise ValueError("json_data is empty or invalid")

    try:
        data = json.loads(json_data)
    except json.JSONDecodeError as exc:
        raise ValueError("json_data is empty or corrupted") from exc

    if not isinstance(key_path, list):
        raise ValueError("key_path must be a list")

    current = data
    for key in key_path:
        if isinstance(current, dict) and key in current:
            current = current[key]
        else:
            raise KeyError(key)

    numeric_values = []

    def collect_numeric(obj):
        if obj is None or isinstance(obj, bool):
            return

        if isinstance(obj, (int, float, np.integer, np.floating)):
            if np.isfinite(obj):
                numeric_values.append(float(obj))
            return

        if isinstance(obj, np.ndarray):
            for item in obj.ravel():
                collect_numeric(item.item() if hasattr(item, "item") else item)
            return

        if isinstance(obj, pd.Series):
            for item in obj.tolist():
                collect_numeric(item)
            return

        if isinstance(obj, dict):
            for value in obj.values():
                collect_numeric(value)
            return

        if isinstance(obj, (list, tuple, set)):
            for item in obj:
                collect_numeric(item)
            return

    collect_numeric(current)

    if not numeric_values:
        raise ValueError("No numeric data found")

    fig, ax = plt.subplots(figsize=(6, 4))
    sns.boxplot(x=np.asarray(numeric_values, dtype=float), ax=ax, color="skyblue")
    ax.set_title("Boxplot of numeric data")
    ax.set_xlabel("Value")
    ax.set_yticks([])
    fig.tight_layout()
    return fig
