import matplotlib.pyplot as plt
import numpy as np

def task_func(ax, radius):
    if getattr(ax, "name", None) != "polar":
        raise TypeError("ax must be a polar Axes")
    if radius < 0:
        raise ValueError("radius must be non-negative")

    theta = np.linspace(0, 2 * np.pi, 361)
    r = np.full_like(theta, radius, dtype=float)
    ax.plot(theta, r)

    if radius == 0:
        ax.set_rticks([0])
        ax.set_ylim(0, 1)
    else:
        ax.set_rticks(np.linspace(0, radius, 5))
        ax.set_ylim(0, radius)

    return ax
