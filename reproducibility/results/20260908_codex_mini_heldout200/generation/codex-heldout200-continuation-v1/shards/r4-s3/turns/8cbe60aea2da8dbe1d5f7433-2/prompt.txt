import json
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd

def task_func(json_data: str, key_path: list):
    """
    Extract numeric values from a JSON structure at the specified key path
    and return a matplotlib Figure containing a boxplot.

    Raises:
        ValueError: If JSON is empty/corrupted, or no numeric data is found.
        KeyError: If a specified key is missing or the path cannot be traversed.
    """
    if json_data is None or not isinstance(json_data, str) or not json_data.strip():
        raise ValueError("JSON data is empty or invalid.")

    try:
        data = json.loads(json_data)
    except json.JSONDecodeError as e:
        raise ValueError("JSON data is corrupted or invalid.") from e

    if key_path is None:
        key_path = []

    current = data
    for key in key_path:
        if not isinstance(current, dict):
            raise KeyError(f"Cannot traverse key '{key}' through a non-dict object.")
        if key not in current:
            raise KeyError(f"Key '{key}' not found.")
        current = current[key]

    def collect_numeric(obj):
        values = []

        if isinstance(obj, dict):
            for v in obj.values():
                values.extend(collect_numeric(v))
        elif isinstance(obj, (list, tuple)):
            for item in obj:
                values.extend(collect_numeric(item))
        else:
            is_number = isinstance(obj, (int, float, np.number)) and not isinstance(obj, (bool, np.bool_))
            if is_number:
                val = float(obj)
                if np.isfinite(val):
                    values.append(val)

        return values

    values = collect_numeric(current)

    if not values:
        raise ValueError("No numeric data found at the specified path.")

    fig, ax = plt.subplots(figsize=(6, 4))
    sns.boxplot(x=values, ax=ax, color="skyblue")
    ax.set_title("Boxplot of Extracted Numeric Data")
    ax.set_xlabel("Values")
    ax.set_yticks([])

    plt.tight_layout()
    return fig
