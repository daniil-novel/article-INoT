import random
import math
import matplotlib.pyplot as plt

def task_func(points_count=1000, radius=1):
    if isinstance(points_count, bool) or not isinstance(points_count, int):
        raise ValueError("points_count must be a non-negative integer")
    if points_count < 0:
        raise ValueError("points_count must be non-negative")

    if isinstance(radius, bool) or not isinstance(radius, (int, float)):
        raise ValueError("radius must be a non-negative number")
    if radius < 0:
        raise ValueError("radius must be non-negative")

    xs = []
    ys = []

    for _ in range(points_count):
        theta = 2 * math.pi * random.random()
        r = radius * math.sqrt(random.random())
        xs.append(r * math.cos(theta))
        ys.append(r * math.sin(theta))

    fig, ax = plt.subplots()
    ax.scatter(xs, ys)
    ax.set_aspect('equal')

    return ax
