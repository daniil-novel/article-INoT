import pandas as pd
import numpy as np
from collections import Counter
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame) -> (Counter, plt.Axes):
    if df.empty:
        raise ValueError("DataFrame is empty")

    # Convert ages to numeric, floor float ages down to integers
    age_numeric = pd.to_numeric(df["age"], errors="raise")
    age_floored = pd.Series(np.floor(age_numeric.astype(float)), index=df.index).astype(int)

    # Reject negative ages
    if (age_floored < 0).any():
        raise ValueError("Age must not be negative")

    # Find all rows whose names are duplicated
    duplicate_mask = df["name"].duplicated(keep=False)
    duplicate_ages = age_floored.loc[duplicate_mask]

    # If no duplicates, return empty counter and None
    if duplicate_ages.empty:
        return Counter(), None

    age_distribution = Counter(duplicate_ages.tolist())

    # Integer-centered bins: [min-0.5, min+0.5, ..., max+0.5]
    min_age = int(duplicate_ages.min())
    max_age = int(duplicate_ages.max())
    bins = np.arange(min_age - 0.5, max_age + 1.5, 1)

    fig, ax = plt.subplots()
    sns.histplot(x=duplicate_ages, bins=bins, ax=ax, stat="count")
    ax.set_xlabel("Age")
    ax.set_ylabel("Count")

    return age_distribution, ax
