import random
import string
import hashlib
import time

def task_func(data_dict: dict, seed=0) -> dict:
    rng = random.Random(seed)
    salt = "".join(rng.choice(string.ascii_lowercase) for _ in range(5))
    
    result = dict(data_dict)
    result["a"] = 1
    
    for key, value in list(result.items()):
        result[key] = hashlib.sha256((str(value) + salt).encode()).hexdigest()
    
    result["timestamp"] = int(time.time())
    return result
