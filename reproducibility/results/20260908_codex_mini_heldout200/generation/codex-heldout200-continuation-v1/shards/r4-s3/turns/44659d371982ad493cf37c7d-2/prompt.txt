import matplotlib.pyplot as plt
import numpy as np
from sklearn.cluster import KMeans

# Constants for configuration
RANGE = 100
SIZE = 1000
CLUSTERS = 5

def task_func():
    if RANGE <= 0:
        raise ValueError("RANGE must be positive.")
    if SIZE <= 0:
        raise ValueError("SIZE must be positive.")
    if CLUSTERS <= 0:
        raise ValueError("CLUSTERS must be positive.")
    if SIZE < CLUSTERS:
        raise ValueError("SIZE must be greater than or equal to CLUSTERS.")

    rng = np.random.default_rng()
    points = rng.uniform(0, RANGE, size=(SIZE, 2))

    model = KMeans(n_clusters=CLUSTERS, random_state=42, n_init=10)
    labels = model.fit_predict(points)

    plt.figure(figsize=(8, 6))
    cmap = plt.cm.get_cmap("tab10", CLUSTERS)

    for cluster_id in range(CLUSTERS):
        cluster_points = points[labels == cluster_id]
        plt.scatter(
            cluster_points[:, 0],
            cluster_points[:, 1],
            s=15,
            color=cmap(cluster_id),
            label=f"Cluster {cluster_id + 1}",
            alpha=0.7,
        )

    plt.scatter(
        model.cluster_centers_[:, 0],
        model.cluster_centers_[:, 1],
        s=200,
        c="black",
        marker="X",
        label="Centroids",
        edgecolors="white",
        linewidths=1.0,
    )

    plt.title("KMeans Clustering of Random 2D Points")
    plt.xlabel("X")
    plt.ylabel("Y")
    plt.legend()
    plt.tight_layout()
    plt.show()

    return points, model
