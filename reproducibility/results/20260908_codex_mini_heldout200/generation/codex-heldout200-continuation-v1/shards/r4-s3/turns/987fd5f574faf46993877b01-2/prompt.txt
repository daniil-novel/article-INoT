import pandas as pd
from itertools import combinations

# Constants
MIN_PERCENTAGE = 0.75

def task_func(data, cols, percentage):
    corr_combinations = []

    # Validate inputs
    if not isinstance(data, pd.DataFrame):
        return corr_combinations

    if percentage is None:
        percentage = MIN_PERCENTAGE

    try:
        percentage = float(percentage)
    except (TypeError, ValueError):
        percentage = MIN_PERCENTAGE

    # Correlation is always in [-1, 1]
    if percentage < 0:
        percentage = 0.0
    if percentage > 1:
        return corr_combinations

    # Normalize and deduplicate columns while preserving order
    try:
        unique_cols = list(dict.fromkeys(cols))
    except TypeError:
        return corr_combinations

    # Keep only existing numeric columns
    usable_cols = []
    for col in unique_cols:
        if col in data.columns:
            series = data[col]
            if isinstance(series, pd.Series) and pd.api.types.is_numeric_dtype(series):
                usable_cols.append(col)

    if len(usable_cols) < 2:
        return corr_combinations

    # Find qualifying pairs
    for col1, col2 in combinations(usable_cols, 2):
        corr_value = data[col1].corr(data[col2])
        if pd.notna(corr_value) and abs(corr_value) > percentage:
            corr_combinations.append((col1, col2))

    return corr_combinations
