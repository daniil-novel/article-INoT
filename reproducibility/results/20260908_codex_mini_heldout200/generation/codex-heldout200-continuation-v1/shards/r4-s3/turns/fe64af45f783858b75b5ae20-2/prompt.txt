import requests
import pandas as pd

def task_func(url: str) -> pd.DataFrame:
    try:
        response = requests.get(url, timeout=5)
        response.raise_for_status()
    except requests.RequestException as exc:
        raise SystemError(f"Failed to fetch data from {url}: {exc}") from exc

    try:
        data = response.json()
        return pd.DataFrame(data)
    except ValueError as exc:
        raise ValueError(f"Invalid JSON data or incompatible structure from {url}: {exc}") from exc
