```python
import re
import string
# Constants
PUNCTUATION = string.punctuation

def task_func(text):
    words_count = len(re.findall(r"\b\w+\b", text))
    punctuation_count = sum(1 for char in text if char in PUNCTUATION)
    return (words_count, punctuation_count)
```
