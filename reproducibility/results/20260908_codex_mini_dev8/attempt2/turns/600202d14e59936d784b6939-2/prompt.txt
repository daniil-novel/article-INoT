import numpy as np
import random
from sklearn.preprocessing import MinMaxScaler

def task_func(list_of_lists, seed=42):
    random.seed(seed)

    filled_lists = []
    lengths = []

    for inner in list_of_lists:
        inner_list = list(inner)
        if len(inner_list) == 0:
            inner_list = [random.randint(0, 100) for _ in range(5)]
        filled_lists.append(inner_list)
        lengths.append(len(inner_list))

    if not filled_lists:
        return []

    flat_values = [value for inner in filled_lists for value in inner]
    if not flat_values:
        return [[] for _ in filled_lists]

    data = np.array(flat_values, dtype=float).reshape(-1, 1)
    scaler = MinMaxScaler()
    scaled_flat = scaler.fit_transform(data).flatten().tolist()

    result = []
    idx = 0
    for length in lengths:
        result.append([float(v) for v in scaled_flat[idx:idx + length]])
        idx += length

    return result
