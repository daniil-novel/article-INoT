import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(df: pd.DataFrame):
    # Select only numeric columns
    numeric_df = df.select_dtypes(include=np.number).copy()

    # Validate feasibility for PCA with 2 components
    if numeric_df.shape[1] < 2:
        raise ValueError("PCA requires at least two numeric columns.")
    if numeric_df.shape[0] < 2:
        raise ValueError("PCA requires at least two rows.")

    # Replace missing values with each column's average
    for col in numeric_df.columns:
        col_mean = numeric_df[col].mean()
        if pd.isna(col_mean):
            # If a column is entirely missing, fall back to 0 to avoid unresolved NaNs
            col_mean = 0.0
        numeric_df[col] = numeric_df[col].fillna(col_mean)

    # Run PCA
    pca = PCA(n_components=2)
    transformed = pca.fit_transform(numeric_df)

    # Build output DataFrame
    pca_df = pd.DataFrame(
        transformed,
        index=df.index,
        columns=["principal component 1", "principal component 2"],
    )

    # Create scatter plot
    ax = sns.scatterplot(
        data=pca_df,
        x="principal component 1",
        y="principal component 2",
    )
    ax.set_xlabel("principal component")
    ax.set_ylabel("principal component 2")

    return pca_df, ax
