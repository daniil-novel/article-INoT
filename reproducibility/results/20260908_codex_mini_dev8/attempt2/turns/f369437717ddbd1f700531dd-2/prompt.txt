import pandas as pd
import logging
# Set up basic configuration for logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def task_func(sheet_name, excel_file_location="test.xlsx", csv_file_location="test.csv"):
    """
    Read data from an Excel spreadsheet, convert it to CSV, then calculate the sum of each column in the CSV.

    Args:
        sheet_name: Name of the Excel sheet to read.
        excel_file_location: Path to the Excel file.
        csv_file_location: Path where the CSV file will be written.

    Returns:
        dict: A dictionary with the sum of each column.

    Raises:
        FileNotFoundError: If the Excel file does not exist at the specified path.
        ValueError: If the specified sheet name is not found in the Excel file.
        ValueError: If the sheet contains non-numeric data.
    """
    logging.info("Starting task_func with excel_file_location=%s, csv_file_location=%s, sheet_name=%s",
                 excel_file_location, csv_file_location, sheet_name)

    # Check that the Excel file exists
    try:
        with open(excel_file_location, "rb"):
            pass
    except FileNotFoundError:
        logging.error("Excel file not found: %s", excel_file_location)
        raise

    # Verify the sheet exists and read it
    try:
        excel_file = pd.ExcelFile(excel_file_location)
    except FileNotFoundError:
        logging.error("Excel file not found while opening: %s", excel_file_location)
        raise

    if sheet_name not in excel_file.sheet_names:
        logging.error("Sheet '%s' not found in Excel file: %s", sheet_name, excel_file_location)
        raise ValueError(f"Sheet '{sheet_name}' not found in Excel file.")

    try:
        df = excel_file.parse(sheet_name=sheet_name)
    except ValueError:
        logging.error("Failed to parse sheet '%s' from file: %s", sheet_name, excel_file_location)
        raise

    logging.info("Excel sheet read successfully with shape %s", df.shape)

    # Ensure all data is numeric
    try:
        numeric_df = df.apply(pd.to_numeric, errors="raise")
    except Exception as exc:
        logging.error("Non-numeric data found in sheet '%s': %s", sheet_name, exc)
        raise ValueError("The Excel sheet must contain only numerical data.") from exc

    # Write the numeric data to CSV
    numeric_df.to_csv(csv_file_location, index=False)
    logging.info("Data written to CSV at %s", csv_file_location)

    # Read the CSV back and calculate column sums
    csv_df = pd.read_csv(csv_file_location)
    sums = csv_df.sum(numeric_only=True).to_dict()

    logging.info("Column sums calculated successfully: %s", sums)
    return sums
