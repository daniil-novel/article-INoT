## Recommended exception handling strategy

Handle errors in this order:

1. `urllib.error.URLError`
   - Return: `URL Error: ...`
2. `zipfile.BadZipFile`
   - Return: ZIP corruption / invalid archive message
3. `OSError` / `IOError`
   - Covers filesystem issues:
     - permission denied
     - missing directories
     - extraction failures
     - deletion failures
     - disk errors
4. Optional broad `Exception`
   - As a final safeguard, return a generic unexpected error message

### URL / network issues

### ZIP issues

import urllib.request
import zipfile
import os
import urllib.error


def task_func(
    url: str,
    save_path: str = "downloaded_file.zip",
    extract_path: str = "extracted_files",
) -> str:
    """
    Download a ZIP file from `url`, extract it into `extract_path`,
    delete the downloaded ZIP, and return `extract_path` on success.

    On failure, returns a descriptive error string.
    """

    def _best_effort_cleanup() -> None:
        try:
            if os.path.exists(save_path):
                os.remove(save_path)
        except OSError:
            pass

    def _safe_extract(zip_obj: zipfile.ZipFile) -> None:
        abs_extract_root = os.path.abspath(extract_path)

        for member in zip_obj.infolist():
            member_target = os.path.abspath(os.path.join(extract_path, member.filename))

            # Prevent path traversal outside the extraction directory.
            if member_target != abs_extract_root and not member_target.startswith(abs_extract_root + os.sep):
                raise OSError(f"Unsafe path detected in ZIP archive: {member.filename}")

            if member.is_dir():
                os.makedirs(member_target, exist_ok=True)
                continue

            parent_dir = os.path.dirname(member_target)
            if parent_dir:
                os.makedirs(parent_dir, exist_ok=True)

            with zip_obj.open(member, "r") as source, open(member_target, "wb") as target:
                while True:
                    chunk = source.read(8192)
                    if not chunk:
                        break
                    target.write(chunk)

    try:
        os.makedirs(extract_path, exist_ok=True)
    except OSError as e:
        return f"File I/O Error: {e}"

    try:
        urllib.request.urlretrieve(url, save_path)
    except urllib.error.URLError as e:
        _best_effort_cleanup()
        reason = getattr(e, "reason", e)
        return f"URL Error: {reason}"
    except OSError as e:
        _best_effort_cleanup()
        return f"File I/O Error: {e}"

    try:
        with zipfile.ZipFile(save_path, "r") as zip_obj:
            _safe_extract(zip_obj)
    except zipfile.BadZipFile:
        _best_effort_cleanup()
        return "ZIP Error: corrupted or invalid ZIP file"
    except OSError as e:
        _best_effort_cleanup()
        return f"File I/O Error: {e}"
    except Exception as e:
        _best_effort_cleanup()
        return f"Unexpected Error: {e}"

    try:
        os.remove(save_path)
    except OSError as e:
        return f"File I/O Error: {e}"

    return extract_path
