```python
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

plt.switch_backend("Agg")


def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    """
    Generate a random sample from a normal distribution, compute skewness and kurtosis,
    and optionally save a histogram and QQ plot.

    Returns:
        tuple[float, float, list]: (skewness, kurtosis, plot_paths)
    """
    if std_dev <= 0:
        raise ValueError("std_dev must be positive.")

    sample_size = 1000
    rng = np.random.default_rng()
    sample = rng.normal(loc=mean, scale=std_dev, size=sample_size)

    skewness = float(stats.skew(sample))
    kurtosis = float(stats.kurtosis(sample))

    plot_paths = []

    # Histogram
    fig, ax = plt.subplots(figsize=(8, 5))
    ax.hist(sample, bins=30, density=True, alpha=0.7, color="steelblue", edgecolor="black")

    x_min, x_max = sample.min(), sample.max()
    x = np.linspace(x_min, x_max, 400)
    pdf = stats.norm.pdf(x, loc=mean, scale=std_dev)
    ax.plot(x, pdf, "r-", linewidth=2, label="Normal PDF")

    ax.set_title("Histogram of Sample")
    ax.set_xlabel("Value")
    ax.set_ylabel("Density")
    ax.grid(True, alpha=0.3)
    ax.legend()

    if save_plots:
        hist_path = "histogram.png"
        fig.savefig(hist_path, dpi=150, bbox_inches="tight")
        plot_paths.append(hist_path)
    plt.close(fig)

    # QQ plot
    fig, ax = plt.subplots(figsize=(8, 5))
    stats.probplot(sample, dist="norm", plot=ax)
    ax.set_title("QQ Plot of Sample")

    if save_plots:
        qq_path = "qq_plot.png"
        fig.savefig(qq_path, dpi=150, bbox_inches="tight")
        plot_paths.append(qq_path)
    plt.close(fig)

    if not save_plots:
        plot_paths = []

    return skewness, kurtosis, plot_paths
```
