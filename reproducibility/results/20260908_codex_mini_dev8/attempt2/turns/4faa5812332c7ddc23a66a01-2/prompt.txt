import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(s1, s2):
    """
    Create a swarm plot comparing two series and highlight their intersecting
    values with red dashed horizontal lines.

    Parameters
    ----------
    s1, s2 : pandas.Series or series-like
        Input data.

    Returns
    -------
    ax : matplotlib.axes.Axes
        The Axes object containing the plot.
    intersection_count : int
        Number of unique intersecting data points between s1 and s2.
    """
    # Normalize inputs to numeric Series; non-numeric values become NaN and are dropped.
    s1 = pd.Series(s1)
    s2 = pd.Series(s2)

    s1_num = pd.to_numeric(s1, errors="coerce").dropna()
    s2_num = pd.to_numeric(s2, errors="coerce").dropna()

    # Compute unique intersection count.
    intersection_values = sorted(set(s1_num.unique()) & set(s2_num.unique()))
    intersection_count = len(intersection_values)

    # Prepare long-form data for swarmplot.
    df = pd.DataFrame(
        {
            "value": pd.concat([s1_num, s2_num], ignore_index=True),
            "series": (["s1"] * len(s1_num)) + (["s2"] * len(s2_num)),
        }
    )

    fig, ax = plt.subplots()

    if not df.empty:
        sns.swarmplot(
            data=df,
            x="series",
            y="value",
            order=["s1", "s2"],
            ax=ax,
            color="#4C72B0",
            size=6,
        )

        # Highlight intersection values with red dashed horizontal lines.
        for value in intersection_values:
            ax.axhline(value, color="red", linestyle="--", linewidth=1, alpha=0.7, zorder=0)
    else:
        ax.set_xticks([0, 1])
        ax.set_xticklabels(["s1", "s2"])

    ax.set_title("Swarm Plot of Two Series with Intersections Highlighted")
    ax.set_xlabel("Series")
    ax.set_ylabel("Value")

    return ax, intersection_count
