import re
import os
from pathlib import Path
import glob


def task_func(directory_path: str, regex_pattern: str = r'\(.+?\)|\w') -> dict:
    """
    Extract matches from each text file in a directory using the provided regex.

    Returns:
        dict: {file_name: [matches_in_order]}
    """
    base_dir = Path(directory_path)
    if not base_dir.exists() or not base_dir.is_dir():
        return {}

    try:
        pattern = re.compile(regex_pattern)
    except re.error:
        raise

    result = {}

    for file_path_str in sorted(glob.glob(str(base_dir / "*"))):
        file_path = Path(file_path_str)
        if not file_path.is_file():
            continue

        try:
            with file_path.open("r", encoding="utf-8", errors="ignore") as f:
                text = f.read()
        except OSError:
            result[file_path.name] = []
            continue

        result[file_path.name] = [match.group(0) for match in pattern.finditer(text)]

    return result
