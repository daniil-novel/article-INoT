import numpy as np
import matplotlib.pyplot as plt
from scipy import stats

def task_func(mean=123456.908, std_dev=1.2, save_plots=False):
    """
    Generate a normal random sample, compute skewness and kurtosis,
    and optionally save a histogram and QQ plot.

    Returns:
        tuple: (skewness: float, kurtosis: float, plot_paths: list[str])
    """
    import os

    mean = float(mean)
    std_dev = float(std_dev)

    if std_dev <= 0:
        raise ValueError("std_dev must be positive.")

    n = 1000
    sample = np.random.normal(loc=mean, scale=std_dev, size=n)

    skewness = float(stats.skew(sample, bias=False))
    kurtosis = float(stats.kurtosis(sample, bias=False))  # excess kurtosis

    plot_paths = []

    # Histogram
    fig1, ax1 = plt.subplots(figsize=(8, 5))
    ax1.hist(sample, bins=30, density=True, alpha=0.7, color="steelblue", edgecolor="black")

    # Overlay theoretical normal PDF for reference
    x = np.linspace(sample.min(), sample.max(), 300)
    pdf = stats.norm.pdf(x, loc=mean, scale=std_dev)
    ax1.plot(x, pdf, "r-", linewidth=2, label="Normal PDF")
    ax1.set_title("Histogram of Normal Sample")
    ax1.set_xlabel("Value")
    ax1.set_ylabel("Density")
    ax1.legend(loc="best")
    fig1.tight_layout()

    # QQ plot
    fig2, ax2 = plt.subplots(figsize=(8, 5))
    stats.probplot(sample, dist="norm", plot=ax2)
    ax2.set_title("QQ Plot of Normal Sample")
    fig2.tight_layout()

    if save_plots:
        hist_path = os.path.abspath("histogram.png")
        qq_path = os.path.abspath("qq_plot.png")

        fig1.savefig(hist_path, dpi=150, bbox_inches="tight")
        fig2.savefig(qq_path, dpi=150, bbox_inches="tight")

        plot_paths = [hist_path, qq_path]

    plt.close(fig1)
    plt.close(fig2)

    return skewness, kurtosis, plot_paths
