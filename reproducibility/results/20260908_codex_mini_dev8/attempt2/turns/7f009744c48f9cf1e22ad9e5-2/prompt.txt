import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def task_func(s1, s2):
    def _to_series(obj):
        if isinstance(obj, pd.Series):
            return obj.copy()
        try:
            return pd.Series(obj)
        except Exception:
            return pd.Series([obj])

    s1 = _to_series(s1).dropna()
    s2 = _to_series(s2).dropna()

    common_values = pd.Index(s1.unique()).intersection(pd.Index(s2.unique()), sort=False)
    intersection_count = len(common_values)

    plot_df = pd.DataFrame(
        {
            "Series": ["s1"] * len(s1) + ["s2"] * len(s2),
            "Value": pd.concat([s1.reset_index(drop=True), s2.reset_index(drop=True)], ignore_index=True),
        }
    )

    fig, ax = plt.subplots()

    if not plot_df.empty:
        sns.swarmplot(data=plot_df, x="Series", y="Value", order=["s1", "s2"], ax=ax)
    else:
        ax.set_xticks([0, 1])
        ax.set_xticklabels(["s1", "s2"])

    for value in common_values:
        ax.axhline(y=value, color="red", linestyle="--", linewidth=1, alpha=0.7)

    ax.set_xlabel("Series")
    ax.set_ylabel("Value")
    ax.set_title("Swarm plot with intersections highlighted")

    return ax, intersection_count
